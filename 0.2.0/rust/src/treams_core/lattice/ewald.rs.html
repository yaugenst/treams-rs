35/// The Ewald parts selected by `evaluation` with split `eta`. Both sums run over shells of
36/// reduced bases, and the reciprocal sum over the Bloch vector reduced into a cell
37/// of the reduced reciprocal basis, so their results do not depend on the given basis
38/// or on the Bloch vector's cell.
39pub(super) fn ewald<const N: usize>(
40    wave: Family,
41    lattice: &BlochLattice,
42    mut inputs: Inputs<N>,
43    eta: Complex,
44    evaluation: Evaluation,
45    shared: Option<&SphericalCache<N>>,
46) -> Result<Jet<N>> {
47    let CellReduction {
48        phase,
49        direct_rows: direct,
50        reciprocal_rows: rows,
51        kpar,
52        kpar_reduced: reduced,
53        given_shift,
54    } = reduce_to_cell(lattice, &mut inputs);
55    let inputs = &inputs;
56    let dim = inputs.dim;
57    let symmetry = Symmetry::of(wave, inputs, &direct, &kpar);
58    if symmetry.vanishes && (N == 0 || matches!(evaluation, Evaluation::EtaDerivative)) {
59        return Ok(Jet::default());
60    }
61    // Only splits below every automatic one check the cancellation of their parts.
62    let small = below_automatic(eta);
63    let checked = symmetry.checked(inputs, evaluation);
64    let real_limit = real_shells(lattice, inputs.k.value, eta);
65    // The shells before which neither part may stop (see `peak_shells`): the real-space
66    // terms fall like `exp(-Re((k eta)^2) |R|^2 / 2)`, the reciprocal ones like
67    // `exp(-Re(1 / (k eta)^2) |q + G|^2 / 2)`.
68    let square = (inputs.k.value * eta).powi(2);
69    let real_floor = peak_shells(
70        wave,
71        dim,
72        N > 0,
73        square.re,
74        norm(inputs.r.map(|r| r.value.re)),
75        (&lattice.reciprocal, lattice.direct_reduction.as_ref()),
76    );
77    let reciprocal_floor = peak_shells(
78        wave,
79        dim,
80        N > 0,
81        square.inv().re,
82        norm(kpar.map(|q| q.value.re)),
83        (&lattice.direct, lattice.reciprocal_reduction.as_ref()),
84    );
85    // Complete sums may settle their real-space shells (see `Shells::sum`) and fail
86    // early (see `EARLY_FAILURE_LOSS`).
87    let settle = small && matches!(evaluation, Evaluation::Part(SumPart::Full));
88    let early = settle && probes::early_failure_enabled();
89    let automatic_split = AutomaticSplit {
90        wave,
91        k: inputs.k.value,
92        lattice,
93        shift: given_shift,
94        evaluation,
95    };
96    let monitor = SmallSplitMonitor::new(
97        (small && !matches!(evaluation, Evaluation::EtaDerivative))
98            .then(|| SmallSplitKambe::new(eta)),
99        early,
100        automatic_split,
101        &checked,
102    );
103    let real_summands = RealSummands {
104        wave,
105        inputs,
106        direct: &direct,
107        eta,
108        evaluation,
109        monitor: &monitor,
110        shared: if small { None } else { shared },
111        index: Cell::new(0),
112    };
113    let real_summand = |n, bounds: &mut Rounding<N>| real_summands.at(n, bounds);
114    let real_part = || {
115        Shells::sum(
116            dim,
117            [real_limit, real_floor],
118            &mut { real_summand },
119            settle,
120            small,
121        )
122    };
123    if !evaluation.reciprocal() {
124        let real = real_part()?;
125        let sum = phase * real.sum;
126        if small {
127            cancellation(&[&real], &Jet::default(), 0.0, &phase, &sum, &checked)?;
128        }
129        return Ok(symmetry.zeroed(sum));
130    }
131    // 1D spherical sums may take the spectral series of the complete sum (less the
132    // real part for the reciprocal part alone), evaluated once.
133    let sw1d = match wave {
134        Family::Spherical { l, m } if dim == 1 && probes::spectral_series_enabled() => Some((l, m)),
135        _ => None,
136    };
137    let series_cell = OnceCell::new();
138    let series = || {
139        *series_cell.get_or_init(|| {
140            let (mut series, mut bound) = spectral_sw1d(
141                sw1d?,
142                inputs.k,
143                inputs.r,
144                kpar[0],
145                rows[0][0],
146                inputs.measure,
147            )?;
148            if !evaluation.real() {
149                let real = real_part().ok()?.sum;
150                series -= real;
151                bound += Rounding::relative(&real, REAL_ULPS);
152            }
153            Some((phase * series, bound.times(&phase)))
154        })
155    };
156    // Far off the axis the Ewald sum cancels (see `sum`), so a series that keeps its
157    // accuracy is taken without it.
158    let rho = (inputs.r[0].value.re).hypot(inputs.r[1].value.re);
159    if sw1d.is_some()
160        && inputs.k.value.norm() * rho * eta.norm() >= SPECTRAL_FIRST_W
161        && let Some((series, bound)) = series()
162        && f64::EPSILON * bound.value <= SPECTRAL_SWITCH_LOSS * series.value.norm().max(1.0)
163    {
164        return Ok(series);
165    }
166    let real = if evaluation.real() {
167        real_part().map(Some)
168    } else {
169        Ok(None)
170    };
171    let [maximum, extension] =
172        reciprocal_shells(lattice, inputs.k.value, eta, kpar.map(|q| q.value.re));
173    // A 1D spherical sum whose reciprocal part passes `extension` shells takes its
174    // spectral series where that is accurate. Calibrated: for a degree-3 chain the series
175    // lies 2e-15 from the sum at the automatic split, and the Ewald sum continued to the
176    // larger radius 3e-13.
177    let maximum = if sw1d.is_some() { extension } else { maximum };
178    // Absolute rounding error bounds of the reciprocal part in units of epsilon.
179    let mut rounding = Rounding::default();
180    let split = Split::new(inputs.k.value, eta);
181    let reciprocal_summands = ReciprocalSummands {
182        wave,
183        inputs,
184        rows: &rows,
185        kpar: &kpar,
186        kpar_reduced: reduced,
187        eta,
188        split: &split,
189        shared: if !small && dim == 2 && inputs.r[2].value == Complex::default() {
190            shared
191        } else {
192            None
193        },
194        index: Cell::new(0),
195    };
196    let mut reciprocal_summand =
197        |n: [i64; 3], _: &mut Rounding<N>| reciprocal_summands.at(n, &mut rounding);
198    let reciprocal = Shells::sum(
199        dim,
200        [maximum, reciprocal_floor],
201        &mut reciprocal_summand,
202        false,
203        small,
204    );
205    let self_term = if inputs.r.iter().all(|r| r.value == Complex::default()) {
206        self_term(wave, inputs.k, inputs.r, eta)
207    } else {
208        Jet::default()
209    };
210    let total = |real: &Option<Shells<N>>, reciprocal: &Shells<N>| {
211        phase * (real.map_or_else(Jet::default, |real| real.sum) + reciprocal.sum + self_term)
212    };
213    // Below every automatic split both parts continue to `SHELL_TOLERANCE` of the
214    // complete sum, which fails where they cancel beyond use (and goes to
215    // `prefer_spectral_sw1d` then).
216    let mut complete = |mut real: Option<Shells<N>>, mut reciprocal: Shells<N>| {
217        let scale = Some(total(&real, &reciprocal).norm().max(1.0));
218        let mut tail = 0.0;
219        if !reciprocal.extend(dim, extension, &mut reciprocal_summand, scale)? {
220            tail += reciprocal.tail();
221        }
222        if let Some(real) = &mut real
223            && !real.extend(dim, real_limit, &mut { real_summand }, scale)?
224        {
225            tail += real.tail();
226        }
227        let sum = total(&real, &reciprocal);
228        let parts: Vec<&Shells<N>> = real.iter().chain([&reciprocal]).collect();
229        let rounding = cancellation(&parts, &self_term, tail, &phase, &sum, &checked)?;
230        Ok((sum, rounding, tail))
231    };
232    // The rounding the cancelling parts predict and what their last shells may leave
233    // out, for `prefer_spectral_sw1d`.
234    let (mut predicted, mut truncation) = (Rounding::default(), 0.0);
235    let settled = matches!(&real, Ok(Some(real)) if real.settled);
236    let result = match (real, reciprocal) {
237        (Err(error), _) | (Ok(_), Err(error)) => Err(error),
238        (Ok(real), Ok(reciprocal)) if small => {
239            complete(real, reciprocal).map(|(sum, rounding, tail)| {
240                (predicted, truncation) = (rounding, tail);
241                sum
242            })
243        }
244        (Ok(real), Ok(reciprocal)) => Ok(total(&real, &reciprocal)),
245    };
246    let result = if sw1d.is_none() {
247        result.map(|sum| symmetry.zeroed(sum))
248    } else {
249        let mut rounding = rounding.times(&phase);
250        rounding += predicted;
251        rounding.value += truncation / f64::EPSILON;
252        prefer_spectral_sw1d(result, rounding, series, inputs.k.value)
253            .map(|sum| symmetry.zeroed(sum))
254    };
255    if !settled {
256        return result;
257    }
258    verify_settled(result?, predicted.value, automatic_split)
259}

261/// Ewald samples of one spherical degree, shared by its orders. Each order still
262/// follows its own shell convergence test; the shell iterator visits the same prefix
263/// of lattice points, so later orders reuse the samples already visited and append
264/// only the farther points they need. The cache lives for one degree evaluation.
265#[derive(Default)]
266pub(super) struct SphericalCache<const N: usize> {
267    samples: RefCell<Vec<SphericalRealSample<N>>>,
268    plane_samples: RefCell<Vec<SphericalReciprocalSample<N>>>,
269}
270
271/// A bounded prefix covers ordinary Ewald sums without retaining a large 3D cube
272/// at the shell limit: at most 1.4 MiB per degree for the largest (16-slot) jets.
273/// Farther samples stream through the scalar arithmetic instead.
274const SHARED_REAL_POINTS: usize = 1024;

325/// What the real-space terms of one sum read.
326struct RealSummands<'a, const N: usize> {
327    wave: Family,
328    inputs: &'a Inputs<N>,
329    /// The reduced direct rows.
330    direct: &'a [[Jet<N>; 3]; 3],
331    eta: Complex,
332    evaluation: Evaluation,
333    monitor: &'a SmallSplitMonitor<'a, N>,
334    shared: Option<&'a SphericalCache<N>>,
335    index: Cell<usize>,
336}
337
338impl<const N: usize> RealSummands<'_, N> {
339    /// The real-space term at the lattice point with the coordinates `n`, adding its
340    /// rounding bounds below every automatic split to `bounds`. Always inlined: it is
341    /// the body of the term loop of [`Shells::sum`].
342    #[allow(clippy::inline_always)]
343    #[inline(always)]
344    fn at(&self, n: [i64; 3], bounds: &mut Rounding<N>) -> Result<Jet<N>> {
345        let (wave, inputs, direct, eta, evaluation, monitor) = (
346            self.wave,
347            self.inputs,
348            self.direct,
349            self.eta,
350            self.evaluation,
351            self.monitor,
352        );
353        if let (Some(shared), Family::Spherical { l, m }) = (self.shared, wave) {
354            return Ok(self.spherical_at(shared, l, m, n));
355        }
356        let point = inputs.point(direct, n);
357        let shift = inputs.image(&point);
358        if shift.iter().all(|r| r.value == Complex::default()) {
359            return Ok(Jet::default());
360        }
361        probes::count_real_term();
362        // Below every automatic split only: the rounding of the largest term so far and
363        // the bound of this one.
364        let mut small = monitor
365            .kambe
366            .as_ref()
367            .map(|kambe| (kambe, monitor.largest.get(), Rounding::default()));
368        let kambe = match (&mut small, evaluation) {
369            (Some((kambe, current, bound)), _) => RealKambe::SmallSplit {
370                kambe,
371                largest: current,
372                bound,
373            },
374            (None, Evaluation::EtaDerivative) => RealKambe::EtaDerivative,
375            (None, _) => RealKambe::Plain,
376        };
377        let term = real_term(wave, inputs.k, shift, eta, kambe);
378        let phase = (Complex::i() * inputs.phase(&point)).exp();
379        let Some((_, current, bound)) = small else {
380            return Ok(term * phase);
381        };
382        monitor
383            .largest
384            .set(current.max(Rounding::relative(&term, 1.0)));
385        let (term, bound) = (term * phase, bound.times(&phase));
386        *bounds += bound;
387        monitor.add_to_running(&term, bound)?;
388        Ok(term)
389    }
390
391    /// The ordinary spherical term, preserving its operation order while sharing its
392    /// geometry, Bloch phase and Kambe integral with the other orders of this degree.
393    fn spherical_at(&self, shared: &SphericalCache<N>, l: i32, m: i32, n: [i64; 3]) -> Jet<N> {
394        let index = self.index.replace(self.index.get() + 1);
395        let mut samples = shared.samples.borrow_mut();
396        let sample = if let Some(&sample) = samples.get(index) {
397            sample
398        } else {
399            let point = self.inputs.point(self.direct, n);
400            let shift = self.inputs.image(&point);
401            let radial = if shift.iter().all(|r| r.value == Complex::default()) {
402                Jet::default()
403            } else {
404                spherical_radial(l, self.inputs.k, shift, self.eta)
405            };
406            let sample = SphericalRealSample {
407                shift,
408                radial,
409                phase: (Complex::i() * self.inputs.phase(&point)).exp(),
410            };
411            if index < SHARED_REAL_POINTS {
412                samples.push(sample);
413            }
414            sample
415        };
416        if sample.shift.iter().all(|r| r.value == Complex::default()) {
417            return Jet::default();
418        }
419        probes::count_real_term();
420        spherical_from_radial(l, m, self.inputs.k, sample.shift, sample.radial) * sample.phase
421    }
422}

424/// What the reciprocal-space terms of one sum read.
425struct ReciprocalSummands<'a, const N: usize> {
426    wave: Family,
427    inputs: &'a Inputs<N>,
428    /// The reduced reciprocal rows.
429    rows: &'a [[Jet<N>; 3]; 3],
430    /// The Bloch vector reduced into a cell of `rows`.
431    kpar: &'a [Jet<N>; 3],
432    /// Whether the reduction moved the Bloch vector.
433    kpar_reduced: bool,
434    eta: Complex,
435    split: &'a Split,
436    shared: Option<&'a SphericalCache<N>>,
437    index: Cell<usize>,
438}
439
440impl<const N: usize> ReciprocalSummands<'_, N> {
441    /// The reciprocal-space term at the reciprocal lattice point with the coordinates
442    /// `n`, adding its absolute rounding bounds in units of epsilon to `rounding`. Always
443    /// inlined: it is the body of the term loop of [`Shells::sum`].
444    #[allow(clippy::inline_always)]
445    #[inline(always)]
446    fn at(&self, n: [i64; 3], rounding: &mut Rounding<N>) -> Result<Jet<N>> {
447        if let (Some(shared), Family::Spherical { l, .. }) = (self.shared, self.wave) {
448            let index = self.index.replace(self.index.get() + 1);
449            // Bound the gamma table by 1024 scalar slots per degree, regardless of
450            // multipole order, and the geometry by at most 512 samples. Farther
451            // reciprocal points use ordinary evaluation.
452            let maximum = 1024 / (usize::try_from(l).unwrap_or_default() + 2);
453            if index < maximum {
454                let mut samples = shared.plane_samples.borrow_mut();
455                if index == samples.len() {
456                    samples.push(self.prepare(n)?);
457                }
458                let sample = &mut samples[index];
459                return self.term(
460                    sample.diffraction,
461                    sample.phase,
462                    rounding,
463                    Some(&mut sample.integrals),
464                );
465            }
466        }
467        let sample = self.prepare(n)?;
468        self.term(sample.diffraction, sample.phase, rounding, None)
469    }
470
471    /// Geometry, threshold argument and Bloch phase shared by all spherical orders.
472    #[allow(clippy::inline_always)]
473    #[inline(always)]
474    fn prepare(&self, n: [i64; 3]) -> Result<SphericalReciprocalSample<N>> {
475        let (inputs, rows, kpar) = (self.inputs, self.rows, self.kpar);
476        let dim = inputs.dim;
477        let vector = inputs.point(rows, n);
478        let mut q = [Jet::default(); 3];
479        for j in 0..dim {
480            q[inputs.axes[j]] = vector[j] + kpar[j];
481        }
482        let phase = (-Complex::i()
483            * q.into_iter()
484                .zip(inputs.r)
485                .map(|(q, r)| q * r)
486                .sum::<Jet<N>>())
487        .exp();
488        Ok(SphericalReciprocalSample {
489            diffraction: Diffraction::new(
490                q,
491                !self.kpar_reduced && n == [0; 3],
492                inputs.k,
493                self.eta,
494            )?,
495            phase,
496            integrals: Vec::new(),
497        })
498    }
499
500    #[allow(clippy::inline_always)]
501    #[inline(always)]
502    fn term(
503        &self,
504        diffraction: Diffraction<N>,
505        phase: Jet<N>,
506        rounding: &mut Rounding<N>,
507        plane_integrals: Option<&mut Vec<Option<Complex>>>,
508    ) -> Result<Jet<N>> {
509        let mut bound = Rounding::default();
510        let term = reciprocal_term(
511            self.wave,
512            self.inputs.dim,
513            self.inputs.k,
514            diffraction,
515            self.inputs.r,
516            self.eta,
517            self.split,
518            self.inputs.measure,
519            &mut bound,
520            plane_integrals,
521        )?;
522        *rounding += bound.times(&phase);
523        Ok(term * phase)
524    }
525}

527/// The shift and the Bloch vector of one sum, each reduced into a cell of its reduced
528/// basis.
529struct CellReduction<const N: usize> {
530    /// The Bloch phase `exp(-i q · R)` that compensates subtracting the lattice point `R`
531    /// from the shift.
532    phase: Jet<N>,
533    /// The reduced direct rows.
534    direct_rows: [[Jet<N>; 3]; 3],
535    /// The reduced reciprocal rows.
536    reciprocal_rows: [[Jet<N>; 3]; 3],
537    /// The Bloch vector in lattice coordinates, reduced into a cell of `reciprocal_rows`.
538    kpar: [Jet<N>; 3],
539    /// Whether the reduction moved the Bloch vector.
540    kpar_reduced: bool,
541    /// The shift as given, for the sum at the automatic split ([`AutomaticSplit`]).
542    given_shift: [f64; 3],
543}
544
545/// Reduces the shift of `inputs` into a cell of the reduced direct basis, in place, and
546/// the Bloch vector into a cell of the reduced reciprocal basis.
547fn reduce_to_cell<const N: usize>(
548    lattice: &BlochLattice,
549    inputs: &mut Inputs<N>,
550) -> CellReduction<N> {
551    let dim = inputs.dim;
552    // Coordinates of a lattice-frame vector in the basis `rows`, whose dual is `dual`.
553    let coordinates = |dual: &[[f64; 3]; 3], vector: [f64; 3]| -> [f64; 3] {
554        std::array::from_fn(|i| {
555            if i < dim {
556                (0..dim).map(|j| dual[i][j] * vector[j]).sum::<f64>() / (2.0 * PI)
557            } else {
558                0.0
559            }
560        })
561    };
562    // A Bloch phase compensates the shift's reduction. The cell point is combined from
563    // the given rows, so a shift equal to one of them reduces to exactly zero.
564    let direct_reduction = lattice.direct_reduction.as_ref();
565    let shift = std::array::from_fn(|j| inputs.r[inputs.axes[j]].value.re);
566    let cells = Reduction::given(
567        direct_reduction,
568        Reduction::round(
569            direct_reduction,
570            coordinates(&lattice.reciprocal, shift),
571            dim,
572        ),
573        dim,
574    );
575    let given_shift = inputs.r.map(|r| r.value.re);
576    let cell = inputs.combination(&inputs.direct, cells);
577    for (&axis, &cell) in inputs.axes.iter().zip(&cell).take(dim) {
578        inputs.r[axis] -= cell;
579    }
580    let phase = (-Complex::i() * inputs.phase(&cell)).exp();
581    let direct_rows = inputs.reduce(&inputs.direct, direct_reduction);
582    let reciprocal_reduction = lattice.reciprocal_reduction.as_ref();
583    let reciprocal_rows = inputs.reduce(&inputs.reciprocal, reciprocal_reduction);
584    let cells = Reduction::round(
585        reciprocal_reduction,
586        coordinates(&lattice.direct, lattice.kpar),
587        dim,
588    );
589    let mut kpar = inputs.kpar;
590    let kpar_reduced = cells.iter().any(|&c| c != 0.0);
591    if kpar_reduced {
592        let point = inputs.combination(&reciprocal_rows, cells);
593        for (kpar, point) in kpar.iter_mut().zip(point).take(dim) {
594            *kpar -= point;
595        }
596    }
597    CellReduction {
598        phase,
599        direct_rows,
600        reciprocal_rows,
601        kpar,
602        kpar_reduced,
603        given_shift,
604    }
605}

607/// A sum that vanishes by symmetry ([`vanishes`]) is exactly zero, as are its derivatives
608/// that move `k` alone; jets keep the other derivatives and their check of the
609/// cancellation.
610struct Symmetry<const N: usize> {
611    /// Whether the sum vanishes by symmetry.
612    vanishes: bool,
613    /// Whether each jet slot moves `k` alone, with the shift, the Bloch vector and the
614    /// lattice vectors fixed.
615    k_alone: [bool; N],
616}
617
618impl<const N: usize> Symmetry<N> {
619    /// The symmetry of the sum with the reduced direct rows `direct` and the reduced
620    /// Bloch vector `kpar`.
621    fn of(wave: Family, inputs: &Inputs<N>, direct: &[[Jet<N>; 3]; 3], kpar: &[Jet<N>; 3]) -> Self {
622        Self {
623            vanishes: vanishes(wave, inputs, direct, kpar),
624            k_alone: std::array::from_fn(|slot| {
625                inputs
626                    .r
627                    .iter()
628                    .chain(&inputs.kpar)
629                    .chain(inputs.direct.iter().flatten())
630                    .all(|input| input.derivative[slot] == Complex::default())
631            }),
632        }
633    }
634
635    /// `sum` with the components that vanish by symmetry set to exactly zero.
636    fn zeroed(&self, mut sum: Jet<N>) -> Jet<N> {
637        if self.vanishes {
638            sum.value = Complex::default();
639            for (derivative, &k_alone) in sum.derivative.iter_mut().zip(&self.k_alone) {
640                if k_alone {
641                    *derivative = Complex::default();
642                }
643            }
644        }
645        sum
646    }
647
648    /// The components whose cancellation a sum below every automatic split checks: those
649    /// that [`Self::zeroed`] keeps, without the split derivative of the parts.
650    fn checked(&self, inputs: &Inputs<N>, evaluation: Evaluation) -> Checked<N> {
651        Checked {
652            value: !self.vanishes,
653            units: std::array::from_fn(|slot| {
654                if matches!(evaluation, Evaluation::EtaDerivative)
655                    || (self.vanishes && self.k_alone[slot])
656                {
657                    None
658                } else {
659                    Some(derivative_scale(slot, inputs.dim, inputs.k.value))
660                }
661            }),
662        }
663    }
664}

666/// The same sum at the automatic split, against which sums below every automatic split
667/// fail early ([`SmallSplitMonitor::hopeless`]) or verify their settled shells
668/// ([`verify_settled`]).
669#[derive(Clone, Copy)]
670struct AutomaticSplit<'a> {
671    wave: Family,
672    k: Complex,
673    lattice: &'a BlochLattice,
674    /// The shift as given, before its reduction into a cell.
675    shift: [f64; 3],
676    evaluation: Evaluation,
677}
678
679impl AutomaticSplit<'_> {
680    fn sum<const N: usize>(&self) -> Result<Jet<N>> {
681        evaluate::<N>(
682            self.wave,
683            self.k,
684            self.lattice,
685            self.shift,
686            Complex::default(),
687            self.evaluation,
688        )
689    }
690}

692/// The real-space terms below every automatic split: the Kambe integrals that keep their
693/// rounding below that of the largest term so far, and the early failure of complete
694/// sums ([`EARLY_FAILURE_LOSS`]).
695struct SmallSplitMonitor<'a, const N: usize> {
696    /// The Kambe integrals of the real-space terms below every automatic split, except
697    /// for the split derivative.
698    kambe: Option<SmallSplitKambe>,
699    /// The largest moduli of the value and derivatives of a real-space term so far,
700    /// whose rounding the closed forms of the later terms may reach.
701    largest: Cell<Rounding<N>>,
702    /// Whether the sum fails early once the rounding its real-space terms predict is
703    /// [`hopeless`](Self::hopeless).
704    early: bool,
705    /// The rounding the real-space terms predict so far.
706    running: Cell<Rounding<N>>,
707    /// The sum at the automatic split, on whose scales the sum fails early, evaluated
708    /// once; `None` where it fails.
709    automatic_sum: OnceCell<Option<Jet<N>>>,
710    automatic_split: AutomaticSplit<'a>,
711    checked: &'a Checked<N>,
712}
713
714impl<'a, const N: usize> SmallSplitMonitor<'a, N> {
715    fn new(
716        kambe: Option<SmallSplitKambe>,
717        early: bool,
718        automatic_split: AutomaticSplit<'a>,
719        checked: &'a Checked<N>,
720    ) -> Self {
721        Self {
722            kambe,
723            largest: Cell::new(Rounding::default()),
724            early,
725            running: Cell::new(Rounding::default()),
726            automatic_sum: OnceCell::new(),
727            automatic_split,
728            checked,
729        }
730    }
731
732    /// With [`Self::early`], adds the rounding of the real-space term `term` and its bound
733    /// `bound` to the running rounding and checks whether it is hopeless.
734    fn add_to_running(&self, term: &Jet<N>, bound: Rounding<N>) -> Result<()> {
735        if self.early {
736            let mut total = self.running.get();
737            total += Rounding::relative(term, TERM_ULPS);
738            total += bound;
739            self.running.set(total);
740            self.hopeless(&total)?;
741        }
742        Ok(())
743    }
744
745    /// Fails where the rounding `bound` exceeds [`EARLY_FAILURE_LOSS`] of the scale of a
746    /// checked component of the sum at the automatic split.
747    fn hopeless(&self, bound: &Rounding<N>) -> Result<()> {
748        let checked = self.checked;
749        let beyond = |bound: f64, scale: f64| f64::EPSILON * bound > EARLY_FAILURE_LOSS * scale;
750        // The scales are at least 1 for the value and `s` for a derivative, so smaller
751        // losses need no automatic sum.
752        let slots = || bound.derivative.iter().zip(checked.units);
753        let candidate = (checked.value && beyond(bound.value, 1.0))
754            || slots().any(|(&bound, unit)| unit.is_some_and(|unit| beyond(bound, unit)));
755        if !candidate {
756            return Ok(());
757        }
758        let Some(automatic) = self
759            .automatic_sum
760            .get_or_init(|| self.automatic_split.sum::<N>().ok())
761        else {
762            return Ok(());
763        };
764        let scale = automatic.value.norm().max(1.0);
765        if checked.value && beyond(bound.value, scale) {
766            return Err(too_small(f64::EPSILON * bound.value / scale, false));
767        }
768        for ((&bound, unit), derivative) in slots().zip(&automatic.derivative) {
769            if let Some(unit) = unit {
770                let scale = scale.mul_add(unit, derivative.norm());
771                if beyond(bound, scale) {
772                    return Err(too_small(f64::EPSILON * bound / scale, true));
773                }
774            }
775        }
776        Ok(())
777    }
778}

780/// `sum`, whose real-space shells settled at their limit ([`Shells::settled`]), if it
781/// agrees with the sum at the automatic split within the rounding `predicted` that its
782/// parts predict for the value (without the truncation that the comparison verifies)
783/// plus twice the convergence tolerance of each shell sum.
784fn verify_settled<const N: usize>(
785    sum: Jet<N>,
786    predicted: f64,
787    automatic_split: AutomaticSplit<'_>,
788) -> Result<Jet<N>> {
789    // A larger split cannot help where the automatic one does not converge either.
790    let automatic = automatic_split.sum::<N>().map_err(|error| {
791        if is_shell_limit(&error) {
792            Error::NotConverged(format!(
793                "{NOT_CONVERGED}, neither at this split nor at the automatic one"
794            ))
795        } else {
796            error
797        }
798    })?;
799    let scale = automatic.value.norm().max(1.0);
800    let relative = f64::EPSILON.mul_add(predicted / scale, 4.0 * SHELL_TOLERANCE);
801    let agrees = |a: Complex, b: Complex| (a - b).norm() <= relative * (b.norm() + scale);
802    if agrees(sum.value, automatic.value)
803        && sum
804            .derivative
805            .iter()
806            .zip(&automatic.derivative)
807            .all(|(&a, &b)| agrees(a, b))
808    {
809        Ok(sum)
810    } else {
811        Err(not_converged())
812    }
813}

815/// Whether the sum vanishes by symmetry.
816///
817/// That holds for odd degrees of spherical and odd orders of cylindrical waves where the
818/// reduced Bloch vector `kpar` is exactly zero and twice the reduced shift is exactly one
819/// of the combinations of the reduced rows `direct` with coefficients in `{-1, 0, 1}`
820/// (`2 r` lies in their cell).
821///
822/// # Why
823///
824/// Inversion maps the images `-r - R` onto themselves and changes the sign of the waves,
825/// so every Ewald part and the self term vanish at any split and `k`, while the computed
826/// parts would leave rounding of the order of the terms. The derivatives that move `k`
827/// alone vanish too; the others keep the check of the predicted loss ([`cancellation`]).
828pub(super) fn vanishes<const N: usize>(
829    wave: Family,
830    inputs: &Inputs<N>,
831    direct: &[[Jet<N>; 3]; 3],
832    kpar: &[Jet<N>; 3],
833) -> bool {
834    let odd = match wave {
835        Family::Spherical { l, .. } => l % 2 != 0,
836        Family::Cylindrical { m } => m % 2 != 0,
837    };
838    let dim = inputs.dim;
839    let lattice_axis = |axis: &usize| inputs.axes[..dim].contains(axis);
840    if !odd
841        || kpar[..dim].iter().any(|q| q.value != Complex::default())
842        || (0..3).any(|axis| !lattice_axis(&axis) && inputs.r[axis].value != Complex::default())
843    {
844        return false;
845    }
846    let twice: [f64; 3] = std::array::from_fn(|j| {
847        if j < dim {
848            2.0 * inputs.r[inputs.axes[j]].value.re
849        } else {
850            0.0
851        }
852    });
853    // Integer coefficients in {-1, 0, 1} for every reduced row.
854    let count = 3_usize.pow(u32::try_from(dim).unwrap_or(3));
855    (0..count).any(|index| {
856        let mut rest = index;
857        let coefficients: [f64; 3] = std::array::from_fn(|i| {
858            if i >= dim {
859                return 0.0;
860            }
861            let digit = rest % 3;
862            rest /= 3;
863            match digit {
864                0 => 0.0,
865                1 => 1.0,
866                _ => -1.0,
867            }
868        });
869        let point = inputs.combination(direct, coefficients);
870        (0..dim).all(|j| point[j].value.re == twice[j])
871    })
872}

876    //! Test probes of the Ewald sum: switches that turn off one decision of [`ewald`], a
877    //! count of its real-space terms, and the reduced integrals and the spectral series

901    /// Restores a switch when dropped, so a panicking test leaves it as it was.

916            switch,
917            previous: switch.replace(true),
918        };

978    /// `F_n(v, t)` of one reciprocal point, of squared transverse modulus `q2` at the
979    /// distance `s` from the lattice plane or axis (`t = (k s eta)^2`), from its series in
980    /// `t` and from Kambe integrals, each with its rounding bound, on the sheet of the
981    /// automatic split.
982    pub(crate) fn reduced_paths(
983        twice_n: i32,
984        k: Complex,
985        q2: f64,
986        eta: Complex,
987        distance: f64,
988    ) -> [(Complex, f64); 2] {
989        let v = lower_side((q2 / (k * k) - 1.0) / (2.0 * eta * eta));
990        let t = (k * distance * eta).powi(2);
991        let sheet = Some(split_sheet(&Split::new(k, eta), q2.into()));
992        let mut reduced = Reduced::<0>::new(Jet::constant(v), Jet::constant(t), sheet);
993        let kambe = if twice_n % 2 == 0 {
994            reduced_kambe(twice_n, v, t.sqrt(), sheet)
995        } else {
996            (reduced.by_even_recurrence(twice_n), 0.0)
997        };
998        [reduced.series(twice_n), kambe]
999    }
1000
1001    /// The spectral series of a 1D spherical sum with its derivatives in every input, as
1002    /// [`derivatives`] returns them, and the rounding bounds of each component in units
1003    /// of epsilon (as real parts, in the same layout), or `None` where [`spectral_sw1d`]
1004    /// is not available.
1005    ///
1006    /// [`derivatives`]: super::super::derivatives
1007    pub(crate) fn spectral_sw1d_derivatives(
1008        (l, m): (i32, i32),
1009        k: Complex,
1010        lattice: &BlochLattice,
1011        r: [f64; 3],
1012    ) -> Option<(Derivatives, Derivatives)> {
1013        let inputs = Inputs::<6>::new(Family::Spherical { l, m }, k, lattice, r, true, true);
1014        let (series, bound) = spectral_sw1d(
1015            (l, m),
1016            inputs.k,
1017            inputs.r,
1018            inputs.kpar[0],
1019            inputs.reciprocal[0][0],
1020            inputs.measure,
1021        )?;
1022        let bound = Jet {
1023            value: bound.value.into(),
1024            derivative: bound.derivative.map(Complex::from),
1025        };
1026        Some((unpack(series, 1), unpack(bound, 1)))
1027    }
1028}
