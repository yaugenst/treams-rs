1//! Multipole rotations in the z-y-z convention, with analytic Euler-angle pullbacks.
2//!
3//! [`sw_rotate`] and [`cw_rotate`] give the coefficient of one mode pair; [`sw_rotation`]
4//! and [`cw_rotation`] give the matrix between two bases. Both matrices return one
5//! [`RotationResidual`], whose pullback gives the cotangents of the Euler angles
6//! `[phi, theta, psi]`, so the two families share this module instead of living in
7//! [`crate::sw`] and [`crate::cw`].
8//!
9//! Upstream: `treams.sw.rotate`, `treams.cw.rotate` and the `treams.Rotate` operator.
10#![allow(clippy::indexing_slicing)] // Validated mode labels and complete angular blocks.
11#![allow(clippy::float_cmp)] // Exact zero rotations and exact axial labels.
12

24/// Entries `(row, column)` whose coupling keys agree, in column-major order; all
25/// other entries of a rotation vanish.
26fn coupled<K: Eq + std::hash::Hash>(rows: &[K], columns: &[K]) -> Vec<(usize, usize)> {
27    let mut groups: HashMap<&K, Vec<usize>> = HashMap::new();
28    for (i, key) in rows.iter().enumerate() {
29        groups.entry(key).or_default().push(i);
30    }
31    columns
32        .iter()
33        .enumerate()
34        .flat_map(|(j, key)| groups.get(key).into_iter().flatten().map(move |&i| (i, j)))
35        .collect()
36}

38/// The Wigner small-d blocks and Euler phases of a spherical rotation.
39#[derive(Debug)]
40struct Spherical {
41    /// Degree of each row.
42    degrees: Vec<i32>,
43    tables: BTreeMap<i32, DMatrix<f64>>,
44    /// Row phases `exp(-i mu phi)` and column phases `exp(-i m psi)`.
45    phases: [Vec<Complex>; 2],
46}
47
48/// What [`sw_rotation`] and [`cw_rotation`] save for their pullback: the rotation
49/// matrix, its coupled entries and, for spherical bases, the angular blocks.
50///
51/// The pullback adds the entries in order on the calling thread, so its result does not
52/// depend on the thread count.
53#[derive(Debug)]
54pub struct RotationResidual {
55    rows: Vec<i32>,
56    columns: Vec<i32>,
57    /// The coupled entries; all others vanish.
58    entries: Vec<(usize, usize)>,
59    spherical: Option<Spherical>,
60    value: DMatrix<Complex>,
61}
62
63/// Spherical Euler rotation; distinct positions, degrees and polarizations decouple.
64///
65/// Upstream: the `treams.Rotate` operator (`treams.operators.rotate`) between spherical
66/// bases.
67pub fn sw_rotation(
68    destination: &sw::Basis,
69    source: &sw::Basis,
70    angles: [f64; 3],
71) -> Result<RotationResidual> {
72    destination.validate()?;
73    source.validate()?;
74    if angles.iter().any(|a| !a.is_finite()) {
75        return Err(Error::InvalidInput("rotation angles must be finite".into()));
76    }
77    let mut tables = BTreeMap::new();
78    for &(_, mode) in &destination.modes {
79        if let std::collections::btree_map::Entry::Vacant(e) = tables.entry(mode.l) {
80            e.insert(wigner_small_d_matrix(mode.l, angles[1])?);
81        }
82    }
83    let key = |basis: &sw::Basis| -> Vec<_> {
84        basis
85            .modes
86            .iter()
87            .map(|&(pidx, mode)| (pidx, mode.l, mode.pol))
88            .collect()
89    };
90    let entries = coupled(&key(destination), &key(source));
91    let orders = |basis: &sw::Basis| -> Vec<_> { basis.modes.iter().map(|(_, m)| m.m).collect() };
92    let (rows, columns) = (orders(destination), orders(source));
93    let phase = |m: i32, angle: f64| (-Complex::i() * f64::from(m) * angle).exp();
94    let spherical = Spherical {
95        degrees: destination.modes.iter().map(|(_, m)| m.l).collect(),
96        tables,
97        phases: [
98            rows.iter().map(|&mu| phase(mu, angles[0])).collect(),
99            columns.iter().map(|&m| phase(m, angles[2])).collect(),
100        ],
101    };
102    let mut value = numerics::zeros(rows.len(), columns.len())?;
103    for &(i, j) in &entries {
104        let l = spherical.degrees[i];
105        let d = spherical.tables[&l][(index(l, rows[i]), index(l, columns[j]))];
106        value[(i, j)] = spherical.phases[0][i] * spherical.phases[1][j] * d;
107    }
108    Ok(RotationResidual {
109        rows,
110        columns,
111        entries,
112        spherical: Some(spherical),
113        value,
114    })
115}

117/// Cylindrical rotation about z; axial labels and the transverse plane stay fixed.
118///
119/// Upstream: the `treams.Rotate` operator (`treams.operators.rotate`) between cylindrical
120/// bases.
121pub fn cw_rotation(
122    destination: &cw::Basis,
123    source: &cw::Basis,
124    angles: [f64; 3],
125) -> Result<RotationResidual> {
126    destination.validate()?;
127    source.validate()?;
128    if angles.iter().any(|a| !a.is_finite()) || angles[1] != 0.0 {
129        return Err(Error::InvalidInput(
130            "cylindrical rotation requires finite angles and theta=0".into(),
131        ));
132    }
133    let key = |basis: &cw::Basis| -> Vec<_> {
134        basis
135            .modes
136            .iter()
137            .map(|&(pidx, mode)| (pidx, label_bits(mode.kz), mode.m, mode.pol))
138            .collect()
139    };
140    let entries = coupled(&key(destination), &key(source));
141    let orders = |basis: &cw::Basis| -> Vec<_> { basis.modes.iter().map(|(_, m)| m.m).collect() };
142    let (rows, columns) = (orders(destination), orders(source));
143    let mut value = numerics::zeros(rows.len(), columns.len())?;
144    for &(i, j) in &entries {
145        value[(i, j)] = (-Complex::i() * f64::from(rows[i]) * (angles[0] + angles[2])).exp();
146    }
147    Ok(RotationResidual {
148        rows,
149        columns,
150        entries,
151        spherical: None,
152        value,
153    })
154}

156impl RotationResidual {
157    /// The rotation from the source to the destination multipole coefficients, which
158    /// the pullback reads.
159    #[must_use]
160    pub const fn value(&self) -> &DMatrix<Complex> {
161        &self.value
162    }
163
164    /// The shape of the rotation: destination and source mode counts.
165    #[must_use]
166    pub fn shape(&self) -> (usize, usize) {
167        self.value.shape()
168    }
169
170    /// Euler-angle cotangents. Cylindrical theta is a fixed zero constraint.
171    pub fn pullback(self, cotangent: &DMatrix<Complex>) -> Result<[f64; 3]> {
172        if cotangent.shape() != self.shape() || cotangent.iter().any(|&z| !finite(z)) {
173            return Err(Error::InvalidInput("invalid rotation cotangent".into()));
174        }
175        let mut result = [0.0; 3];
176        for &(i, j) in &self.entries {
177            let (mu, m) = (self.rows[i], self.columns[j]);
178            let g = cotangent[(i, j)].conj();
179            // `d/dphi D = -i mu D` and `d/dpsi D = -i m D`.
180            let paired = (g * (-Complex::i()) * self.value[(i, j)]).re;
181            result[0] += f64::from(mu) * paired;
182            result[2] += f64::from(m) * paired;
183            if let Some(s) = &self.spherical {
184                // `d/dtheta d_(mu m) = c(mu) d_(mu+1, m) - c(mu-1) d_(mu-1, m)` with the
185                // ladder coefficients `c`.
186                let l = s.degrees[i];
187                let d = &s.tables[&l];
188                let column = index(l, m);
189                let raised = if mu < l {
190                    ladder(l, mu) * d[(index(l, mu + 1), column)]
191                } else {
192                    0.0
193                };
194                let lowered = if mu > -l {
195                    ladder(l, mu - 1) * d[(index(l, mu - 1), column)]
196                } else {
197                    0.0
198                };
199                let phase = s.phases[0][i] * s.phases[1][j];
200                result[1] += (g * phase * (raised - lowered)).re;
201            }
202        }
203        Ok(result)
204    }
205}

207/// Rotation coefficient of one spherical mode pair.
208///
209/// For the Euler angles `[phi, theta, psi]`, the coefficient from `source` to
210/// `destination` is the Wigner D-matrix element `D^l_(m' m)(phi, theta, psi)`
211/// ([`wigner_d`]) for equal degrees and polarizations, and zero otherwise; `m'` is the
212/// order of `destination` and `m` that of `source`.
213///
214/// Upstream: `treams.sw.rotate`.
215/// Differences: modes with `l = 0`, `l > 128` or `|m| > l` give an error, and so do
216/// non-finite angles at equal degrees and polarizations; treams returns a value or NaN for
217/// them, for example 1 at `l = 0` and zero for `|m| > l`.
218#[inline]
219pub fn sw_rotate(
220    destination: sw::Mode,
221    source: sw::Mode,
222    [phi, theta, psi]: [f64; 3],
223) -> Result<Complex> {
224    destination.validate()?;
225    source.validate()?;
226    if destination.l != source.l || destination.pol != source.pol {
227        Ok(Complex::default())
228    } else {
229        wigner_d(
230            destination.l,
231            destination.m,
232            source.m,
233            [phi.into(), theta.into(), psi.into()],
234        )
235    }
236}

238/// Rotation coefficient of one cylindrical mode pair about the z axis.
239///
240/// For the angle `phi`, the coefficient from `(qz, m, q)` to `(kz, mu, p)` is
241/// `exp(-i m phi)` for equal modes and zero otherwise. The polarizations `p` and `q`
242/// must be 0 or 1, the axial wavenumbers and `phi` finite, and the orders satisfy
243/// `|mu|, |m| <= 128` ([`MAX_DEGREE`]). The labels are `i64` so that every integer label
244/// gets these checks before its conversion.
245///
246/// Upstream: `treams.cw.rotate`.
247/// Differences: treams accepts orders above 128 and non-finite arguments, where this
248/// function gives an error.
249#[inline]
250pub fn cw_rotate(kz: f64, mu: i64, p: i64, qz: f64, m: i64, q: i64, phi: f64) -> Result<Complex> {
251    let p = pol_index(p)?;
252    let q = pol_index(q)?;
253    let bound = u64::from(MAX_DEGREE.unsigned_abs());
254    if !kz.is_finite()
255        || !qz.is_finite()
256        || !phi.is_finite()
257        || mu.unsigned_abs() > bound
258        || m.unsigned_abs() > bound
259    {
260        return Err(Error::InvalidInput(
261            // 128 is MAX_DEGREE.
262            "finite axial/angle arguments and |orders| <= 128 required".into(),
263        ));
264    }
265    if kz != qz || mu != m || p != q {
266        Ok(Complex::default())
267    } else {
268        let order =
269            f64::from(i32::try_from(m).map_err(|_| Error::InvalidInput("invalid order".into()))?);
270        let (sin, cos) = (-order * phi).sin_cos();
271        Ok(Complex::new(cos, sin))
272    }
