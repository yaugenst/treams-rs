1//! Bessel functions and radial jets with analytic argument derivatives.
2//!
3//! Cylindrical and spherical Bessel functions of every kind ([`bessel`]), regular and
4//! singular radial jets ([`spherical_radial`], [`cylindrical_radial`]), and their
5//! broadcasting residual. Away from the small-argument power series they call the AMOS
6//! library (the `complex_bessel` crate), as scipy does.
7//!
8//! Upstream: `treams.special.jv`, `yv`, `hankel1`, `hankel2`, `spherical_jn` and
9//! `spherical_yn` (scipy's functions), and `jv_d`, `yv_d`, `hankel1_d`, `hankel2_d`,
10//! `spherical_hankel1`, `spherical_hankel2` and their `_d` derivatives (`_bessel.pyx`).

29/// Cylindrical Bessel solution, also used at half order for spherical functions.
30///
31/// Upstream: `treams.special.jv` (`J`), `yv` (`Y`), `hankel1` (`H1`) and `hankel2`
32/// (`H2`).
33#[derive(Clone, Copy, Debug, PartialEq, Eq)]
34pub enum Bessel {
35    /// Regular first-kind solution.
36    J,
37    /// Second-kind solution.
38    Y,
39    /// Outgoing Hankel solution.
40    H1,
41    /// Incoming Hankel solution.
42    H2,
43}

45impl From<Radial> for Bessel {
46    /// Regular waves are J, singular (outgoing) waves H1.
47    fn from(radial: Radial) -> Self {
48        match radial {
49            Radial::Regular => Self::J,
50            Radial::Singular => Self::H1,
51        }
52    }
53}

55fn amos_error(error: complex_bessel::Error) -> Error {
56    Error::SpecialFunction(error.to_string())
57}

59/// `sqrt(pi / 2z)`, which turns cylindrical functions of order `l + 1/2` into
60/// spherical functions of order `l`.
61///
62/// On the negative real axis AMOS takes the limit from above for either sign of a
63/// zero imaginary part (it tests `Im z < 0`), where `pi / 2z` approaches the axis from
64/// below. Real-by-complex division gives `pi / 2z` a `+0` imaginary part there, so the
65/// prefactor takes that side explicitly; the spherical functions are single-valued,
66/// and only the two factors' agreement matters.
67fn spherical_prefactor(z: Complex) -> Complex {
68    let mut w = PI / (2.0 * z);
69    if z.im == 0.0 && z.re < 0.0 {
70        w.im = -0.0;
71    }
72    crate::numerics::complex_sqrt(w)
73}

75/// One AMOS evaluation of a cylindrical function, or of a spherical one at order + 1/2.
76#[inline]
77fn bessel_raw(order: f64, z: Complex, kind: Bessel, spherical: bool) -> Result<Complex> {
78    let order = if spherical { order + 0.5 } else { order };
79    let value = match kind {
80        Bessel::J => complex_bessel::besselj(order, z),
81        Bessel::Y => complex_bessel::bessely(order, z),
82        Bessel::H1 => complex_bessel::hankel1(order, z),
83        Bessel::H2 => complex_bessel::hankel2(order, z),
84    }
85    .map_err(amos_error)?;
86    Ok(if spherical {
87        spherical_prefactor(z) * value
88    } else {
89        value
90    })
91}

93/// Cylindrical functions of orders `start, start + 1, ...` from one AMOS recurrence.
94fn sequence(kind: Bessel, start: f64, z: Complex, length: usize) -> Result<Vec<Complex>> {
95    let unscaled = complex_bessel::Scaling::Unscaled;
96    match kind {
97        Bessel::J => complex_bessel::besselj_seq(start, z, length, unscaled),
98        Bessel::Y => complex_bessel::bessely_seq(start, z, length, unscaled),
99        Bessel::H1 => complex_bessel::hankel1_seq(start, z, length, unscaled),
100        Bessel::H2 => complex_bessel::hankel2_seq(start, z, length, unscaled),
101    }
102    .map(|sequence| sequence.values)
103    .map_err(amos_error)
104}

106/// The two values of a two-term sequence.
107fn pair(values: Vec<Complex>) -> Result<[Complex; 2]> {
108    <[Complex; 2]>::try_from(values)
109        .map_err(|_| Error::SpecialFunction("invalid Bessel sequence length".into()))
110}

112/// Outgoing cylindrical functions `[H_(m-1)(z), H_m(z)]` from one AMOS sequence, with
113/// `H_-1 = -H_1`: the derivative `u^m H_(m-1)(u)` of `u^m H_m(u)` without the
114/// cancelling product rule.
115pub(crate) fn hankel_below(order: u32, z: Complex) -> Result<[Complex; 2]> {
116    if order == 0 {
117        let [value, upper] = pair(sequence(Bessel::H1, 0.0, z, 2)?)?;
118        return Ok([-upper, value]);
119    }
120    pair(sequence(Bessel::H1, f64::from(order - 1), z, 2)?)
121}

123/// Spherical functions of orders `order` and `order + 1`. Hankel functions share one
124/// AMOS sequence, which agrees with single evaluations to 2e-14 relative. J and Y
125/// sequences differ from single evaluations by up to 1e-12, so these stay single.
126fn spherical_pair(kind: Bessel, order: f64, z: Complex) -> Result<[Complex; 2]> {
127    let values = if matches!(kind, Bessel::H1 | Bessel::H2) {
128        pair(sequence(kind, order + 0.5, z, 2)?)?
129    } else {
130        let single = |order: f64| bessel_raw(order + 0.5, z, kind, false);
131        [single(order)?, single(order + 1.0)?]
132    };
133    let prefactor = spherical_prefactor(z);
134    Ok(values.map(|value| prefactor * value))
135}

137/// `jet`, or an overflow error where one of its parts is not finite. Near zero the
138/// prefactor `sqrt(pi / 2z)` and the derivatives' powers of `1/z` overflow where AMOS
139/// does not, as for `h_0` below `|z| = 1e-103`.
140fn checked(jet: RadialJet) -> Result<RadialJet> {
141    if finite(jet.value) && finite(jet.first) && finite(jet.second) {
142        Ok(jet)
143    } else {
144        Err(Error::SpecialFunction(
145            "radial function or derivative overflow".into(),
146        ))
147    }
148}

150/// Derivatives of a spherical function from the next order, `f' = (l/z) f - f_(l+1)`,
151/// and the spherical Bessel equation.
152fn jet_from_next_order(order: f64, z: Complex, [value, next]: [Complex; 2]) -> RadialJet {
153    let first = order * ratio(value, z) - next;
154    let second = -2.0 * ratio(first, z) + order * (order + 1.0) * ratio(ratio(value, z), z) - value;
155    RadialJet {
156        value,
157        first,
158        second,
159    }
160}

162/// Bessel value or one of its first two complex-argument derivatives.
163///
164/// `kind` selects J, Y, H1 or H2. `spherical` selects the spherical function of order
165/// `order` (`j`, `y`, `h1` or `h2`): the cylindrical function of order `order + 1/2`
166/// times `sqrt(pi / (2z))`. `derivative` is 0, 1 or 2, always with respect to `z`.
167/// Spherical regular functions use analytic origin limits.
168///
169/// Upstream: `treams.special.jv`, `yv`, `hankel1`, `hankel2` and with `spherical`
170/// `spherical_jn`, `spherical_yn`, `spherical_hankel1`, `spherical_hankel2`; their `_d`
171/// variants for `derivative = 1`. treams has no second derivative.
172#[allow(clippy::cast_possible_truncation, clippy::cast_sign_loss)] // Integral order in 0..=MAX_ORDER checked before conversion.
173#[inline]
174pub fn bessel(
175    order: f64,
176    z: Complex,
177    kind: Bessel,
178    spherical: bool,
179    derivative: u8,
180) -> Result<Complex> {
181    if !order.is_finite() || !finite(z) || derivative > 2 {
182        return Err(Error::InvalidInput(
183            "finite order/argument and derivative 0, 1 or 2 required".into(),
184        ));
185    }
186    // AMOS-based kernels lose accuracy for subnormal orders (including NaNs in
187    // SciPy). Their zero-order limit is identical at double precision: even at
188    // extreme finite z, order * log(z) is far below a representable correction.
189    let order = if order.is_subnormal() { 0.0 } else { order };
190    let jet = |jet: RadialJet| match derivative {
191        0 => jet.value,
192        1 => jet.first,
193        _ => jet.second,
194    };
195    let evaluate = |v| bessel_raw(v, z, kind, spherical);
196    let value = if spherical
197        && kind == Bessel::J
198        && z.norm() < SERIES_RADIUS
199        && (0.0..=f64::from(MAX_ORDER)).contains(&order)
200        && order.fract() == 0.0
201    {
202        jet(spherical_radial(order as u32, z, Radial::Regular)?)
203    } else if derivative == 0 {
204        evaluate(order)?
205    } else if spherical {
206        jet(jet_from_next_order(
207            order,
208            z,
209            spherical_pair(kind, order, z)?,
210        ))
211    } else if matches!(kind, Bessel::H1 | Bessel::H2) && (order != 0.0 || derivative == 2) {
212        let length = usize::from(2 * derivative + 1);
213        match sequence(kind, order - f64::from(derivative), z, length)?.as_slice() {
214            [left, _, right] => 0.5 * (left - right),
215            [left, _, middle, _, right] => 0.25 * (left - 2.0 * middle + right),
216            _ => {
217                return Err(Error::SpecialFunction(
218                    "invalid Hankel sequence length".into(),
219                ));
220            }
221        }
222    } else if derivative == 1 {
223        if order == 0.0 {
224            -evaluate(1.0)?
225        } else {
226            0.5 * (evaluate(order - 1.0)? - evaluate(order + 1.0)?)
227        }
228    } else {
229        0.25 * (evaluate(order - 2.0)? - 2.0 * evaluate(order)? + evaluate(order + 2.0)?)
230    };
231    if !finite(value) {
232        return Err(Error::SpecialFunction("non-finite Bessel result".into()));
233    }
234    Ok(value)
235}

237/// What [`bessel_array`] saves for its pullback: the orders, the arguments and the
238/// function. Only the arguments get gradients.
239#[derive(Debug)]
240pub struct BesselResidual {
241    orders: Vec<f64>,
242    arguments: Vec<Complex>,
243    kind: Bessel,
244    spherical: bool,
245    derivative: u8,
246    size: usize,
247}

249/// Evaluate borrowed elementwise Bessel arguments without recording a pullback; the
250/// forward pass of [`bessel_array`].
251pub(crate) fn bessel_values(
252    orders: &[f64],
253    arguments: &[Complex],
254    kind: Bessel,
255    spherical: bool,
256    derivative: u8,
257) -> Result<Vec<Complex>> {
258    let message = "Bessel arrays must have equal lengths or scalar inputs, and derivative 0 or 1";
259    if derivative > 1 {
260        return Err(Error::InvalidInput(message.into()));
261    }
262    let size = broadcast::size(&[orders.len(), arguments.len()], message)?;
263    broadcast::map(size, PARALLEL, |i| {
264        bessel(
265            element(orders, i),
266            element(arguments, i),
267            kind,
268            spherical,
269            derivative,
270        )
271    })
272}

274/// Evaluate elementwise Bessel functions, broadcasting either scalar input.
275///
276/// `kind` and `spherical` select the function as in [`bessel`]. `derivative` is 0 or 1,
277/// one below the cap of [`bessel`], because the pullback evaluates derivative + 1.
278///
279/// Upstream: the functions of [`bessel`] over arrays.
280pub fn bessel_array(
281    orders: Vec<f64>,
282    arguments: Vec<Complex>,
283    kind: Bessel,
284    spherical: bool,
285    derivative: u8,
286) -> Result<(Vec<Complex>, BesselResidual)> {
287    let value = bessel_values(&orders, &arguments, kind, spherical, derivative)?;
288    let size = value.len();
289    Ok((
290        value,
291        BesselResidual {
292            orders,
293            arguments,
294            kind,
295            spherical,
296            derivative,
297            size,
298        },
299    ))
300}
301
302impl BesselResidual {
303    fn evaluate(&self, i: usize, derivative: u8) -> Result<Complex> {
304        bessel(
305            element(&self.orders, i),
306            element(&self.arguments, i),
307            self.kind,
308            self.spherical,
309            derivative,
310        )
311    }
312
313    /// The gradient with respect to the arguments, given the gradient `cotangent` with
314    /// respect to the values: `cotangent` times the conjugate of the next derivative. An
315    /// argument given as one value for all elements receives the sum of its element
316    /// gradients; the orders have no gradient.
317    pub fn pullback(self, cotangent: &[Complex]) -> Result<Vec<Complex>> {
318        let [gradient] = broadcast::pullback(
319            cotangent,
320            self.size,
321            "Bessel cotangent must be finite and match output",
322            [self.arguments.len()],
323            PARALLEL,
324            |i, &g| {
325                if g == Complex::default() {
326                    Ok([g])
327                } else {
328                    Ok([g * self.evaluate(i, self.derivative + 1)?.conj()])
329                }
330            },
331        )?;
332        Ok(gradient)
333    }
334}

350/// A radial function and its first two argument derivatives: a second-order jet in one
351/// complex variable, unlike the crate's first-order jets `numerics::Jet` in several.
352#[derive(Clone, Copy, Debug)]
353pub struct RadialJet {
354    /// Function value.
355    pub value: Complex,
356    /// First derivative with respect to the complex argument.
357    pub first: Complex,
358    /// Second derivative with respect to the complex argument.
359    pub second: Complex,
360}

362/// Terms of the small-argument power series of [`spherical_radial`] and
363/// [`cylindrical_radial`]. Below `|z| = SERIES_RADIUS` (0.5) each term is at most
364/// `1/(16 q^2)` of its predecessor, so the last one is below 1e-40 of the first.
365const SERIES_TERMS: u32 = 16;
366
367/// The even power series `sum_q t_q` with `t_0 = leading z^power` and
368/// `t_q = factor(q) z^2 t_(q-1)`, differentiated term by term: every term reuses its
369/// predecessor's powers.
370fn even_series(power: u32, leading: f64, z: Complex, factor: impl Fn(f64) -> f64) -> RadialJet {
371    let mut term = RadialJet {
372        value: leading * z.powu(power),
373        first: if power > 0 {
374            leading * f64::from(power) * z.powu(power - 1)
375        } else {
376            Complex::default()
377        },
378        second: if power > 1 {
379            leading * f64::from(power * (power - 1)) * z.powu(power - 2)
380        } else {
381            Complex::default()
382        },
383    };
384    let mut sum = term;
385    let square = z * z;
386    for q in 1..SERIES_TERMS {
387        let factor = factor(f64::from(q));
388        term = RadialJet {
389            value: factor * square * term.value,
390            first: factor * (square * term.first + 2.0 * z * term.value),
391            second: factor * (square * term.second + 4.0 * z * term.first + 2.0 * term.value),
392        };
393        sum.value += term.value;
394        sum.first += term.first;
395        sum.second += term.second;
396    }
397    sum
398}

400/// Evaluate a spherical radial function and its first two derivatives: `j_l` for
401/// regular and `h1_l` for singular waves.
402///
403/// The regular solution uses its convergent power series near zero; away from
404/// zero, complex Bessel evaluation and the spherical differential equation apply.
405/// Singular (outgoing) waves fail at zero; near it, where the value or a derivative
406/// overflows, they fail with an overflow error.
407///
408/// Upstream: `treams.special.spherical_jn` and `spherical_hankel1` with their `_d`
409/// derivatives.
410pub fn spherical_radial(l: u32, z: Complex, radial: Radial) -> Result<RadialJet> {
411    if l > MAX_ORDER.unsigned_abs() || !finite(z) {
412        return Err(Error::InvalidInput(
413            // 256 is MAX_ORDER.
414            "require finite argument and 0 <= l <= 256".into(),
415        ));
416    }
417    if radial == Radial::Regular && z.norm() < SERIES_RADIUS {
418        // j_l(z) = sum_q (-z^2/2)^q z^l / (q! (2l + 2q + 1)!!).
419        let double_factorial: f64 = (0..=l).map(|i| f64::from(2 * i + 1)).product();
420        let degree = f64::from(l);
421        return Ok(even_series(l, double_factorial.recip(), z, |q| {
422            -1.0 / (2.0 * q * (2.0 * (degree + q) + 1.0))
423        }));
424    }
425    if z == Complex::default() {
426        return Err(Error::InvalidInput(
427            "outgoing spherical waves are singular at zero".into(),
428        ));
429    }
430    let order = f64::from(l);
431    checked(jet_from_next_order(
432        order,
433        z,
434        spherical_pair(radial.into(), order, z)?,
435    ))
436}

438/// [`spherical_radial`] jets of every degree `0..=order` at one argument. Away from the
439/// regular power series, adjacent degrees share their AMOS evaluations: regular jets
440/// reuse single J evaluations and equal [`spherical_radial`] bit for bit, and singular
441/// jets come from one Hankel sequence instead of `order + 1` two-term sequences. Against
442/// 30-digit values at `|Im z| <= 5` the long sequence stays within 2e-13 relative up to
443/// order 80 (two-term sequences: 8e-14); below `Im z = 0` it loses up to a digit to
444/// them, still within 1e-13 of each jet's size up to order 40.
445pub(crate) fn spherical_radial_sequence(
446    order: u32,
447    z: Complex,
448    radial: Radial,
449) -> Result<Vec<RadialJet>> {
450    if order > MAX_ORDER.unsigned_abs() || !finite(z) {
451        return Err(Error::InvalidInput(
452            // 256 is MAX_ORDER.
453            "require finite argument and 0 <= l <= 256".into(),
454        ));
455    }
456    if radial == Radial::Regular && z.norm() < SERIES_RADIUS {
457        return (0..=order)
458            .map(|l| spherical_radial(l, z, radial))
459            .collect();
460    }
461    if z == Complex::default() {
462        return Err(Error::InvalidInput(
463            "outgoing spherical waves are singular at zero".into(),
464        ));
465    }
466    let values = match radial {
467        Radial::Regular => {
468            let prefactor = spherical_prefactor(z);
469            (0..=order + 1)
470                .map(|l| Ok(prefactor * bessel_raw(f64::from(l) + 0.5, z, Bessel::J, false)?))
471                .collect::<Result<Vec<_>>>()?
472        }
473        Radial::Singular => spherical_hankels(0, order + 1, z)?,
474    };
475    (0..=order)
476        .zip(values.iter().zip(values.iter().skip(1)))
477        .map(|(l, (&value, &next))| checked(jet_from_next_order(f64::from(l), z, [value, next])))
478        .collect()
479}

481/// Spherical Hankel functions `h_p(z)` of the first kind for `p = first..=last` at a
482/// nonzero argument, from one AMOS sequence.
483pub(crate) fn spherical_hankels(first: u32, last: u32, z: Complex) -> Result<Vec<Complex>> {
484    let count = usize::try_from(last.saturating_sub(first) + 1)
485        .map_err(|_| Error::InvalidInput("invalid radial order".into()))?;
486    let prefactor = spherical_prefactor(z);
487    Ok(sequence(Bessel::H1, f64::from(first) + 0.5, z, count)?
488        .into_iter()
489        .map(|h| prefactor * h)
490        .collect())
491}

493/// Integer-order cylindrical J (regular) or outgoing H1 (singular), with its first two
494/// derivatives.
495///
496/// Upstream: `treams.special.jv` and `hankel1` with their `_d` derivatives.
497pub fn cylindrical_radial(order: i32, z: Complex, radial: Radial) -> Result<RadialJet> {
498    if order.unsigned_abs() > MAX_ORDER.unsigned_abs() || !finite(z) {
499        return Err(Error::InvalidInput(
500            // 256 is MAX_ORDER.
501            "require finite argument and |m| <= 256".into(),
502        ));
503    }
504    let m = order.unsigned_abs();
505    let sign = if order < 0 && m % 2 == 1 { -1.0 } else { 1.0 };
506    if z == Complex::default() {
507        if radial == Radial::Singular {
508            return Err(Error::InvalidInput(
509                "outgoing cylindrical waves are singular at zero".into(),
510            ));
511        }
512        return Ok(RadialJet {
513            value: Complex::new(if m == 0 { 1.0 } else { 0.0 }, 0.0),
514            first: Complex::new(if m == 1 { 0.5 * sign } else { 0.0 }, 0.0),
515            second: Complex::new(
516                if m == 0 {
517                    -0.5
518                } else if m == 2 {
519                    0.25
520                } else {
521                    0.0
522                },
523                0.0,
524            ),
525        });
526    }
527    let degree = f64::from(m);
528    if radial == Radial::Regular && z.norm() < SERIES_RADIUS {
529        // J_m(z) = sum_q (-z^2/4)^q (z/2)^m / (q! (m + q)!).
530        let leading = sign * (-log_factorial(order.abs())).exp() / 2.0_f64.powf(degree);
531        return Ok(even_series(m, leading, z, |q| {
532            -1.0 / (4.0 * q * (degree + q))
533        }));
534    }
535    let [value, upper] = pair(sequence(radial.into(), degree, z, 2)?)?;
536    let value = sign * value;
537    let first = degree * value / z - sign * upper;
538    let second = (degree.powi(2) / z.powu(2) - 1.0) * value - first / z;
539    Ok(RadialJet {
540        value,
541        first,
542        second,
543    })
544}

546/// A spherical or cylindrical radial function of integer `order` at a jet argument:
547/// without derivatives a single value-only Bessel evaluation, otherwise the value
548/// and first derivative from [`spherical_radial`] or [`cylindrical_radial`].
549pub(crate) fn radial_jet<const N: usize>(
550    order: i32,
551    z: Jet<N>,
552    radial: Radial,
553    spherical: bool,
554) -> Result<Jet<N>> {
555    if N == 0 {
556        let value = bessel(f64::from(order), z.value, radial.into(), spherical, 0)?;
557        return Ok(Jet::constant(value));
558    }
559    let jet = if spherical {
560        spherical_radial(order.unsigned_abs(), z.value, radial)?
561    } else {
562        cylindrical_radial(order, z.value, radial)?
563    };
564    Ok(z.chain(jet.value, jet.first))
565}

594            ],
595            radial in radial(),

610    /// Each jet of a spherical sequence equals the single-degree jet: bit for bit for
611    /// regular waves, whose J evaluations are shared, and to 1e-13 of the jet's size for
612    /// singular waves (in the right half-plane of translation arguments), whose Hankel
613    /// functions come from one AMOS sequence. The jet's size covers values that are
614    /// small next to their neighbours, where both evaluations lose relative accuracy.
615    fn check_spherical_radial_sequence(
616        order: u32,
617        z: Complex,
618        radial: Radial,
619    ) -> Result<(), TestCaseError> {
620        let sequence = spherical_radial_sequence(order, z, radial).unwrap();
621        prop_assert_eq!(sequence.len(), usize::try_from(order + 1).unwrap());
622        for (l, jet) in (0..).zip(sequence) {
623            let single = spherical_radial(l, z, radial).unwrap();
624            let pairs = [
625                (jet.value, single.value),
626                (jet.first, single.first),
627                (jet.second, single.second),
628            ];
629            let size = single.value.norm() + single.first.norm() + single.second.norm();
630            for (d, (actual, expected)) in pairs.into_iter().enumerate() {
631                if radial == Radial::Regular {
632                    prop_assert_eq!(actual, expected, "degree {} derivative {}", l, d);
633                } else {
634                    let tolerance = 1e-13 * size;
635                    prop_assert_close!(
636                        actual,
637                        expected,
638                        tolerance,
639                        "degree {} derivative {}",
640                        l,
641                        d
642                    );
643                }
644            }
645        }
646        Ok(())
647    }

649    /// The cylindrical jet is a power series below `|z| = SERIES_RADIUS` (0.5) and a
650    /// two-term AMOS sequence above it. Both match single-order [`bessel`] values and
651    /// their order-difference or Hankel-sequence derivatives.
652    fn check_cylindrical_jet(order: i32, z: Complex, radial: Radial) -> Result<(), TestCaseError> {
653        let jet = cylindrical_radial(order, z, radial).unwrap();
654        let kind = Bessel::from(radial);
655        // Two-term J sequences differ from single J evaluations by up to 4e-13.
656        let tolerance = if radial == Radial::Regular && z.norm() < SERIES_RADIUS {
657            2e-13
658        } else {
659            2e-12
660        };
661        for (d, actual) in [0, 1, 2]
662            .into_iter()
663            .zip([jet.value, jet.first, jet.second])
664        {
665            let expected = bessel(f64::from(order), z, kind, false, d).unwrap();
666            prop_assert_close!(
667                actual,
668                expected,
669                tolerance * expected.norm(),
670                "derivative {}",
671                d
672            );
673        }
674        Ok(())
675    }

690    #[test]
691    fn regular_jets_at_the_origin_and_singular_outgoing_waves() {
692        // j_l^(k)(0) = k! [z^k] j_l: 1, 0, -1/3 for l = 0; 0, 1/3, 0 for l = 1; 0, 0, 2/15 for l = 2.
693        let exact = [
694            [1.0, 0.0, -1.0 / 3.0],
695            [0.0, 1.0 / 3.0, 0.0],
696            [0.0, 0.0, 2.0 / 15.0],
697        ];
698        for (l, expected) in (0..).zip(exact) {
699            let jet = spherical_radial(l, Complex::default(), Radial::Regular).unwrap();
700            for (actual, expected) in [jet.value, jet.first, jet.second].into_iter().zip(expected) {
701                assert!(
702                    (actual - expected).norm() < 1e-16,
703                    "l = {l}: {actual} != {expected}"
704                );
705            }
706        }
707        // J_m^(k)(0) = k! [z^k] J_m: 1, 0, -1/2 for m = 0; 0, 1/2, 0 for m = 1; 0, 0, 1/4 for m = 2.
708        let exact = [[1.0, 0.0, -0.5], [0.0, 0.5, 0.0], [0.0, 0.0, 0.25]];
709        for (m, expected) in (0..).zip(exact) {
710            for (order, sign) in [(m, 1.0), (-m, if m % 2 == 0 { 1.0 } else { -1.0 })] {
711                let jet = cylindrical_radial(order, Complex::default(), Radial::Regular).unwrap();
712                for (actual, expected) in
713                    [jet.value, jet.first, jet.second].into_iter().zip(expected)
714                {
715                    assert_eq!(actual, Complex::new(sign * expected, 0.0), "m = {order}");
716                }
717            }
718        }
719        assert!(spherical_radial(1, Complex::default(), Radial::Singular).is_err());
720        assert!(cylindrical_radial(1, Complex::default(), Radial::Singular).is_err());
721        // Only zero itself is singular. Near it the jets overflow instead, in AMOS (h_2 ~
722        // 3i/z^3) or in the derivatives' powers of 1/z: h_0 = -i e^(iz)/z has h_0'' ~
723        // 2i/z^3, beyond 1e308 below |z| = 1e-103.
724        let overflows = |result: crate::Result<()>, case: &str| {
725            let Err(Error::SpecialFunction(message)) = result else {
726                panic!("{case}: expected an overflow, got {result:?}");
727            };
728            assert!(message.contains("overflow"), "{case}: {message}");
729        };
730        for (radius, phase) in [(1e-200, -0.8), (1e-200, 2.9), (1e-120, 0.3), (1e-120, -2.2)] {
731            let tiny = Complex::from_polar(radius, phase);
732            for l in 0..=2 {
733                let case = format!("h_{l}({tiny:e})");
734                overflows(spherical_radial(l, tiny, Radial::Singular).map(drop), &case);
735                overflows(
736                    spherical_radial_sequence(l, tiny, Radial::Singular).map(drop),
737                    &case,
738                );
739            }
740        }
741    }
