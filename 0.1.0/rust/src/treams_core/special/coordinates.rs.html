1//! Coordinate and vector-component transformations in orthonormal local frames.
2//!
3//! Upstream: `treams.special.car2cyl`, `car2sph`, `cyl2car`, `cyl2sph`, `sph2car`,
4//! `sph2cyl`, `car2pol`, `pol2car` ([`point`]) and their `v`-prefixed vector twins such
5//! as `vcar2sph` ([`vector`]), from `_coord.pyx`. Spherical coordinates are
6//! `(r, theta, phi)` with the polar angle `theta`, cylindrical ones `(rho, phi, z)`, as
7//! in treams.
8#![allow(clippy::indexing_slicing)] // Fixed Cartesian triples.
9
10use crate::{Complex, Error, Result, numerics::finite};

12/// Coordinate conversion; vector positions are expressed in the input system.
13///
14/// Upstream: each variant names a function of `treams.special` for [`point`]
15/// (`CarToCyl` is `car2cyl`); [`vector`] takes the `v`-prefixed one (`vcar2cyl`).
16#[derive(Clone, Copy, Debug)]
17pub enum Transform {
18    /// Cartesian (x,y,z) to cylindrical (rho,phi,z).
19    CarToCyl,
20    /// Cartesian (x,y,z) to spherical (r,theta,phi).
21    CarToSph,
22    /// Cylindrical to Cartesian.
23    CylToCar,
24    /// Cylindrical to spherical.
25    CylToSph,
26    /// Spherical to Cartesian.
27    SphToCar,
28    /// Spherical to cylindrical.
29    SphToCyl,
30    /// Cartesian (x,y) to polar (rho,phi).
31    CarToPol,
32    /// Polar to Cartesian (x,y).
33    PolToCar,
34}
35use Transform::{CarToCyl, CarToPol, CarToSph, CylToCar, CylToSph, PolToCar, SphToCar, SphToCyl};
36
37impl Transform {
38    /// Number of coordinate or vector components.
39    #[must_use]
40    pub fn dimension(self) -> usize {
41        match self {
42            Self::CarToPol | Self::PolToCar => 2,
43            _ => 3,
44        }
45    }
46}

48/// `sqrt(a^2 + b^2)`: a fused multiply-add where the squares stay normal, `hypot`
49/// elsewhere.
50#[inline]
51fn radius(a: f64, b: f64) -> f64 {
52    let scale = a.abs().max(b.abs());
53    if (1e-150..=1e150).contains(&scale) {
54        a.mul_add(a, b * b).sqrt()
55    } else {
56        a.hypot(b)
57    }
58}

60/// Checks that every component is finite.
61fn validate(position: [f64; 3]) -> Result<()> {
62    if position.iter().any(|v| !v.is_finite()) {
63        return Err(Error::InvalidInput("coordinates must be finite".into()));
64    }
65    Ok(())
66}

68/// Cylindrical `(rho, phi, z)` of Cartesian `(x, y, z)`.
69#[inline]
70fn cylindrical_from_cartesian([a, b, c]: [f64; 3]) -> [f64; 3] {
71    [radius(a, b), b.atan2(a), c]
72}
73
74/// Cartesian `(x, y, z)` of cylindrical `(rho, phi, z)`.
75#[inline]
76fn cartesian_from_cylindrical([a, b, c]: [f64; 3]) -> [f64; 3] {
77    let (sin, cos) = b.sin_cos();
78    [a * cos, a * sin, c]
79}
80
81/// Spherical `(r, theta, phi)` of cylindrical `(rho, phi, z)`.
82#[inline]
83fn spherical_from_cylindrical([a, b, c]: [f64; 3]) -> [f64; 3] {
84    [radius(a, c), a.atan2(c), b]
85}
86
87/// Cylindrical `(rho, phi, z)` of spherical `(r, theta, phi)`.
88#[inline]
89fn cylindrical_from_spherical([a, b, c]: [f64; 3]) -> [f64; 3] {
90    let (sin, cos) = b.sin_cos();
91    [a * sin, c, a * cos]
92}

94/// Transform a point. Polar conversions ignore and return zero in the third slot.
95///
96/// Cartesian and spherical coordinates convert through cylindrical coordinates.
97///
98/// Upstream: `treams.special.car2cyl` and the other point conversions of
99/// [`Transform`]. Differences: non-finite coordinates give an error.
100// Always inlined: the ufunc loops pass a constant transform, which then selects
101// one chart at compile time (a call per point costs a third more).
102#[allow(clippy::inline_always)]
103#[inline(always)]
104pub fn point(position: [f64; 3], transform: Transform) -> Result<[f64; 3]> {
105    validate(position)?;
106    Ok(match transform {
107        CarToCyl => cylindrical_from_cartesian(position),
108        CarToPol => {
109            let [rho, phi, _] = cylindrical_from_cartesian(position);
110            [rho, phi, 0.0]
111        }
112        CylToCar => cartesian_from_cylindrical(position),
113        PolToCar => {
114            let [x, y, _] = cartesian_from_cylindrical(position);
115            [x, y, 0.0]
116        }
117        CylToSph => spherical_from_cylindrical(position),
118        SphToCyl => cylindrical_from_spherical(position),
119        // A cylindrical radius that overflows gives an infinite spherical radius with
120        // exact angles, as in treams.
121        CarToSph => spherical_from_cylindrical(cylindrical_from_cartesian(position)),
122        SphToCar => cartesian_from_cylindrical(cylindrical_from_spherical(position)),
123    })
124}

126/// The gradient with respect to the input coordinates of [`point`], given the gradient
127/// `cotangent` with respect to its output.
128///
129/// Angular derivatives are undefined on the axes; they give an error only where their
130/// `cotangent` component is nonzero.
131pub fn point_pullback(
132    position: [f64; 3],
133    transform: Transform,
134    cotangent: [f64; 3],
135) -> Result<[f64; 3]> {
136    validate(position)?;
137    validate(cotangent)?;
138    let [a, b, c] = position;
139    let undefined =
140        || Error::InvalidInput("coordinate derivative is undefined at this axis or origin".into());
141    Ok(match transform {
142        CarToCyl | CarToPol => {
143            let rho = radius(a, b);
144            if rho == 0.0 && (cotangent[0] != 0.0 || cotangent[1] != 0.0) {
145                return Err(undefined());
146            }
147            let (sp, cp) = if rho == 0.0 {
148                (0.0, 1.0)
149            } else {
150                (b / rho, a / rho)
151            };
152            let angular = if cotangent[1] == 0.0 {
153                0.0
154            } else {
155                cotangent[1] / rho
156            };
157            [
158                cotangent[0] * cp - angular * sp,
159                cotangent[0] * sp + angular * cp,
160                if matches!(transform, CarToPol) {
161                    0.0
162                } else {
163                    cotangent[2]
164                },
165            ]
166        }
167        CylToCar | PolToCar => {
168            let (sin, cos) = b.sin_cos();
169            [
170                cotangent[0] * cos + cotangent[1] * sin,
171                a * (-cotangent[0] * sin + cotangent[1] * cos),
172                if matches!(transform, PolToCar) {
173                    0.0
174                } else {
175                    cotangent[2]
176                },
177            ]
178        }
179        CylToSph => {
180            let r = radius(a, c);
181            if r == 0.0 && (cotangent[0] != 0.0 || cotangent[1] != 0.0) {
182                return Err(undefined());
183            }
184            if r == 0.0 {
185                [0.0, cotangent[2], 0.0]
186            } else {
187                [
188                    cotangent[0] * (a / r) + cotangent[1] * (c / r) / r,
189                    cotangent[2],
190                    cotangent[0] * (c / r) - cotangent[1] * (a / r) / r,
191                ]
192            }
193        }
194        SphToCyl => {
195            let (sin, cos) = b.sin_cos();
196            [
197                cotangent[0] * sin + cotangent[2] * cos,
198                a * (cotangent[0] * cos - cotangent[2] * sin),
199                cotangent[1],
200            ]
201        }
202        CarToSph => {
203            // Polar and azimuthal derivatives are undefined on the whole z axis. Check
204            // here: the intermediate cylindrical radial cotangent may underflow to zero.
205            if radius(a, b) == 0.0 && (cotangent[1] != 0.0 || cotangent[2] != 0.0) {
206                return Err(undefined());
207            }
208            let cylindrical = point(position, CarToCyl)?;
209            point_pullback(
210                position,
211                CarToCyl,
212                point_pullback(cylindrical, CylToSph, cotangent)?,
213            )?
214        }
215        SphToCar => {
216            let cylindrical = point(position, SphToCyl)?;
217            point_pullback(
218                position,
219                SphToCyl,
220                point_pullback(cylindrical, CylToCar, cotangent)?,
221            )?
222        }
223    })
224}

226/// Rotation of vector components into the output frame of `transform`, and its
227/// derivatives with respect to the (at most two) angles it depends on. Inverse
228/// transforms use the transposed matrices. Polar vectors use the first two
229/// rows/columns; no radial metric factors occur.
230#[inline]
231fn frame(position: [f64; 3], transform: Transform) -> ([[f64; 3]; 3], [[[f64; 3]; 3]; 2]) {
232    let [a, b, c] = position;
233    let inverse = matches!(transform, CylToCar | PolToCar | SphToCyl | SphToCar);
234    let zero = [[0.0; 3]; 3];
235    let (rotation, derivatives) = match transform {
236        CarToCyl | CarToPol | CylToCar | PolToCar => {
237            let phi = if inverse { b } else { b.atan2(a) };
238            let (sin, cos) = phi.sin_cos();
239            (
240                [[cos, sin, 0.0], [-sin, cos, 0.0], [0.0, 0.0, 1.0]],
241                [[[-sin, cos, 0.0], [-cos, -sin, 0.0], [0.0; 3]], zero],
242            )
243        }
244        CylToSph | SphToCyl => {
245            let theta = if inverse { b } else { a.atan2(c) };
246            let (sin, cos) = theta.sin_cos();
247            (
248                [[sin, 0.0, cos], [cos, 0.0, -sin], [0.0, 1.0, 0.0]],
249                [[[cos, 0.0, -sin], [-sin, 0.0, -cos], [0.0; 3]], zero],
250            )
251        }
252        CarToSph | SphToCar => {
253            let theta = if inverse { b } else { radius(a, b).atan2(c) };
254            let phi = if inverse { c } else { b.atan2(a) };
255            let (st, ct) = theta.sin_cos();
256            let (sp, cp) = phi.sin_cos();
257            (
258                [
259                    [st * cp, st * sp, ct],
260                    [ct * cp, ct * sp, -st],
261                    [-sp, cp, 0.0],
262                ],
263                [
264                    [[ct * cp, ct * sp, -st], [-st * cp, -st * sp, -ct], [0.0; 3]],
265                    [
266                        [-st * sp, st * cp, 0.0],
267                        [-ct * sp, ct * cp, 0.0],
268                        [-cp, -sp, 0.0],
269                    ],
270                ],
271            )
272        }
273    };
274    if inverse {
275        (transpose(rotation), derivatives.map(transpose))
276    } else {
277        (rotation, derivatives)
278    }
279}

281/// The transpose of a 3 x 3 matrix.
282fn transpose(matrix: [[f64; 3]; 3]) -> [[f64; 3]; 3] {
283    std::array::from_fn(|i| std::array::from_fn(|j| matrix[j][i]))
284}

286/// The product of a real 3 x 3 matrix and a complex vector.
287fn apply(matrix: [[f64; 3]; 3], vector: [Complex; 3]) -> [Complex; 3] {
288    matrix.map(|row| row.into_iter().zip(vector).map(|(a, b)| a * b).sum())
289}

291/// Transform vector components at a point in the source coordinate system.
292///
293/// Upstream: `treams.special.vcar2cyl` and the other vector conversions of
294/// [`Transform`]. Differences: non-finite components or coordinates give an error.
295// Always inlined, as `point`: a constant transform selects one frame.
296#[allow(clippy::inline_always)]
297#[inline(always)]
298pub fn vector(
299    value: [Complex; 3],
300    position: [f64; 3],
301    transform: Transform,
302) -> Result<[Complex; 3]> {
303    validate(position)?;
304    if value.into_iter().any(|v| !finite(v)) {
305        return Err(Error::InvalidInput(
306            "vector components must be finite".into(),
307        ));
308    }
309    // Apply `frame`'s rotation without its structural zeros and ones: the dense
310    // product costs a third more per point, since x * 0.0 cannot be folded.
311    let r = frame(position, transform).0;
312    let [x, y, z] = value;
313    let row = |i: usize| x * r[i][0] + y * r[i][1];
314    Ok(match transform {
315        CarToCyl | CarToPol | CylToCar | PolToCar => [row(0), row(1), z],
316        CylToSph => [x * r[0][0] + z * r[0][2], x * r[1][0] + z * r[1][2], y],
317        SphToCyl => [row(0), z, row(2)],
318        CarToSph | SphToCar => [row(0) + z * r[0][2], row(1) + z * r[1][2], row(2)],
319    })
320}

322/// The gradients with respect to the vector components and the position of [`vector`],
323/// given the gradient `cotangent` with respect to its output components.
324pub fn vector_pullback(
325    value: [Complex; 3],
326    position: [f64; 3],
327    transform: Transform,
328    cotangent: [Complex; 3],
329) -> Result<([Complex; 3], [f64; 3])> {
330    validate(position)?;
331    if value.into_iter().chain(cotangent).any(|v| !finite(v)) {
332        return Err(Error::InvalidInput(
333            "vector and cotangent must be finite".into(),
334        ));
335    }
336    let (rotation, derivatives) = frame(position, transform);
337    let angles = derivatives.map(|d| {
338        apply(d, value)
339            .into_iter()
340            .zip(cotangent)
341            .map(|(d, g)| (g.conj() * d).re)
342            .sum::<f64>()
343    });
344    let point = match transform {
345        CarToCyl | CarToPol | CylToSph => {
346            point_pullback(position, transform, [0.0, angles[0], 0.0])?
347        }
348        CarToSph => point_pullback(position, transform, [0.0, angles[0], angles[1]])?,
349        CylToCar | PolToCar | SphToCyl => [0.0, angles[0], 0.0],
350        SphToCar => [0.0, angles[0], angles[1]],
351    };
352    Ok((apply(transpose(rotation), cotangent), point))
353}

437    /// The point and vector components of `transform`'s input system, restricted to
438    /// the plane for polar charts.
439    fn input(
440        position: [f64; 3],
441        value: [Complex; 3],
442        transform: Transform,
443    ) -> ([f64; 3], [Complex; 3]) {
444        let source = match transform {
445            CylToCar | CylToSph => Some(CarToCyl),
446            SphToCar | SphToCyl => Some(CarToSph),
447            PolToCar => Some(CarToPol),
448            CarToCyl | CarToSph | CarToPol => None,
449        };
450        let planar = transform.dimension() == 2;
451        let (position, value) = if planar {
452            (
453                [position[0], position[1], 0.0],
454                [value[0], value[1], Complex::default()],
455            )
456        } else {
457            (position, value)
458        };
459        match source {
460            Some(source) => (
461                point(position, source).unwrap(),
462                vector(value, position, source).unwrap(),
463            ),
464            None => (position, value),
465        }
466    }

468    /// Per-slot tolerances of coordinates in the output system of `transform` for
469    /// points of norm `scale`: relative for lengths and absolute for angles.
470    fn slot_tolerances(transform: Transform, scale: f64) -> [f64; 3] {
471        let length = 1e-14 * scale + f64::MIN_POSITIVE;
472        match transform {
473            CarToCyl | SphToCyl | CarToPol => [length, 1e-14, length],
474            CarToSph | CylToSph => [length, 1e-14, 1e-14],
475            CylToCar | SphToCar | PolToCar => [length; 3],
476        }
477    }

479    /// Every chart inverts, `CylToSph ∘ CarToCyl = CarToSph` and
480    /// `CylToCar ∘ SphToCyl = SphToCar` exactly, and vector frames are orthogonal:
481    /// they invert, preserve the norm and compose.
482    #[allow(clippy::float_cmp)] // Chart compositions are exact.
483    fn check_charts(cartesian: [f64; 3], value: [Complex; 3]) -> Result<(), TestCaseError> {
484        let scale = cartesian.iter().map(|v| v.abs()).fold(0.0, f64::max);
485        for (forward, inverse) in PAIRS {
486            for (from, to) in [(forward, inverse), (inverse, forward)] {
487                let (position, value) = input(cartesian, value, from);
488                let image = point(position, from).unwrap();
489                let back = point(image, to).unwrap();
490                let tolerances = slot_tolerances(to, scale);
491                for ((back, position), tolerance) in back.into_iter().zip(position).zip(tolerances)
492                {
493                    prop_assert_close!(back, position, tolerance, "{from:?} then {to:?}");
494                }
495                let rotated = vector(value, position, from).unwrap();
496                let norm = re_dot(value, value);
497                prop_assert_close!(re_dot(rotated, rotated), norm, 1e-14 * norm, "{from:?}");
498                let restored = vector(rotated, image, to).unwrap();
499                prop_assert_close!(restored, value, 1e-14 * norm.sqrt(), "{from:?} then {to:?}");
500            }
501        }
502        let cylindrical = point(cartesian, CarToCyl).unwrap();
503        let spherical = point(cartesian, CarToSph).unwrap();
504        prop_assert_eq!(point(cylindrical, CylToSph).unwrap(), spherical);
505        prop_assert_eq!(
506            point(point(spherical, SphToCyl).unwrap(), CylToCar).unwrap(),
507            point(spherical, SphToCar).unwrap()
508        );
509        let tolerance = 1e-14 * re_dot(value, value).sqrt();
510        let via_cylinder = vector(
511            vector(value, cartesian, CarToCyl).unwrap(),
512            cylindrical,
513            CylToSph,
514        );
515        prop_assert_close!(
516            via_cylinder.unwrap(),
517            vector(value, cartesian, CarToSph).unwrap(),
518            tolerance
519        );
520        let components = vector(value, spherical, SphToCyl).unwrap();
521        let cylinder = point(spherical, SphToCyl).unwrap();
522        let via_cylinder = vector(components, cylinder, CylToCar).unwrap();
523        prop_assert_close!(
524            via_cylinder,
525            vector(value, spherical, SphToCar).unwrap(),
526            tolerance
527        );
528        Ok(())
529    }

531    /// Point pullbacks match central differences and invert with their chart, and
532    /// vector pullbacks are the transposed rotation (exactly, since `vector` is linear
533    /// in its components) plus a position term matching central differences.
534    fn check_adjoints(
535        cartesian: [f64; 3],
536        value: [Complex; 3],
537        cotangent: [Complex; 3],
538    ) -> Result<(), TestCaseError> {
539        let h = 1e-6;
540        for (forward, inverse) in PAIRS {
541            for (transform, other) in [(forward, inverse), (inverse, forward)] {
542                let (position, value) = input(cartesian, value, transform);
543                let planar = transform.dimension() == 2;
544                let cotangent = if planar {
545                    [cotangent[0], cotangent[1], Complex::default()]
546                } else {
547                    cotangent
548                };
549                let g = cotangent.map(|g| g.re);
550                let point_g = point_pullback(position, transform, g).unwrap();
551                let (vector_g, position_g) =
552                    vector_pullback(value, position, transform, cotangent).unwrap();
553                let rotated = vector(value, position, transform).unwrap();
554                prop_assert_close!(
555                    re_dot(cotangent, rotated),
556                    re_dot(vector_g, value),
557                    1e-14 * (1.0 + re_dot(cotangent, cotangent) + re_dot(value, value)),
558                    "{transform:?}"
559                );
560                let image = point(position, transform).unwrap();
561                let chained = point_pullback(
562                    position,
563                    transform,
564                    point_pullback(image, other, g).unwrap(),
565                );
566                prop_assert_close!(
567                    chained.unwrap(),
568                    g,
569                    1e-13 * (1.0 + g.iter().map(|v| v.abs()).sum::<f64>()),
570                    "{transform:?}"
571                );
572                for axis in 0..transform.dimension() {
573                    let step = |t: f64| {
574                        let mut shifted = position;
575                        shifted[axis] += t * h;
576                        shifted
577                    };
578                    let numeric = (point(step(1.0), transform).unwrap().into_iter())
579                        .zip(point(step(-1.0), transform).unwrap())
580                        .zip(g)
581                        .map(|((p, m), g)| g * (p - m) / (2.0 * h))
582                        .sum::<f64>();
583                    prop_assert_close!(
584                        point_g[axis],
585                        numeric,
586                        2e-8 * (1.0 + numeric.abs()),
587                        "{transform:?}"
588                    );
589                    let numeric = re_dot(
590                        cotangent,
591                        (0..3).map(|c| {
592                            (vector(value, step(1.0), transform).unwrap()[c]
593                                - vector(value, step(-1.0), transform).unwrap()[c])
594                                / (2.0 * h)
595                        }),
596                    );
597                    prop_assert_close!(
598                        position_g[axis],
599                        numeric,
600                        2e-8 * (1.0 + numeric.abs()),

602                    );
603                }
604            }
605        }
606        Ok(())
607    }
