11/// Normalized measures of a global helicity T-matrix.
12#[derive(Clone, Copy, Debug)]
13pub enum Metric {
14    /// Absorption circular dichroism in a real embedding medium. Upstream:
15    /// `treams.TMatrix.cd`.
16    CircularDichroism,
17    /// Fraction of scattering that changes helicity. Upstream: `treams.TMatrix.db`.
18    DualityBreaking,
19    /// Electromagnetic chirality from the singular values of the four helicity blocks.
20    /// Upstream: `treams.TMatrix.chi`.
21    Chirality,
22}
23
24/// What [`metric`] saves for its pullback: the gradient itself, since the value is a
25/// real scalar; no singular vectors are kept.
26#[derive(Debug)]
27pub struct MetricResidual {
28    /// The matrix gradient, or why it is undefined: chirality may be zero where its
29    /// derivative is undefined.
30    gradient: Result<DMatrix<Complex>>,
31    ks_gradient: [f64; 2],
32    dimension: usize,
33}
34
35/// Gradients of the inputs of [`metric`], in the order of its arguments.
36#[derive(Clone, Debug)]
37pub struct MetricGradient {
38    /// Gradient of the helicity T-matrix.
39    pub matrix: DMatrix<Complex>,
40    /// Gradients of the real embedding wavenumbers; zero except for circular
41    /// dichroism.
42    pub ks: [f64; 2],
43}
44
45impl MetricResidual {
46    /// Gradients of the matrix and the embedding wavenumbers from `cotangent`, the
47    /// gradient of a real loss with respect to the metric.
48    ///
49    /// The pullback scales the saved gradient on the calling thread, so the Rayon pool
50    /// does not change it.
51    pub fn pullback(self, cotangent: f64) -> Result<MetricGradient> {
52        if !cotangent.is_finite() {
53            return Err(Error::InvalidInput(
54                "metric cotangent must be finite".into(),
55            ));
56        }
57        if cotangent == 0.0 {
58            return Ok(MetricGradient {
59                matrix: DMatrix::zeros(self.dimension, self.dimension),
60                ks: [0.0; 2],
61            });
62        }
63        Ok(MetricGradient {
64            matrix: self.gradient?.map(|z| z * cotangent),
65            ks: self.ks_gradient.map(|k| k * cotangent),
66        })
67    }
68}

70/// One [`Metric`] of a global helicity T-matrix.
71///
72/// `pol` is the polarization index of each row and `ks` holds the negative and positive
73/// helicity wavenumbers of the embedding, which enter circular dichroism only.
74///
75/// Upstream: `treams.TMatrix.cd`, `treams.TMatrix.db` and `treams.TMatrix.chi`.
76/// Differences: circular dichroism requires positive real wavenumbers where treams
77/// requires a real material; the other two scale the matrix to a largest entry of 1
78/// first; the chirality gradient is an error at zero helicity contrast.
79pub fn metric(
80    matrix: &DMatrix<Complex>,
81    pol: &[u8],
82    ks: [f64; 2],
83    kind: Metric,
84) -> Result<(f64, MetricResidual)> {
85    let dimension = matrix.nrows();
86    if dimension == 0
87        || !matrix.is_square()
88        || pol.len() != dimension
89        || pol.iter().any(|&p| p > 1)
90        || matrix.iter().any(|&z| !finite(z))
91        || ks.iter().any(|k| !k.is_finite() || *k <= 0.0)
92    {
93        return Err(Error::InvalidInput("require a finite square helicity matrix, matching polarizations and positive real wavenumbers".into()));
94    }
95    if matches!(kind, Metric::CircularDichroism) {
96        return circular_dichroism(matrix, pol, ks);
97    }
98    // Normalized metrics are invariant to a common matrix scale. Work at unit
99    // scale so norm squares do not overflow or underflow for weak scattering.
100    let scale = matrix.iter().map(|z| z.norm()).fold(0.0, f64::max);
101    if scale == 0.0 {
102        return Err(Error::InvalidInput(
103            "normalized metric requires nonzero scattering".into(),
104        ));
105    }
106    let matrix = matrix.map(|z| z / scale);
107    let (value, gradient) = if matches!(kind, Metric::DualityBreaking) {
108        let (value, gradient) = duality_breaking(&matrix, pol);
109        (value, Ok(gradient))
110    } else {
111        chirality(&matrix, pol)?
112    };
113    Ok((
114        value,
115        MetricResidual {
116            gradient: gradient.map(|g| g.map(|z| z / scale)),
117            ks_gradient: [0.0; 2],
118            dimension,
119        },
120    ))
121}

123/// Absorption circular dichroism with its matrix and wavenumber gradients.
124fn circular_dichroism(
125    matrix: &DMatrix<Complex>,
126    pol: &[u8],
127    ks: [f64; 2],
128) -> Result<(f64, MetricResidual)> {
129    let dimension = matrix.nrows();
130    let weights: Vec<_> = pol.iter().map(|&p| ks[usize::from(p)].powi(-2)).collect();
131    let mut absorption = [0.0; 2];
132    for j in 0..dimension {
133        absorption[usize::from(pol[j])] -= matrix[(j, j)].re * weights[j]
134            + (0..dimension)
135                .map(|i| matrix[(i, j)].norm_sqr() * weights[i])
136                .sum::<f64>();
137    }
138    let total = absorption.iter().sum::<f64>();
139    if total == 0.0 || !total.is_finite() {
140        return Err(Error::InvalidInput(
141            "circular dichroism requires nonzero total absorption".into(),
142        ));
143    }
144    let value = (absorption[1] - absorption[0]) / total;
145    let g = [
146        -2.0 * absorption[1] / total / total,
147        2.0 * absorption[0] / total / total,
148    ];
149    let mut gradient = DMatrix::from_fn(dimension, dimension, |i, j| {
150        -2.0 * g[usize::from(pol[j])] * weights[i] * matrix[(i, j)]
151    });
152    let mut gks = [0.0; 2];
153    for i in 0..dimension {
154        let p = usize::from(pol[i]);
155        gradient[(i, i)] -= g[p] * weights[i];
156        gks[p] += 2.0 * weights[i] / ks[p]
157            * (g[p] * matrix[(i, i)].re
158                + (0..dimension)
159                    .map(|j| g[usize::from(pol[j])] * matrix[(i, j)].norm_sqr())
160                    .sum::<f64>());
161    }
162    Ok((
163        value,
164        MetricResidual {
165            gradient: Ok(gradient),
166            ks_gradient: gks,
167            dimension,
168        },
169    ))
170}

172/// The fraction of scattering that flips helicity and its gradient, for a matrix at
173/// unit scale.
174fn duality_breaking(matrix: &DMatrix<Complex>, pol: &[u8]) -> (f64, DMatrix<Complex>) {
175    let dimension = matrix.nrows();
176    let norm = matrix.norm_squared();
177    let flipped = (0..dimension)
178        .flat_map(|j| (0..dimension).map(move |i| (i, j)))
179        .filter(|&(i, j)| pol[i] != pol[j])
180        .map(|(i, j)| matrix[(i, j)].norm_sqr())
181        .sum::<f64>();
182    let value = flipped / norm;
183    let gradient = DMatrix::from_fn(dimension, dimension, |i, j| {
184        matrix[(i, j)] * (2.0 * (f64::from(pol[i] != pol[j]) - value) / norm)
185    });
186    (value, gradient)
187}

189/// One helicity block of the chirality metric and its place in the singular-value
190/// differences.
191#[derive(Clone, Copy)]
192struct HelicityBlock {
193    /// Helicity of the rows.
194    row: usize,
195    /// Helicity of the columns.
196    col: usize,
197    /// The difference of spectra the block enters: 0 is `spectrum(1, 1) -
198    /// spectrum(0, 0)`, 1 is `spectrum(1, 0) - spectrum(0, 1)`.
199    difference: usize,
200    /// The sign of the block's spectrum in that difference.
201    sign: f64,
202}
203
204/// The four helicity blocks, ordered so that block `3 - p` minus block `p` is
205/// difference `p`, as treams pairs them in `TMatrix.chi`.
206const CHIRALITY_BLOCKS: [HelicityBlock; 4] = [
207    HelicityBlock {
208        row: 0,
209        col: 0,
210        difference: 0,
211        sign: -1.0,
212    },
213    HelicityBlock {
214        row: 0,
215        col: 1,
216        difference: 1,
217        sign: -1.0,
218    },
219    HelicityBlock {
220        row: 1,
221        col: 0,
222        difference: 1,
223        sign: 1.0,
224    },
225    HelicityBlock {
226        row: 1,
227        col: 1,
228        difference: 0,
229        sign: 1.0,
230    },
231];

233/// Electromagnetic chirality from the singular values of the four helicity blocks,
234/// for a matrix at unit scale. Its gradient is undefined at zero contrast.
235fn chirality(matrix: &DMatrix<Complex>, pol: &[u8]) -> Result<(f64, Result<DMatrix<Complex>>)> {
236    let indices: [Vec<_>; 2] = std::array::from_fn(|p| {
237        pol.iter()
238            .enumerate()
239            .filter_map(|(i, &q)| (usize::from(q) == p).then_some(i))
240            .collect()
241    });
242    let count = indices[0].len();
243    if count == 0 || count != indices[1].len() {
244        return Err(Error::InvalidInput(
245            "chirality requires equally sized nonempty helicity groups".into(),
246        ));
247    }
248    let block = |row: usize, col: usize| {
249        DMatrix::from_fn(count, count, |i, j| {
250            matrix[(indices[row][i], indices[col][j])]
251        })
252    };
253    let spectra = CHIRALITY_BLOCKS
254        .iter()
255        .map(|&HelicityBlock { row, col, .. }| crate::linalg::svdvals(&block(row, col)))
256        .collect::<Result<Vec<_>>>()?;
257    let differences: [Vec<_>; 2] = std::array::from_fn(|p| {
258        spectra[3 - p]
259            .values()
260            .iter()
261            .zip(spectra[p].values())
262            .map(|(a, b)| a - b)
263            .collect()
264    });
265    let contrast = differences
266        .iter()
267        .flatten()
268        .map(|x| x * x)
269        .sum::<f64>()
270        .sqrt();
271    let value = contrast / matrix.norm_squared().sqrt();
272    // The value stays available where the gradient is undefined.
273    let gradient = if contrast == 0.0 {
274        Err(Error::InvalidInput(
275            "chirality is not differentiable at zero contrast".into(),
276        ))
277    } else {
278        chirality_gradient(matrix, &indices, spectra, &differences, contrast)
279    };
280    Ok((value, gradient))
281}

283/// The chirality gradient through the singular values of the helicity blocks.
284fn chirality_gradient(
285    matrix: &DMatrix<Complex>,
286    indices: &[Vec<usize>; 2],
287    spectra: Vec<SvdvalsResidual>,
288    differences: &[Vec<f64>; 2],
289    contrast: f64,
290) -> Result<DMatrix<Complex>> {
291    let norm = matrix.norm_squared();
292    let value = contrast / norm.sqrt();
293    let mut gradient = matrix.map(|z| -value * z / norm);
294    for (
295        residual,
296        &HelicityBlock {
297            row,
298            col,
299            difference,
300            sign,
301        },
302    ) in spectra.into_iter().zip(&CHIRALITY_BLOCKS)
303    {
304        let weights: Vec<_> = differences[difference]
305            .iter()
306            .map(|x| sign * x / contrast / norm.sqrt())
307            .collect();
308        let block = residual.pullback(&weights)?;
309        for (j, &target_col) in indices[col].iter().enumerate() {
310            for (i, &target_row) in indices[row].iter().enumerate() {
311                gradient[(target_row, target_col)] += block[(i, j)];
312            }
313        }
314    }
315    Ok(gradient)
