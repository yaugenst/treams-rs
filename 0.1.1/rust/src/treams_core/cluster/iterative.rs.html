30/// The matrix-free counterpart of [`sphere_cluster`](super::sphere_cluster): the same
31/// spheres, solved with GMRES for the requested incident columns only.
32///
33/// It keeps the geometry and the local Mie coefficients, never a particle-pair matrix.
34///
35/// treams-rs extension; treams has no matrix-free cluster solver.
36#[derive(Debug)]
37pub struct IterativeSphereCluster {
38    k0: f64,
39    radii: Vec<f64>,
40    positions: Vec<[f64; 3]>,
41    degrees: Vec<Vec<MieResidual>>,
42    mode_degrees: Vec<usize>,
43    plan: TranslationPlan,
44    dimension: usize,
45}
46
47impl IterativeSphereCluster {
48    /// Homogeneous nonmagnetic spheres in vacuum, with the arguments of
49    /// [`sphere_cluster`](super::sphere_cluster). Each particle's modes follow
50    /// [`sw::modes`] in helicity convention.
51    pub fn new(
52        lmax: u32,
53        k0: f64,
54        radii: &[f64],
55        epsilon: &[Complex],
56        positions: &[[f64; 3]],
57    ) -> Result<Self> {
58        crate::cluster::validate_spheres(k0, radii, epsilon, positions)?;
59        let modes = sw::modes(lmax)?;
60        let dimension = modes
61            .len()
62            .checked_mul(radii.len())
63            .ok_or_else(|| Error::InvalidInput("cluster is too large".into()))?;
64        let degrees = radii
65            .iter()
66            .zip(epsilon)
67            .map(|(&radius, &epsilon)| {
68                let materials = crate::cluster::vacuum_sphere(epsilon);
69                (1..=lmax)
70                    .map(|l| mie(l, &[k0 * radius], &materials))
71                    .collect::<Result<Vec<_>>>()
72            })
73            .collect::<Result<Vec<_>>>()?;
74        let count = degrees.first().map_or(0, Vec::len);
75        Ok(Self {
76            k0,
77            radii: radii.to_vec(),
78            positions: positions.to_vec(),
79            degrees,
80            mode_degrees: crate::tmatrix::block_degrees(count).collect(),
81            plan: TranslationPlan::between(&modes, &modes, true)?,
82            dimension,
83        })
84    }
85
86    /// Number of local multipoles across all particles.
87    #[must_use]
88    pub const fn dimension(&self) -> usize {
89        self.dimension
90    }
91
92    /// `T input`, or `Tᴴ input` if `adjoint`, from the 2 x 2 Mie block of each degree
93    /// and order, for one or more column-major vectors.
94    ///
95    /// [`sphere`](crate::tmatrix::sphere) places the same blocks in a dense T-matrix
96    /// for [`sphere_cluster`](super::sphere_cluster); this applies them in place.
97    fn local(&self, input: &[Complex], adjoint: bool) -> Vec<Complex> {
98        let mut output = vec![Complex::default(); input.len()];
99        let modes = self.modes_per_particle();
100        for (output, input) in output
101            .chunks_exact_mut(self.dimension)
102            .zip(input.chunks_exact(self.dimension))
103        {
104            for (particle, degrees) in self.degrees.iter().enumerate() {
105                for (block, &degree) in self.mode_degrees.iter().enumerate() {
106                    let matrix = to_mode_order(degrees[degree].value());
107                    let offset = particle * modes + block * 2;
108                    for i in 0..2 {
109                        for j in 0..2 {
110                            let t = if adjoint {
111                                matrix[(j, i)].conj()
112                            } else {
113                                matrix[(i, j)]
114                            };
115                            output[offset + i] += t * input[offset + j];
116                        }
117                    }
118                }
119            }
120        }
121        output
122    }
123
124    /// Multipoles per particle.
125    const fn modes_per_particle(&self) -> usize {
126        self.mode_degrees.len() * 2
127    }
128
129    /// The particle-pair coupling `C`, or `Cᴴ`, applied to `columns` column-major
130    /// vectors. Each translation block is evaluated once for all columns.
131    fn coupling(&self, input: &[Complex], columns: usize, adjoint: bool) -> Result<Vec<Complex>> {
132        let modes = self.modes_per_particle();
133        let particles = self.positions.len();
134        let input = MatRef::from_column_major_slice(input, self.dimension, columns);
135        let rows: Vec<Vec<Complex>> = crate::threads::install(|| {
136            (0..particles)
137                .into_par_iter()
138                .map(|i| -> Result<Vec<Complex>> {
139                    // Rows of particle `i`, column-major over the input columns.
140                    let mut output = vec![Complex::default(); modes * columns];
141                    for j in (0..particles).filter(|&j| j != i) {
142                        let (to, from) = if adjoint { (j, i) } else { (i, j) };
143                        let displacement = std::array::from_fn(|axis| {
144                            self.positions[to][axis] - self.positions[from][axis]
145                        });
146                        self.plan.apply(
147                            Complex::new(self.k0, 0.0),
148                            displacement,
149                            Radial::Singular,
150                            input.subrows(j * modes, modes),
151                            MatMut::from_column_major_slice_mut(&mut output, modes, columns),
152                            adjoint,
153                        )?;
154                    }
155                    Ok(output)
156                })
157                .collect::<Result<_>>()
158        })?;
159        let mut output = vec![Complex::default(); self.dimension * columns];
160        for (i, rows) in rows.iter().enumerate() {
161            for (column, values) in rows.chunks_exact(modes).enumerate() {
162                output[column * self.dimension + i * modes..][..modes].copy_from_slice(values);
163            }
164        }
165        Ok(output)
166    }
167
168    /// `(I - T C) input`, or `(I - T C)ᴴ input = (I - Cᴴ Tᴴ) input` if `adjoint`: the
169    /// operator that GMRES inverts.
170    fn apply(&self, input: &[Complex], columns: usize, adjoint: bool) -> Result<Vec<Complex>> {
171        let product = if adjoint {
172            self.coupling(&self.local(input, true), columns, true)?
173        } else {
174            self.local(&self.coupling(input, columns, false)?, false)
175        };
176        Ok(input.iter().zip(product).map(|(x, y)| x - y).collect())
177    }
178
179    /// Solve `(I - T C) X = rhs`, or `(I - T C)ᴴ X = rhs` if `adjoint`, one column of
180    /// `rhs` independently, sharing pair translations across bounded column batches.
181    fn solve_rhs(
182        &self,
183        rhs: &DMatrix<Complex>,
184        options: GmresOptions,
185        adjoint: bool,
186    ) -> Result<IterativeSolution> {
187        let mut value = DMatrix::zeros(self.dimension, rhs.ncols());
188        let mut convergence = Vec::with_capacity(rhs.ncols());
189        // Retain at most eight independent Krylov bases regardless of the number of
190        // illuminations. Each costly pair translation serves all active columns.
191        const BATCH_COLUMNS: usize = 8;
192        for (input, output) in rhs.as_slice().chunks(self.dimension * BATCH_COLUMNS).zip(
193            value
194                .as_mut_slice()
195                .chunks_mut(self.dimension * BATCH_COLUMNS),
196        ) {
197            let columns = input.len() / self.dimension;
198            let (answer, reports) = gmres_batch(input, columns, options, |x, columns| {
199                self.apply(x, columns, adjoint)
200            })?;
201            output.copy_from_slice(&answer);
202            convergence.extend(reports);
203        }
204        Ok(IterativeSolution { value, convergence })
205    }

207    /// Solve `(I - T C) X = T B` for only the columns `B` supplied in `incident`.
208    /// An unconverged solve returns an error containing the measured residual.
209    pub fn solve(
210        &self,
211        incident: &DMatrix<Complex>,
212        options: GmresOptions,
213    ) -> Result<IterativeSolution> {
214        options.validate()?;
215        if incident.nrows() != self.dimension
216            || incident.ncols() == 0
217            || incident.iter().any(|&z| !finite(z))
218        {
219            return Err(Error::InvalidInput(
220                "incident coefficients must be finite, with one row per local multipole and at least one column".into(),
221            ));
222        }
223        let rhs = DMatrix::from_vec(
224            self.dimension,
225            incident.ncols(),
226            self.local(incident.as_slice(), false),
227        );
228        self.solve_rhs(&rhs, options, false)
229    }

231    /// Solve like [`solve`](Self::solve) and keep what the pullback needs. The
232    /// residual shares this cluster through the [`Arc`]; the Krylov vectors are dropped.
233    pub fn record(
234        self: &Arc<Self>,
235        incident: DMatrix<Complex>,
236        options: GmresOptions,
237    ) -> Result<IterativeResidual> {
238        let solution = self.solve(&incident, options)?;
239        Ok(IterativeResidual {
240            operator: Arc::clone(self),
241            incident,
242            options,
243            solution,
244        })
245    }

258/// Gradients of an [`IterativeResidual`], without a dense coupling matrix or its
259/// gradient.
260///
261/// The fields follow the inputs: `cluster` holds the gradients of the arguments of
262/// [`IterativeSphereCluster::new`] in their order (`k0`, `radii`, `epsilon`,
263/// `positions`), then come those of the incident fields of
264/// [`record`](IterativeSphereCluster::record) and the convergence of the adjoint
265/// solves.
266#[derive(Debug)]
267pub struct IterativeGradient {
268    /// Gradients of the vacuum wavenumber, radii, permittivities and positions.
269    pub cluster: SphereClusterGradient,
270    /// Gradient of the incident fields.
271    pub incident: DMatrix<Complex>,
272    /// Independent convergence certificates of the adjoint solves.
273    pub convergence: Vec<Convergence>,
274}
275
276impl IterativeResidual {
277    /// The forward solution and its convergence certificates; the pullback reads the
278    /// solution.
279    #[must_use]
280    pub const fn solution(&self) -> &IterativeSolution {
281        &self.solution
282    }
283
284    /// The shape of the solution: local multipoles and incident columns.
285    #[must_use]
286    pub fn shape(&self) -> (usize, usize) {
287        self.solution.value.shape()
288    }
289
290    /// Gradients of the cluster and the incident fields from `cotangent`, the gradient
291    /// of a real loss with respect to the solution.
292    ///
293    /// One adjoint GMRES solve per column gives `Y = (I - T C)⁻ᴴ G`; each particle pair
294    /// then contributes to the position and `k0` gradients. Those contributions add in
295    /// chunks of particles fixed by the particle count, and the chunk sums in chunk
296    /// order; the radius and permittivity gradients add in particle order. The thread
297    /// count changes none of the gradients.
298    pub fn pullback(self, cotangent: &DMatrix<Complex>) -> Result<IterativeGradient> {
299        if cotangent.shape() != self.shape() || cotangent.iter().any(|&z| !finite(z)) {
300            return Err(Error::InvalidInput("invalid illumination cotangent".into()));
301        }
302        let operator = self.operator;
303        let adjoint = operator.solve_rhs(cotangent, self.options, true)?;
304        let columns = self.incident.ncols();
305        let value = &self.solution.value;
306        // The local matrices multiply the response B + C X; the coupling sees T^H Y.
307        let mut response = self.incident;
308        let scattered = operator.coupling(value.as_slice(), columns, false)?;
309        for (response, scattered) in response.iter_mut().zip(scattered) {
310            *response += scattered;
311        }
312        let incident = DMatrix::from_vec(
313            operator.dimension,
314            columns,
315            operator.local(adjoint.value.as_slice(), true),
316        );
317        let (positions, k0) = operator.pair_gradients(&incident, value)?;
318        let mut cluster = SphereClusterGradient {
319            k0,
320            radii: vec![0.0; operator.positions.len()],
321            epsilon: vec![Complex::default(); operator.positions.len()],
322            positions,
323        };
324        operator.local_gradients(&adjoint.value, &response, &mut cluster)?;
325        Ok(IterativeGradient {
326            cluster,
327            incident,
328            convergence: adjoint.convergence,
329        })
330    }
331}

333impl IterativeSphereCluster {
334    /// Position and wavenumber gradients through the coupling `C`, whose cotangent is
335    /// `(Tᴴ Y) Xᴴ` for `local_adjoint = Tᴴ Y` and the solution `value = X`, summed one
336    /// particle-pair block at a time.
337    ///
338    /// Chunks of consecutive particles `i`, fixed by the particle count, add their
339    /// pairs `(i, j)` in order into one partial sum each, and the partial sums add in
340    /// chunk order (`numerics::parallel::try_fold_ordered`), so the thread count does
341    /// not change the result.
342    fn pair_gradients(
343        &self,
344        local_adjoint: &DMatrix<Complex>,
345        value: &DMatrix<Complex>,
346    ) -> Result<(Vec<[f64; 3]>, f64)> {
347        let modes = self.modes_per_particle();
348        let particles = self.positions.len();
349        try_fold_ordered(
350            (0..particles).collect(),
351            true,
352            || (vec![[0.0; 3]; particles], 0.0),
353            |(mut positions, mut k0): (Vec<[f64; 3]>, f64), _, i: usize| -> Result<_> {
354                let mut cotangent = vec![Complex::default(); modes * modes];
355                for j in (0..particles).filter(|&j| j != i) {
356                    // The rows of particle i against the columns of particle j,
357                    // sequentially: the particles already run in parallel.
358                    matmul(
359                        MatMut::from_column_major_slice_mut(&mut cotangent, modes, modes),
360                        Accum::Replace,
361                        view(local_adjoint).subrows(i * modes, modes),
362                        view(value).subrows(j * modes, modes).adjoint(),
363                        Complex::new(1.0, 0.0),
364                        Par::Seq,
365                    );
366                    let displacement = std::array::from_fn(|axis| {
367                        self.positions[i][axis] - self.positions[j][axis]
368                    });
369                    let (position, frequency) = self
370                        .plan
371                        .pullback(
372                            Complex::new(self.k0, 0.0),
373                            displacement,
374                            Radial::Singular,
375                            &cotangent,
376                        )
377                        .map(|(position, k)| (position, k.re))?;
378                    k0 += frequency;
379                    for (axis, derivative) in position.into_iter().enumerate() {
380                        positions[i][axis] += derivative;
381                        positions[j][axis] -= derivative;
382                    }
383                }
384                Ok((positions, k0))
385            },
386            |(mut positions, k0), (partial, partial_k0)| {
387                for (a, b) in positions.iter_mut().flatten().zip(partial.iter().flatten()) {
388                    *a += b;
389                }
390                (positions, k0 + partial_k0)
391            },
392        )
393    }
394
395    /// Radius, permittivity and wavenumber gradients through the Mie coefficients,
396    /// whose local cotangent is `Y (B + C X)ᴴ` restricted to each 2 x 2 block.
397    ///
398    /// [`SphereResidual::pullback`](crate::tmatrix::SphereResidual::pullback) does the
399    /// same for the dense path of [`sphere_cluster`](super::sphere_cluster), with two
400    /// differences in order. It sums the Mie gradients of all degrees and then applies
401    /// the chain rule of the size parameter `k0 r` once; here each degree adds its own
402    /// `k0 g` and `r g`. It also swaps each 2 x 2 block to helicity order before
403    /// summing, where this sums in mode order and swaps once, which gives the same
404    /// values. The first difference changes the last bits, so the two paths agree to
405    /// rounding, and each keeps its order so that its results stay reproducible.
406    fn local_gradients(
407        &self,
408        adjoint: &DMatrix<Complex>,
409        response: &DMatrix<Complex>,
410        cluster: &mut SphereClusterGradient,
411    ) -> Result<()> {
412        let modes = self.modes_per_particle();
413        for (particle, degrees) in self.degrees.iter().enumerate() {
414            let mut degree_cotangents = vec![Matrix2::zeros(); degrees.len()];
415            for (block, &degree) in self.mode_degrees.iter().enumerate() {
416                let offset = particle * modes + block * 2;
417                for i in 0..2 {
418                    for j in 0..2 {
419                        for p in 0..adjoint.ncols() {
420                            degree_cotangents[degree][(i, j)] +=
421                                adjoint[(offset + i, p)] * response[(offset + j, p)].conj();
422                        }
423                    }
424                }
425            }
426            for (residual, cotangent) in degrees.iter().zip(degree_cotangents) {
427                let gradient = residual.clone().pullback(&to_mode_order(&cotangent))?;
428                cluster.radii[particle] += self.k0 * gradient.sizes[0];
429                cluster.k0 += self.radii[particle] * gradient.sizes[0];
430                cluster.epsilon[particle] += gradient.epsilon[0];
431            }
432        }
433        Ok(())
434    }
435}

474    /// One to three lossy or lossless spheres, each placed a random gap beyond touching
475    /// its predecessor, along the polar axis or in a random direction.
476    fn cluster() -> impl Strategy<Value = Cluster> {
477        let direction = prop_oneof![
478            Just([0.0, 0.0, 1.0]),
479            Just([0.0, 0.0, -1.0]),
480            (0.0..PI, -PI..PI).prop_map(|(theta, phi): (f64, f64)| {
481                [
482                    theta.sin() * phi.cos(),
483                    theta.sin() * phi.sin(),
484                    theta.cos(),
485                ]
486            }),
487        ];
488        (1_usize..=3, 1_u32..=2)
489            .prop_flat_map(move |(count, lmax)| {
490                (
491                    Just(lmax),
492                    prop::collection::vec(0.1_f64..0.35, count),
493                    prop::collection::vec(
494                        (1.2_f64..4.0, prop_oneof![Just(0.0), 0.0_f64..0.05]),
495                        count,
496                    ),
497                    prop::array::uniform3(-1.0_f64..1.0),
498                    prop::collection::vec((0.05_f64..1.0, direction.clone()), count - 1),
499                )
500            })
501            .prop_map(|(lmax, radii, epsilon, origin, steps)| {
502                let mut positions = vec![origin];
503                for (i, (gap, direction)) in steps.into_iter().enumerate() {
504                    let distance = radii[i] + radii[i + 1] + gap;
505                    let previous = positions[i];
506                    positions.push(std::array::from_fn(|a| {
507                        previous[a] + distance * direction[a]
508                    }));
509                }
510                Cluster {
511                    lmax,
512                    radii,
513                    epsilon: epsilon
514                        .into_iter()
515                        .map(|(re, im)| Complex::new(re, im))
516                        .collect(),
517                    positions,
518                }
519            })

522    impl Cluster {
523        fn operator(&self, k0: f64) -> IterativeSphereCluster {
524            IterativeSphereCluster::new(self.lmax, k0, &self.radii, &self.epsilon, &self.positions)
525                .unwrap()

527
528        /// The signed permutation `S` mapping `(particle, l, m, pol)` to
529        /// `(particle, l, -m, pol)` with sign `(-1)^m`.
530        fn reciprocity(&self) -> DMatrix<Complex> {
531            let modes = sw::modes(self.lmax).unwrap();
532            let modes_per_particle = modes.len();
533            let dimension = modes_per_particle * self.radii.len();
534            let mut s = DMatrix::zeros(dimension, dimension);
535            for particle in 0..self.radii.len() {
536                let offset = particle * modes_per_particle;
537                for (from, mode) in modes.iter().enumerate() {
538                    let to = modes
539                        .iter()
540                        .position(|other| {
541                            other.l == mode.l && other.m == -mode.m && other.pol == mode.pol
542                        })
543                        .unwrap();
544                    let sign = if mode.m % 2 == 0 { 1.0 } else { -1.0 };
545                    s[(offset + to, offset + from)] = Complex::from(sign);
546                }
547            }
548            s
549        }
550    }

567    /// Matrix-free GMRES solves equal the dense cluster T-matrix applied to the
568    /// requested columns and certify their residuals; for a single sphere both equal
569    /// its Mie T-matrix and its product with the columns. The dense T-matrix obeys
570    /// Lorentz reciprocity `T = S Tᵀ S`, and so do the requested columns:
571    /// `uᵀ S X_v = vᵀ S X_u`. Solves are invariant under translation, joint scaling
572    /// and relabelling of the particles, and the pullback equals the dense one, with
573    /// the incident cotangent `Tᴴ G`, the loss pairing, zero total translation
574    /// gradient and the Euler identity of joint length and wavenumber scaling.
575    fn check_requested_columns(
576        cluster: &Cluster,
577        seed: f64,
578        scale: f64,
579        offset: [f64; 3],
580    ) -> Result<(), TestCaseError> {
581        let Cluster {
582            lmax,
583            radii,
584            epsilon,
585            positions,
586        } = cluster;
587        for i in 0..positions.len() {
588            for j in 0..i {
589                let d = (0..3).fold(0.0_f64, |d, a| d.hypot(positions[i][a] - positions[j][a]));
590                prop_assume!(d > radii[i] + radii[j] + 0.01);
591            }
592        }
593        let k0 = 1.2;
594        let operator = cluster.operator(k0);
595        let incident = patterned(operator.dimension(), 2, seed);
596        let result = operator.solve(&incident, options()).unwrap();
597        let dense = crate::cluster::sphere_cluster(*lmax, k0, radii, epsilon, positions).unwrap();
598        let t = dense.value().clone();
599        let expected = &t * &incident;
600        prop_assert_close!(&result.value, &expected, 2e-10 * expected.norm());
601        for report in &result.convergence {
602            prop_assert!(report.residual_norm <= options().rtol * report.rhs_norm);
603        }
604        if let [radius] = radii.as_slice() {
605            let material = Material {
606                epsilon: epsilon[0],
607                ..Material::default()
608            };
609            let (sphere, _) =
610                crate::tmatrix::sphere(*lmax, k0, &[*radius], &[material, Material::default()])
611                    .unwrap();
612            prop_assert_close!(&sphere, &t, 1e-13);
613            prop_assert_close!(&result.value, &(&sphere * &incident), 1e-13);
614        }
615
616        let s = cluster.reciprocity();
617        prop_assert_close!(&(&s * t.transpose() * &s), &t, 1e-12 * t.norm());
618        let (u, v) = (incident.column(0), incident.column(1));
619        let forward = (u.transpose() * &s * result.value.column(1))[0];
620        let backward = (v.transpose() * &s * result.value.column(0))[0];
621        prop_assert_close!(forward, backward, 1e-10 * u.norm() * v.norm() * t.norm());
622
623        let translated = Cluster {
624            positions: positions
625                .iter()
626                .map(|p| std::array::from_fn(|a| p[a] + offset[a]))
627                .collect(),
628            ..cluster.clone()
629        };
630        let other = translated.operator(k0).solve(&incident, options()).unwrap();
631        prop_assert_close!(&other.value, &result.value, 2e-12);
632        let scaled = Cluster {
633            radii: radii.iter().map(|r| r * scale).collect(),
634            positions: positions.iter().map(|p| p.map(|x| x * scale)).collect(),
635            ..cluster.clone()
636        };
637        let other = scaled
638            .operator(k0 / scale)
639            .solve(&incident, options())
640            .unwrap();
641        prop_assert_close!(&other.value, &result.value, 2e-11);
642        let reversed = Cluster {
643            radii: radii.iter().rev().copied().collect(),
644            epsilon: epsilon.iter().rev().copied().collect(),
645            positions: positions.iter().rev().copied().collect(),
646            ..cluster.clone()
647        };
648        let modes = operator.modes_per_particle();
649        let count = radii.len();
650        let reverse = |x: &DMatrix<Complex>| {
651            DMatrix::from_fn(x.nrows(), x.ncols(), |i, j| {
652                x[((count - 1 - i / modes) * modes + i % modes, j)]
653            })
654        };
655        let other = reversed
656            .operator(k0)
657            .solve(&reverse(&incident), options())
658            .unwrap();
659        prop_assert_close!(&other.value, &reverse(&result.value), 1e-12);
660
661        let g = patterned(operator.dimension(), 2, seed + 0.3);
662        let grad = Arc::new(operator)
663            .record(incident.clone(), options())
664            .unwrap()
665            .pullback(&g)
666            .unwrap();
667        prop_assert_close!(&grad.incident, &(t.adjoint() * &g), 3e-10 * g.norm());
668        let dense_grad = dense.pullback(&(&g * incident.adjoint())).unwrap();
669        let cluster_grad = &grad.cluster;
670        prop_assert_close!(&cluster_grad.radii, &dense_grad.radii, 3e-10);
671        prop_assert_close!(&cluster_grad.epsilon, &dense_grad.epsilon, 3e-10);
672        prop_assert_close!(&cluster_grad.positions, &dense_grad.positions, 3e-10);
673        prop_assert_close!(cluster_grad.k0, dense_grad.k0, 3e-10);
674        for axis in 0..3 {
675            let total: f64 = cluster_grad.positions.iter().map(|p| p[axis]).sum();
676            prop_assert_close!(total, 0.0, 1e-12, "axis {}", axis);
677        }
678        let loss = re_dot(&g, &result.value);
679        prop_assert_close!(loss, re_dot(&grad.incident, &incident), 3e-10);
680        let scale_gradient = dot(radii, &cluster_grad.radii)
681            + dot(
682                positions.iter().flatten(),
683                cluster_grad.positions.iter().flatten(),
684            )
685            - k0 * cluster_grad.k0;
686        prop_assert_close!(scale_gradient, 0.0, 3e-10);
687        Ok(())
688    }

690    /// Near-touching dielectric spheres require several restart cycles. Eleven
691    /// columns exercise a full batch and its tail, with zero and widely differing
692    /// amplitudes in both the forward and adjoint systems.
693    #[test]
694    fn restarted_batches_match_dense_for_strongly_coupled_spheres() -> Result<(), TestCaseError> {
695        let cluster = Cluster {
696            lmax: 3,
697            radii: vec![0.63, 0.60, 0.59],
698            epsilon: vec![
699                Complex::new(5.0, 0.0),
700                Complex::new(6.0, 0.05),
701                Complex::new(4.0, 0.02),
702            ],
703            positions: vec![[0.0; 3], [1.3, 0.0, 0.0], [0.65, 1.15, 0.1]],
704        };
705        let k0 = 1.4;
706        let options = GmresOptions {
707            restart: 3,
708            max_iterations: 300,
709            ..options()
710        };
711        let operator = Arc::new(cluster.operator(k0));
712        let scales = [1e-8, 0.0, 1.0, 1e8, 0.1, 7.0, 0.0, 1e-4, 1e4, 0.0, 2.0];
713        let mut incident = patterned(operator.dimension(), scales.len(), 0.4);
714        let mut g = patterned(operator.dimension(), scales.len(), 1.1);
715        for (column, scale) in scales.into_iter().enumerate() {
716            incident.column_mut(column).scale_mut(scale);
717            // Reciprocal amplitudes keep each nonzero loss contribution comparable.
718            let cotangent_scale = if [4, 9].contains(&column) {
719                0.0
720            } else if scale == 0.0 {
721                0.01
722            } else {
723                0.01 / scale
724            };
725            g.column_mut(column).scale_mut(cotangent_scale);
726        }
727        let record = operator.record(incident.clone(), options).unwrap();
728        let result = record.solution();
729        let dense = crate::cluster::sphere_cluster(
730            cluster.lmax,
731            k0,
732            &cluster.radii,
733            &cluster.epsilon,
734            &cluster.positions,
735        )
736        .unwrap();
737        let expected = dense.value() * &incident;
738        for (column, report) in result.convergence.iter().enumerate() {
739            let actual = result.value.column(column);
740            let expected = expected.column(column);
741            prop_assert_close!(
742                actual.as_slice(),
743                expected.as_slice(),
744                2e-10 * expected.norm() + f64::MIN_POSITIVE,
745                "forward column {}",
746                column
747            );
748            prop_assert!(report.residual_norm <= options.rtol * report.rhs_norm);
749            if scales[column] == 0.0 {
750                prop_assert_eq!(report.iterations, 0);
751                prop_assert!(actual.iter().all(|&z| z == Complex::default()));
752            } else {
753                prop_assert!(report.iterations > options.restart);
754            }
755        }
756        let loss = re_dot(&g, &result.value);
757        let grad = record.pullback(&g).unwrap();
758        let expected_incident = dense.value().adjoint() * &g;
759        for (column, report) in grad.convergence.iter().enumerate() {
760            let actual = grad.incident.column(column);
761            let expected = expected_incident.column(column);
762            prop_assert_close!(
763                actual.as_slice(),
764                expected.as_slice(),
765                3e-10 * g.column(column).norm() + f64::MIN_POSITIVE,
766                "adjoint column {}",
767                column
768            );
769            prop_assert!(report.residual_norm <= options.rtol * report.rhs_norm);
770            if [4, 9].contains(&column) {
771                prop_assert_eq!(report.iterations, 0);
772                prop_assert!(actual.iter().all(|&z| z == Complex::default()));
773            } else {
774                prop_assert!(report.iterations > options.restart);
775            }
776        }
777        let dense_grad = dense.pullback(&(&g * incident.adjoint())).unwrap();
778        let cluster_grad = &grad.cluster;
779        prop_assert_close!(&cluster_grad.radii, &dense_grad.radii, 3e-10);
780        prop_assert_close!(&cluster_grad.epsilon, &dense_grad.epsilon, 3e-10);
781        prop_assert_close!(&cluster_grad.positions, &dense_grad.positions, 3e-10);
782        prop_assert_close!(cluster_grad.k0, dense_grad.k0, 3e-10);
783        prop_assert_close!(loss, re_dot(&grad.incident, &incident), 3e-10);
784        for axis in 0..3 {
785            let total: f64 = cluster_grad.positions.iter().map(|p| p[axis]).sum();
786            prop_assert_close!(total, 0.0, 1e-12, "axis {}", axis);
787        }
788        let scale_gradient = dot(&cluster.radii, &cluster_grad.radii)
789            + dot(
790                cluster.positions.iter().flatten(),
791                cluster_grad.positions.iter().flatten(),
792            )
793            - k0 * cluster_grad.k0;
794        prop_assert_close!(scale_gradient, 0.0, 3e-10);
795        Ok(())
796    }
