3//! Upstream: `treams.SMatrices.from_array`.

5use faer::MatRef;
6use nalgebra::DMatrix;

8use super::{Blocks, checked_dimension, dimension};
9use crate::{

11    linalg::{product, product_adjoint_right, product_views, view},
12    numerics::finite,
13};

15/// The four multipole-by-plane-wave channel arrays of [`from_array`], in the order
16/// incident up, incident down, emitted up, emitted down.
17///
18/// Each array has one row per multipole and one column per plane-wave mode. Unlike
19/// [`Blocks`], the arrays need not be square.
20pub type Channels = [DMatrix<Complex>; 4];
21
22/// What [`from_array`] saves for its pullback: the response, the channel arrays and
23/// the scattered fields.
24#[derive(Debug)]
25pub struct FromArrayResidual {
26    response: DMatrix<Complex>,
27    channels: Channels,
28    scattered: [DMatrix<Complex>; 2],
29}
30
31/// Compose an effective response with incident/emitted, up/down channel arrays.
32///
33/// Returns four plane-wave scattering blocks, including direct transmission.
34///
35/// Upstream: `treams.SMatrices.from_array`, given the channels of
36/// [`spherical_channels`](crate::channels::spherical_channels) or
37/// [`cylindrical_channels`](crate::channels::cylindrical_channels).
38pub fn from_array(
39    response: DMatrix<Complex>,
40    channels: Channels,
41) -> Result<(Blocks, FromArrayResidual)> {
42    let d = response.nrows();
43    let c = channels[0].ncols();
44    if d == 0
45        || !response.is_square()
46        || c == 0
47        || response.iter().any(|&v| !finite(v))
48        || channels
49            .iter()
50            .any(|a| a.shape() != (d, c) || a.iter().any(|&v| !finite(v)))
51    {
52        return Err(Error::InvalidInput(
53            "require a finite square response and four matching multipole-by-plane channel arrays"
54                .into(),
55        ));
56    }
57    let scattered = std::array::from_fn(|direction| product(&response, &channels[direction]));
58    let mut value = std::array::from_fn(|b| {
59        product_views(
60            view(&channels[2 + b / 2]).transpose(),
61            view(&scattered[b % 2]),
62        )
63    });
64    for block in [0, 3] {
65        for i in 0..c {
66            value[block][(i, i)] += 1.0;
67        }
68    }
69    dimension(&value).map_err(|_| Error::NonFinite("non-finite array scattering matrix".into()))?;
70    Ok((
71        value,
72        FromArrayResidual {
73            response,
74            channels,
75            scattered,
76        },
77    ))
78}

80/// Cotangents of the inputs of [`from_array`], in the order of its arguments.
81#[derive(Debug)]
82pub struct FromArrayGradient {
83    /// Cotangent of the effective multipole response.
84    pub response: DMatrix<Complex>,
85    /// Cotangents of the four channel arrays.
86    pub channels: Channels,
87}
88
89impl FromArrayResidual {
90    /// The shape `(n, n)` of each scattering block, with `n` plane-wave modes in each
91    /// direction.
92    #[must_use]
93    pub fn shape(&self) -> (usize, usize) {
94        (self.channels[0].ncols(), self.channels[0].ncols())
95    }
96
97    /// Effective-response and four-channel cotangents, consuming the residual.
98    pub fn pullback(self, cotangent: &Blocks) -> Result<FromArrayGradient> {
99        let c = checked_dimension(
100            cotangent,
101            self.channels[0].ncols(),
102            "invalid array scattering cotangent shape",
103        )?;
104        // Blocks are C_{2+e}^T S_s with S_s = R C_s, so the emission cotangents
105        // conjugate without transposing: bar S_s = conj(C_2) g_s + conj(C_3) g_{2+s}.
106        let conjugate =
107            |a: &DMatrix<Complex>, b: MatRef<'_, Complex>| product_views(view(a).conjugate(), b);
108        let mut response = DMatrix::zeros(self.response.nrows(), self.response.ncols());
109        let mut channels: Channels =
110            std::array::from_fn(|_| DMatrix::zeros(self.response.nrows(), c));
111        for direction in 0..2 {
112            let adjoint = conjugate(&self.channels[2], view(&cotangent[direction]))
113                + conjugate(&self.channels[3], view(&cotangent[2 + direction]));
114            response += product_adjoint_right(&adjoint, &self.channels[direction]);
115            channels[direction] = product_views(view(&self.response).adjoint(), view(&adjoint));
116            channels[2 + direction] = conjugate(
117                &self.scattered[0],
118                view(&cotangent[2 * direction]).transpose(),
119            ) + conjugate(
120                &self.scattered[1],
121                view(&cotangent[2 * direction + 1]).transpose(),
122            );
123        }
124        Ok(FromArrayGradient { response, channels })
125    }
126}
