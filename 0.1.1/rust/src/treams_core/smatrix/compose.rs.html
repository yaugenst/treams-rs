19/// What [`add`] saves for its pullback: four of the eight input blocks, the internal
20/// fields and their factorization.
21///
22/// The pullback reads only blocks 1 and 3 of the lower S-matrix and blocks 0 and 2
23/// of the upper one; the other four input blocks are released after the forward pass.
24#[derive(Debug)]
25pub struct AddResidual {
26    /// Lower blocks 1 and 3: reflection of waves incident from the positive side (above)
27    /// and downward transmission.
28    lower: [DMatrix<Complex>; 2],
29    /// Upper blocks 0 and 2: upward transmission and reflection of waves incident from
30    /// the negative side (below).
31    upper: [DMatrix<Complex>; 2],
32    /// Upgoing internal fields `(I - L1 U2)^-1 [L0, L1 U3]` and their factorization.
33    solve: InternalSolve,
34    /// Downgoing internal fields `U2 up + [0, U3]`.
35    down: DMatrix<Complex>,
36}
37
38/// Place `upper` above `lower`, eliminating their internal incident fields.
39///
40/// Upstream: `treams.SMatrices.add`, called as `lower.add(upper)`.
41pub fn add(lower: Blocks, upper: Blocks) -> Result<(Blocks, AddResidual)> {
42    let n = dimension(&lower)?;
43    if dimension(&upper)? != n {
44        return Err(Error::InvalidInput("S matrix dimensions must match".into()));
45    }
46    let [l0, l1, mut l2, l3] = lower;
47    let [u0, mut u1, u2, u3] = upper;
48    let mut rhs = DMatrix::zeros(n, 2 * n);
49    rhs.columns_mut(0, n).copy_from(&l0);
50    rhs.columns_mut(n, n).copy_from(&product(&l1, &u3));
51    // The Krylov solve needs at least ITERATIVE_MIN_ROWS = 512 rows and at most
52    // ITERATIVE_MAX_COLUMNS = 8 right-hand sides, so 2n columns never qualify and
53    // the direct solve gives the same bits as `InternalSolve::new`.
54    let solve = InternalSolve::direct(view(&l1), view(&u2), rhs)?;
55    let up = view(&solve.value);
56    let mut down = product_views(view(&u2), up);
57    down.columns_mut(n, n).add_assign(&u3);
58    let down_view = view(&down);
59    let value = [
60        product_views(view(&u0), up.subcols(0, n)),
61        {
62            accumulate_product(&mut u1, view(&u0), up.subcols(n, n));
63            u1
64        },
65        {
66            accumulate_product(&mut l2, view(&l3), down_view.subcols(0, n));
67            l2
68        },
69        product_views(view(&l3), down_view.subcols(n, n)),
70    ];
71    dimension(&value).map_err(|_| Error::Singular)?;
72    Ok((
73        value,
74        AddResidual {
75            lower: [l1, l3],
76            upper: [u0, u2],
77            solve,
78            down,
79        },
80    ))
81}

83/// `target += left * right`.
84fn accumulate_product(
85    target: &mut DMatrix<Complex>,
86    left: MatRef<'_, Complex>,
87    right: MatRef<'_, Complex>,
88) {
89    let (m, n, k) = (target.nrows(), target.ncols(), left.ncols());
90    crate::threads::product(m, n, k, |par| {
91        faer::linalg::matmul::matmul(
92            view_mut(target),
93            faer::Accum::Add,
94            left,
95            right,
96            Complex::new(1.0, 0.0),
97            par,
98        );
99    });
100}

102/// Cotangents of the inputs of [`add`], in the order of its arguments.
103#[derive(Debug)]
104pub struct AddGradient {
105    /// Cotangents of the blocks of the lower S-matrix.
106    pub lower: Blocks,
107    /// Cotangents of the blocks of the upper S-matrix.
108    pub upper: Blocks,
109}
110
111impl AddResidual {
112    /// The shape `(n, n)` of each scattering block, with `n` modes in each direction.
113    #[must_use]
114    pub fn shape(&self) -> (usize, usize) {
115        (self.down.nrows(), self.down.nrows())
116    }
117
118    /// Input cotangents under `dL = Re(sum(conj(g) * dx))`; consumes the residual.
119    pub fn pullback(self, cotangent: &Blocks) -> Result<AddGradient> {
120        let n = checked_dimension(
121            cotangent,
122            self.down.nrows(),
123            "invalid S matrix cotangent shape",
124        )?;
125        let Self {
126            lower: [mut reflection, mut transmission],
127            upper: [mut top_transmission, mut bottom_reflection],
128            solve,
129            down: down_fields,
130        } = self;
131        let mut top = DMatrix::zeros(n, 2 * n);
132        top.columns_mut(0, n).copy_from(&cotangent[0]);
133        top.columns_mut(n, n).copy_from(&cotangent[1]);
134        let mut bottom = DMatrix::zeros(n, 2 * n);
135        bottom.columns_mut(0, n).copy_from(&cotangent[2]);
136        bottom.columns_mut(n, n).copy_from(&cotangent[3]);
137        let down = product_views(view(&transmission).adjoint(), view(&bottom));
138        let adjoint = product_views(view(&top_transmission).adjoint(), view(&top))
139            + product_views(view(&bottom_reflection).adjoint(), view(&down));
140        let AdjointSolve { adjoint, value: up } =
141            solve.solve_adjoint(view(&reflection), view(&bottom_reflection), adjoint)?;
142        let incident = down + product_views(view(&reflection).adjoint(), view(&adjoint));
143        // The saved input blocks are read for the last time above; overwrite them with
144        // their gradients.
145        product_adjoint_right_into(&mut reflection, &adjoint, &down_fields);
146        product_adjoint_right_into(&mut transmission, &bottom, &down_fields);
147        product_adjoint_right_into(&mut top_transmission, &top, &up);
148        product_adjoint_right_into(&mut bottom_reflection, &incident, &up);
149        let lower = [
150            adjoint.columns(0, n).into_owned(),
151            reflection,
152            cotangent[2].clone(),
153            transmission,
154        ];
155        let upper = [
156            top_transmission,
157            cotangent[1].clone(),
158            bottom_reflection,
159            incident.columns(n, n).into_owned(),
160        ];
161        Ok(AddGradient { lower, upper })
162    }
163}
