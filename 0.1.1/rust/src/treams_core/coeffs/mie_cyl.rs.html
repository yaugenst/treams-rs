1//! Mie coefficients of multilayer chiral cylinders and their analytic pullback.
2//!
3//! Upstream: `treams.coeffs.mie_cyl`. The pullback is a treams-rs extension.
4#![allow(clippy::indexing_slicing)] // Fixed four-channel interface matrices.

10    special::{Radial, RadialJet, cylindrical_radial},

13/// One boundary between two media, as the pullback reads it.
14#[derive(Clone, Debug)]
15struct Interface {
16    /// The inner medium at the boundary.
17    inside: Side,
18    /// The outer medium at the boundary.
19    outside: Side,
20    /// The inverse of the outer boundary matrix.
21    inverse_outer: Matrix4,
22    /// The map from inner to outer coefficients: `inverse_outer` times the inner
23    /// boundary matrix.
24    transfer: Matrix4,
25    /// The product of the transfers inside this boundary, applied to the two regular
26    /// core channels.
27    before: Matrix42,
28}
29
30/// What [`mie_cyl`] saves for its pullback: the four-channel interface states and
31/// radial functions, so the pullback evaluates no Bessel functions.
32#[derive(Clone, Debug)]
33pub struct MieCylResidual {
34    order: i32,
35    kz: f64,
36    k0: f64,
37    radii: Vec<f64>,
38    materials: Vec<Material>,
39    interfaces: Vec<Interface>,
40    inverse: Matrix2,
41    value: Matrix2,
42}

44/// Gradients of the inputs of [`mie_cyl`], in the order of its arguments.
45#[derive(Clone, Debug)]
46pub struct MieCylGradient {
47    /// Gradient of the axial wavenumber.
48    pub kz: f64,
49    /// Gradient of the vacuum wavenumber.
50    pub k0: f64,
51    /// Gradients of the radii and the materials.
52    pub layers: LayerGradient,
53}
54
55/// A direction of change of one medium at one boundary: of its helicity wavenumbers,
56/// the axial wavenumber, the radius and its impedance.
57#[derive(Clone, Copy, Debug, Default)]
58struct Direction {
59    ks: [Complex; 2],
60    kz: Complex,
61    radius: Complex,
62    impedance: Complex,
63}
64
65/// One medium at a boundary: its helicity wavenumbers, impedance, radial wavenumbers
66/// `kr` and arguments `x = kr radius`, and the regular and (except in the core)
67/// singular radial functions of each helicity.
68#[derive(Clone, Copy, Debug)]
69struct Side {
70    ks: [Complex; 2],
71    impedance: Complex,
72    kr: [Complex; 2],
73    x: [Complex; 2],
74    radial: [[Option<RadialJet>; 2]; 2],
75}
76
77impl Side {
78    fn new(
79        order: i32,
80        kz: f64,
81        ks: [Complex; 2],
82        radius: f64,
83        impedance: Complex,
84        regular_only: bool,
85    ) -> Result<Self> {
86        let mut side = Self {
87            ks,
88            impedance,
89            kr: [Complex::default(); 2],
90            x: [Complex::default(); 2],
91            radial: [[None; 2]; 2],
92        };
93        for (pol, &k) in ks.iter().enumerate() {
94            let kr = (k * k - kz * kz).sqrt();
95            if k.norm_sqr() == 0.0 || kr.norm_sqr() == 0.0 {
96                return Err(Error::InvalidInput(
97                    "cylindrical cutoff k_rho=0 is not supported".into(),
98                ));
99            }
100            let x = kr * radius;
101            (side.kr[pol], side.x[pol]) = (kr, x);
102            let kinds: &[Radial] = if regular_only {
103                &[Radial::Regular]
104            } else {
105                &[Radial::Regular, Radial::Singular]
106            };
107            for (kind, &radial) in kinds.iter().enumerate() {
108                side.radial[pol][kind] = Some(cylindrical_radial(order, x, radial)?);
109            }
110        }
111        Ok(side)
112    }
113
114    /// The columns `(pol, kind)` of the boundary matrix with their radial function.
115    fn columns(&self) -> impl Iterator<Item = (usize, usize, RadialJet)> + '_ {
116        (0..2).flat_map(move |kind| {
117            (0..2).filter_map(move |pol| self.radial[pol][kind].map(|f| (pol, kind, f)))
118        })
119    }
120
121    /// The boundary matrix: tangential field components of each wave.
122    fn matrix(&self, order: i32, kz: f64) -> Matrix4 {
123        let mut value = Matrix4::zeros();
124        for (pol, kind, f) in self.columns() {
125            let (k, kr, x) = (self.ks[pol], self.kr[pol], self.x[pol]);
126            let phi = -kr * f.value / k;
127            let sign = if pol == 0 { 1.0 } else { -1.0 };
128            let z = -kz * f64::from(order) * f.value / (k * x) + sign * f.first;
129            let col = pol + 2 * kind;
130            value[(0, col)] = phi;
131            value[(1, col)] = z;
132            value[(2, col)] = -sign * phi / self.impedance;
133            value[(3, col)] = -sign * z / self.impedance;
134        }
135        value
136    }
137
138    /// The directional derivative of [`Self::matrix`], linear in the direction.
139    fn derivative(&self, order: i32, kz: f64, radius: f64, direction: Direction) -> Matrix4 {
140        let mut derivative = Matrix4::zeros();
141        for (pol, kind, f) in self.columns() {
142            let (k, kr, x) = (self.ks[pol], self.kr[pol], self.x[pol]);
143            let dk = direction.ks[pol];
144            let dkr = (k * dk - kz * direction.kz) / kr;
145            let dx = dkr * radius + kr * direction.radius;
146            let phi = -kr * f.value / k;
147            let dphi = -(dkr * f.value + kr * f.first * dx) / k - phi * dk / k;
148            let sign = if pol == 0 { 1.0 } else { -1.0 };
149            let z = -kz * f64::from(order) * f.value / (k * x) + sign * f.first;
150            let dz = -f64::from(order) / (k * x)
151                * (direction.kz * f.value + kz * f.first * dx - kz * f.value * (dk / k + dx / x))
152                + sign * f.second * dx;
153            let (impedance, dimpedance) = (self.impedance, direction.impedance);
154            let col = pol + 2 * kind;
155            derivative[(0, col)] = dphi;
156            derivative[(1, col)] = dz;
157            derivative[(2, col)] = -sign * (dphi - phi * dimpedance / impedance) / impedance;
158            derivative[(3, col)] = -sign * (dz - z * dimpedance / impedance) / impedance;
159        }
160        derivative
161    }
162}

164/// Mie coefficients of concentric chiral cylinders at the axial wavenumber `kz` and the
165/// azimuthal order `m` (`order`).
166///
167/// `radii` are the boundary radii, from inner to outer, and `materials` has one more
168/// entry, the embedding medium last.
169///
170/// Upstream: `treams.coeffs.mie_cyl`.
171pub fn mie_cyl(
172    kz: f64,
173    order: i32,
174    k0: f64,
175    radii: &[f64],
176    materials: &[Material],
177) -> Result<MieCylResidual> {
178    crate::coeffs::validate_layers(radii, materials)?;
179    if !kz.is_finite()
180        || !k0.is_finite()
181        || k0 <= 0.0
182        || order.unsigned_abs() > MAX_DEGREE.unsigned_abs()
183    {
184        return Err(Error::InvalidInput(
185            // 128 is MAX_DEGREE.
186            "require finite kz, positive k0 and |m| <= 128".into(),
187        ));
188    }
189    let mut q = Matrix42::zeros();
190    q[(0, 0)] = Complex::new(1.0, 0.0);
191    q[(1, 1)] = Complex::new(1.0, 0.0);
192    let mut interfaces = Vec::with_capacity(radii.len());
193    for (i, &radius) in radii.iter().enumerate() {
194        let side = |m: Material, regular_only| {
195            let ks = [k0 * (m.index() - m.kappa), k0 * (m.index() + m.kappa)];
196            Side::new(order, kz, ks, radius, m.impedance(), regular_only)
197        };
198        let inside = side(materials[i], i == 0)?;
199        let outside = side(materials[i + 1], false)?;
200        let inverse_outer = outside
201            .matrix(order, kz)
202            .try_inverse()
203            .ok_or(Error::Singular)?;
204        let transfer = inverse_outer * inside.matrix(order, kz);
205        interfaces.push(Interface {
206            inside,
207            outside,
208            inverse_outer,
209            transfer,
210            before: q,
211        });
212        q = transfer * q;
213    }
214    let inverse = q
215        .fixed_rows::<2>(0)
216        .into_owned()
217        .try_inverse()
218        .ok_or(Error::Singular)?;
219    let value = q.fixed_rows::<2>(2) * inverse;
220    if value.iter().any(|&z| !finite(z)) {
221        return Err(Error::Singular);
222    }
223    Ok(MieCylResidual {
224        order,
225        kz,
226        k0,
227        radii: radii.to_vec(),
228        materials: materials.to_vec(),
229        interfaces,
230        inverse,
231        value,
232    })
233}
234impl MieCylResidual {
235    /// The coefficients in negative, positive helicity order, which the pullback reads.
236    #[must_use]
237    pub const fn value(&self) -> &Matrix2 {
238        &self.value
239    }
240
241    /// Gradients of `kz`, `k0`, the radii and the materials from `cotangent`, the
242    /// gradient of a real loss with respect to the coefficient matrix.
243    ///
244    /// The pullback runs on the calling thread and adds the boundaries in order, from
245    /// the outermost in.
246    pub fn pullback(self, cotangent: &Matrix2) -> Result<MieCylGradient> {
247        self.gradient(cotangent)
248    }
249
250    /// [`pullback`](Self::pullback) without consuming the residual, so the T-matrix
251    /// pullback can reuse one solve for a block and its mirror.
252    pub(crate) fn gradient(&self, cotangent: &Matrix2) -> Result<MieCylGradient> {
253        if cotangent.iter().any(|&z| !finite(z)) {
254            return Err(Error::InvalidInput("cotangent must be finite".into()));
255        }
256        let mut result = MieCylGradient {
257            kz: 0.0,
258            k0: 0.0,
259            layers: LayerGradient::zeros(self.radii.len()),
260        };
261        let lower = cotangent * self.inverse.adjoint();
262        let mut gq = Matrix42::zeros();
263        gq.fixed_rows_mut::<2>(0)
264            .copy_from(&(-self.value.adjoint() * lower));
265        gq.fixed_rows_mut::<2>(2).copy_from(&lower);
266        for (i, interface) in self.interfaces.iter().enumerate().rev() {
267            let gf = gq * interface.before.adjoint();
268            let inner = interface.inverse_outer.adjoint() * gf;
269            let outer = -inner * interface.transfer.adjoint();
270            let sides = [(&interface.inside, inner), (&interface.outside, outer)];
271            for (side, (boundary, g)) in sides.into_iter().enumerate() {
272                let material = self.materials[i + side];
273                let n = material.index();
274                let contract = |direction: Direction| -> Complex {
275                    let df = boundary.derivative(self.order, self.kz, self.radii[i], direction);
276                    g.dotc(&df).conj()
277                };
278                let one = Complex::new(1.0, 0.0);
279                result.layers.radii[i] += contract(Direction {
280                    radius: one,
281                    ..Direction::default()
282                })
283                .re;
284                result.kz += contract(Direction {
285                    kz: one,
286                    ..Direction::default()
287                })
288                .re;
289                result.k0 += contract(Direction {
290                    ks: [n - material.kappa, n + material.kappa],
291                    ..Direction::default()
292                })
293                .re;
294                let layers = &mut result.layers;
295                let targets = [&mut layers.epsilon, &mut layers.mu, &mut layers.kappa];
296                for (tangent, target) in material.tangents().into_iter().zip(targets) {
297                    target[i + side] += contract(Direction {
298                        ks: [
299                            self.k0 * (tangent.index - tangent.kappa),
300                            self.k0 * (tangent.index + tangent.kappa),
301                        ],
302                        impedance: tangent.impedance,
303                        ..Direction::default()
304                    });
305                }
306            }
307            gq = interface.transfer.adjoint() * gq;
308        }
309        Ok(result)
310    }
