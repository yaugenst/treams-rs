1//! Lattice sums over arrays of inputs, with their gradients.
2//!
3//! treams-rs extension: the `lsum*` ufuncs of treams broadcast their inputs but have no
4//! gradients.
5
6use super::{SumGradient, SumPart, cell::BlochLattice, derivatives_part, sum_part, wave::Family};
7use crate::{
8    Complex, Error, Result,
9    numerics::{broadcast, finite, parallel::Parallel},
10};
11
12/// Batched sums split adaptively across the rayon pool from this many outputs.
13const PARALLEL: Parallel = Parallel::AtLeast(8);
14
15/// [`sum_part`] over arrays, with the residual for the pullback.
16///
17/// Each input holds one value for all outputs or one per output. The residual keeps
18/// the inputs only; the pullback recomputes the local derivatives.
19///
20/// treams-rs extension: the `lsum*` ufuncs of `treams.lattice` broadcast their inputs
21/// but have no gradients.
22pub fn sum_array(
23    waves: Vec<Family>,
24    wavenumbers: Vec<Complex>,
25    lattices: Vec<BlochLattice>,
26    shifts: Vec<[f64; 3]>,
27    etas: Vec<Complex>,
28    parts: Vec<SumPart>,
29) -> Result<(Vec<Complex>, SumResidual)> {
30    let size = broadcast::size(
31        &[
32            waves.len(),
33            wavenumbers.len(),
34            lattices.len(),
35            shifts.len(),
36            etas.len(),
37            parts.len(),
38        ],
39        "arrays must have equal lengths or scalar inputs",
40    )?;
41    if parts
42        .iter()
43        .any(|p| matches!(p, SumPart::Real | SumPart::Reciprocal))
44        && etas.contains(&Complex::default())
45    {
46        return Err(Error::InvalidInput(
47            "component adjoints require an explicit nonzero Ewald split".into(),
48        ));
49    }
50    let residual = SumResidual {
51        waves,
52        wavenumbers,
53        lattices,
54        shifts,
55        etas,
56        parts,
57        size,
58    };
59    let values = broadcast::map(size, PARALLEL, |i| {
60        let (wave, k, lattice, r, eta, part) = residual.element(i);
61        sum_part(wave, k, lattice, r, eta, part)
62    })?;
63    Ok((values, residual))
64}
65
66/// What [`sum_array`] saves for the pullback of a batch of lattice sums: the inputs,
67/// each one value for all outputs or one per output.
68///
69/// The pullback computes the gradient of every output on its own, so its result does
70/// not depend on the thread count.
71#[derive(Debug)]
72pub struct SumResidual {
73    waves: Vec<Family>,
74    wavenumbers: Vec<Complex>,
75    lattices: Vec<BlochLattice>,
76    shifts: Vec<[f64; 3]>,
77    etas: Vec<Complex>,
78    parts: Vec<SumPart>,
79    size: usize,
80}
81
82impl SumResidual {
83    /// The inputs of output `i`.
84    fn element(&self, i: usize) -> (Family, Complex, &BlochLattice, [f64; 3], Complex, SumPart) {
85        (
86            broadcast::element(&self.waves, i),
87            broadcast::element(&self.wavenumbers, i),
88            &self.lattices[if self.lattices.len() == 1 { 0 } else { i }],
89            broadcast::element(&self.shifts, i),
90            broadcast::element(&self.etas, i),
91            broadcast::element(&self.parts, i),
92        )
93    }
94
95    /// One gradient per output, from `cotangent`, the gradient of a real loss with
96    /// respect to the sums.
97    pub fn pullback(self, cotangent: &[Complex]) -> Result<Vec<SumGradient>> {
98        if cotangent.len() != self.size || cotangent.iter().any(|&g| !finite(g)) {
99            return Err(Error::InvalidInput(
100                "cotangent must be finite and match lattice output".into(),
101            ));
102        }
103        broadcast::map(self.size, PARALLEL, |i| {
104            let g = cotangent[i];
105            if g == Complex::default() {
106                return Ok(SumGradient::default());
107            }
108            let (wave, k, lattice, r, eta, part) = self.element(i);
109            Ok(derivatives_part(wave, k, lattice, r, eta, part)?.pullback(g))
110        })
111    }
112}
