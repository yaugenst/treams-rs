1//! The internal fields between two adjacent stacks: a certified Krylov or pivoted LU
2//! solve of `(I - L1 U2) up = rhs`, kept for the pullback.
3//!
4//! Upstream: the `numpy.linalg.solve` inside `treams.SMatrices.illuminate` with a
5//! second S-matrix. The Krylov branch and its contraction test are treams-rs
6//! extensions.
7//!
8//! `L1` is the reflection of the lower stack for waves incident from the positive side
9//! (above) and `U2` the reflection of the upper stack for waves incident from the
10//! negative side (below), so `up` holds the upgoing fields between the stacks.
11

21/// The internal operator `I - lower upper`.
22fn internal_operator(lower: MatRef<'_, Complex>, upper: MatRef<'_, Complex>) -> DMatrix<Complex> {
23    let mut value = -product_views(lower, upper);
24    for i in 0..value.nrows() {
25        value[(i, i)] += 1.0;
26    }
27    value
28}

30/// What the pullback reuses to solve the adjoint system.
31///
32/// A certified contraction is invertible, including for zero or dependent right-hand
33/// sides, so its Krylov solve needs no factorization. Every other operator keeps the
34/// pivoted LU and its singular-system behavior.
35#[derive(Debug)]
36enum InternalFactor {
37    /// The operator itself, for Krylov adjoint solves with a dense LU fallback.
38    Krylov(DMatrix<Complex>),
39    /// The pivoted LU factorization of the operator.
40    Lu(Lu),
41}
42
43/// The forward solution of the internal system and what its adjoint solve reuses.
44#[derive(Debug)]
45pub(super) struct InternalSolve {
46    /// Solution of the internal system for the forward right-hand sides.
47    pub(super) value: DMatrix<Complex>,
48    factor: InternalFactor,
49}

51/// Whether `||L U||_inf < 1` holds for the operator `I - L U`, which proves it invertible.
52///
53/// Each row sum of `|re| + |im|` over the entries of `L U` rounds upward, so the sums
54/// bound the infinity norm from above: rounding can make the test fail, never pass.
55fn is_contraction(operator: &DMatrix<Complex>) -> bool {
56    let n = operator.nrows();
57    let mut rows = vec![0.0_f64; n];
58    for (j, column) in operator.as_slice().chunks_exact(n).enumerate() {
59        for (i, (&z, row)) in column.iter().zip(&mut rows).enumerate() {
60            let delta = if i == j {
61                Complex::new(1.0, 0.0) - z
62            } else {
63                z
64            };
65            // |re|+|im| bounds the complex modulus. Upward-rounded positive
66            // additions keep this an upper bound rather than a norm estimate.
67            *row = (*row + (delta.re.abs() + delta.im.abs()).next_up()).next_up();
68        }
69    }
70    rows.iter().all(|&sum| sum < 1.0)
71}

73/// The unrestarted Krylov solve of the operator, or of its adjoint, for a thin batch.
74fn internal_iteration(
75    operator: &DMatrix<Complex>,
76    rhs: &DMatrix<Complex>,
77    adjoint: bool,
78) -> Result<DMatrix<Complex>> {
79    // One Krylov iteration applies the operator to the entire thin batch.
80    // Its total residual is bounded by the smallest nonzero column's tolerance,
81    // which also certifies every individual illumination at 8 epsilon.
82    let column_norm = |j| norm(rhs.column(j).as_slice());
83    let minimum = (0..rhs.ncols())
84        .map(column_norm)
85        .filter(|&v| v > 0.0)
86        .fold(f64::INFINITY, f64::min);
87    if rhs.iter().all(|&z| z == Complex::default()) {
88        return Ok(DMatrix::zeros(rhs.nrows(), rhs.ncols()));
89    }
90    if !minimum.is_finite() {
91        return Err(Error::Singular);
92    }
93    let options = GmresOptions {
94        rtol: 0.0,
95        atol: 8.0 * f64::EPSILON * minimum,
96        restart: KRYLOV_STEPS,
97        max_iterations: KRYLOV_STEPS,
98    };
99    let (answer, _) = gmres(rhs.as_slice(), options, |x| {
100        let x_view = MatRef::from_column_major_slice(x, rhs.nrows(), rhs.ncols());
101        let result = if adjoint {
102            product_views(view(operator).adjoint(), x_view)
103        } else {
104            product_views(view(operator), x_view)
105        };
106        Ok(Vec::from(result.data))
107    })?;
108    Ok(DMatrix::from_vec(rhs.nrows(), rhs.ncols(), answer))
109}

111/// Rows from which a certified contraction tries the bounded Krylov solve before
112/// falling back to a dense LU factorization.
113const ITERATIVE_MIN_ROWS: usize = 512;
114
115/// Most right-hand sides for which the Krylov solve is tried; wider batches factor.
116const ITERATIVE_MAX_COLUMNS: usize = 8;
117
118/// Krylov iterations, without restart, of one internal solve before the dense fallback.
119const KRYLOV_STEPS: usize = 16;
120
121impl InternalSolve {
122    pub(super) fn new(
123        lower: MatRef<'_, Complex>,
124        upper: MatRef<'_, Complex>,
125        rhs: DMatrix<Complex>,
126    ) -> Result<Self> {
127        Self::with_threshold(lower, upper, rhs, ITERATIVE_MIN_ROWS)
128    }
129
130    /// The pivoted LU solve, the branch [`Self::new`] takes whenever the Krylov
131    /// conditions fail.
132    pub(super) fn direct(
133        lower: MatRef<'_, Complex>,
134        upper: MatRef<'_, Complex>,
135        rhs: DMatrix<Complex>,
136    ) -> Result<Self> {
137        if rhs.iter().any(|&z| !finite(z)) {
138            return Err(Error::Singular);
139        }
140        Self::lu(internal_operator(lower, upper), rhs)
141    }
142
143    /// [`Self::new`] with a lower row threshold, which tests use to reach every
144    /// dispatch branch with small systems.
145    fn with_threshold(
146        lower: MatRef<'_, Complex>,
147        upper: MatRef<'_, Complex>,
148        rhs: DMatrix<Complex>,
149        min_rows: usize,
150    ) -> Result<Self> {
151        if rhs.iter().any(|&z| !finite(z)) {
152            return Err(Error::Singular);
153        }
154        let operator = internal_operator(lower, upper);
155        if rhs.nrows() >= min_rows
156            && rhs.ncols() <= ITERATIVE_MAX_COLUMNS
157            && is_contraction(&operator)
158            && let Ok(value) = internal_iteration(&operator, &rhs, false)
159        {
160            return Ok(Self {
161                value,
162                factor: InternalFactor::Krylov(operator),
163            });
164        }
165        Self::lu(operator, rhs)
166    }
167
168    /// Factor the operator and solve for `rhs` in place.
169    fn lu(operator: DMatrix<Complex>, mut rhs: DMatrix<Complex>) -> Result<Self> {
170        let lu = Lu::new(operator)?;
171        lu.solve_in_place(view_mut(&mut rhs))?;
172        if rhs.iter().any(|&z| !finite(z)) {
173            return Err(Error::Singular);
174        }
175        Ok(Self {
176            value: rhs,
177            factor: InternalFactor::Lu(lu),
178        })
179    }
180
181    /// Solve the adjoint system for `rhs`, consuming the forward solve.
182    pub(super) fn solve_adjoint(self, mut rhs: DMatrix<Complex>) -> Result<AdjointSolve> {
183        let lu = match self.factor {
184            InternalFactor::Lu(lu) => lu,
185            InternalFactor::Krylov(operator) => {
186                if let Ok(adjoint) = internal_iteration(&operator, &rhs, true) {
187                    return Ok(AdjointSolve {
188                        adjoint,
189                        value: self.value,
190                    });
191                }
192                Lu::new(operator)?
193            }
194        };
195        lu.solve_adjoint_in_place(view_mut(&mut rhs))?;
196        if rhs.iter().any(|&z| !finite(z)) {
197            return Err(Error::Singular);
198        }
199        Ok(AdjointSolve {
200            adjoint: rhs,
201            value: self.value,
202        })
203    }
204}

210    pub(super) adjoint: DMatrix<Complex>,

212    pub(super) value: DMatrix<Complex>,
213}

223    /// The complex Hermitian pairing `Σ conj(a) b`, from its real and imaginary parts.
224    fn pairing(a: &DMatrix<Complex>, b: &DMatrix<Complex>) -> Complex {
225        let rotated = a * Complex::i();
226        Complex::new(re_dot(a.iter(), b.iter()), re_dot(rotated.iter(), b.iter()))
227    }

229    /// For a weak `n`-by-`n` reflection pair, the Krylov solves of the operator and of
230    /// its adjoint match the dense LU solves and pair as adjoints:
231    /// `<g, A^-1 rhs> = <A^-H g, rhs>`.
232    fn check_krylov_matches_dense_solve(
233        n: usize,
234        p: usize,
235        seed: f64,
236    ) -> Result<(), TestCaseError> {
237        let weight = Complex::new(0.1 / f64::from(u32::try_from(n).unwrap()).sqrt(), 0.0);
238        let lower = patterned(n, n, seed) * weight;
239        let upper = patterned(n, n, seed + 0.4) * weight;
240        let rhs = patterned(n, p, seed + 0.2);
241        let g = patterned(n, p, seed + 0.7);
242        let operator = internal_operator(view(&lower), view(&upper));
243        let dense = linalg::solve_owned(operator.clone(), rhs.clone()).unwrap();
244        prop_assert!(is_contraction(&operator));
245        let value = internal_iteration(&operator, &rhs, false).unwrap();
246        let adjoint = internal_iteration(&operator, &g, true).unwrap();
247        prop_assert_close!(&value, dense.value(), 2e-14 * dense.value().norm());
248        let dense_adjoint = dense.adjoint_rhs(g.clone()).unwrap();
249        prop_assert_close!(&adjoint, &dense_adjoint, 2e-14 * g.norm());
250        prop_assert_close!(
251            pairing(&g, &value),
252            pairing(&adjoint, &rhs),
253            2e-13 * rhs.norm() * g.norm()
254        );
255        Ok(())
256    }

268    #[test]
269    fn production_threshold_solves_thin_contractions_iteratively() -> Result<(), TestCaseError> {
270        // Pin the production threshold from both sides: the same thin contraction
271        // iterates at `ITERATIVE_MIN_ROWS` rows and factors one row below it.
272        let coefficient = Complex::new(1.0, 0.0) - Complex::new(0.1, 0.03).powi(2);
273        for (n, iterates) in [(ITERATIVE_MIN_ROWS, true), (ITERATIVE_MIN_ROWS - 1, false)] {
274            let weak = DMatrix::identity(n, n) * Complex::new(0.1, 0.03);
275            let rhs = patterned(n, 2, 0.4);
276            let solve = InternalSolve::new(view(&weak), view(&weak), rhs.clone()).unwrap();
277            prop_assert_eq!(matches!(solve.factor, InternalFactor::Krylov(_)), iterates);
278            prop_assert_close!(&solve.value * coefficient, &rhs, 2e-13 * rhs.norm());
279            let gradient = solve.solve_adjoint(rhs.clone()).unwrap().adjoint;
280            prop_assert_close!(gradient * coefficient.conj(), &rhs, 2e-13 * rhs.norm());
281        }
282        Ok(())
283    }

285    #[test]
286    fn thin_large_solve_skips_lu_and_keeps_dense_fallback() -> Result<(), TestCaseError> {
287        // Every dispatch branch at a lowered threshold; the production threshold is
288        // pinned by `production_threshold_solves_thin_contractions_iteratively`.
289        // The cyclic reflection below has n distinct eigenvalues on a circle, so the
290        // Krylov trial cannot converge only while n exceeds its iteration budget.
291        let (n, min_rows) = (24, 16);
292        prop_assert!(n > KRYLOV_STEPS);
293        let solve = |lower: &DMatrix<Complex>, upper: &DMatrix<Complex>, rhs, min_rows| {
294            InternalSolve::with_threshold(view(lower), view(upper), rhs, min_rows)
295        };
296        let identity = DMatrix::identity(n, n);
297        let weak = &identity * Complex::new(0.1, 0.03);
298        let rhs = patterned(n, 2, 0.4);
299        let iterative = solve(&weak, &weak, rhs.clone(), min_rows).unwrap();
300        prop_assert!(matches!(iterative.factor, InternalFactor::Krylov(_)));
301        let coefficient = Complex::new(1.0, 0.0) - Complex::new(0.1, 0.03).powi(2);
302        prop_assert_close!(&iterative.value * coefficient, &rhs, 2e-13 * rhs.norm());
303        let gradient = iterative.solve_adjoint(rhs.clone()).unwrap().adjoint;
304        prop_assert_close!(gradient * coefficient.conj(), &rhs, 2e-13 * rhs.norm());
305
306        // The threshold is inclusive, and wide batches or short systems use LU.
307        let at = solve(&weak, &weak, rhs.clone(), n).unwrap();
308        prop_assert!(matches!(at.factor, InternalFactor::Krylov(_)));
309        let below = solve(&weak, &weak, rhs.clone(), n + 1).unwrap();
310        prop_assert!(matches!(below.factor, InternalFactor::Lu(_)));
311        prop_assert_close!(&below.value, &at.value, 2e-13 * rhs.norm());
312        let wide = patterned(n, ITERATIVE_MAX_COLUMNS + 1, 0.4);
313        let batch = solve(&weak, &weak, wide.clone(), min_rows).unwrap();
314        prop_assert!(matches!(batch.factor, InternalFactor::Lu(_)));
315        prop_assert_close!(&batch.value * coefficient, &wide, 2e-13 * wide.norm());
316
317        // A strongly reflecting cyclic system needs more iterations than the bounded
318        // Krylov trial allows, so it takes the direct solve.
319        let mut cyclic = DMatrix::zeros(n, n);
320        for i in 0..n {
321            cyclic[(i, (i + 1) % n)] = Complex::new(0.95, 0.0);
322        }
323        let dense = solve(&cyclic, &identity, rhs.clone(), min_rows).unwrap();
324        prop_assert!(matches!(dense.factor, InternalFactor::Lu(_)));
325        let operator = internal_operator(view(&cyclic), view(&identity));
326        prop_assert!(is_contraction(&operator));
327        prop_assert_close!(&operator * &dense.value, &rhs, 2e-13 * rhs.norm());
328        let eigenvector = DMatrix::from_element(n, 2, Complex::new(1.0, 0.0));
329        let easy_forward = solve(&cyclic, &identity, eigenvector, min_rows).unwrap();
330        prop_assert!(matches!(easy_forward.factor, InternalFactor::Krylov(_)));
331        // A forward eigenvector can converge immediately while an unrelated
332        // adjoint RHS needs the direct fallback. Certify that path separately.
333        let reverse = easy_forward.solve_adjoint(rhs.clone()).unwrap().adjoint;
334        prop_assert_close!(operator.adjoint() * reverse, &rhs, 2e-13 * rhs.norm());
335
336        // Singular systems stay errors, including consistent and zero right-hand sides.
337        prop_assert!(solve(&identity, &identity, rhs, min_rows).is_err());
338        let mut partial = DMatrix::zeros(n, n);
339        partial[(0, 0)] = Complex::new(1.0, 0.0);
340        let mut in_range = DMatrix::zeros(n, 1);
341        in_range[(1, 0)] = Complex::new(1.0, 0.0);
342        prop_assert!(solve(&partial, &identity, in_range, min_rows).is_err());
343        prop_assert!(solve(&identity, &identity, DMatrix::zeros(n, 1), min_rows).is_err());
344        Ok(())
345    }
