1//! Fixed-shape metric gradient state, including deferred nonsmoothness errors.
2//! treams-rs extension.
3
4use super::MetricResidual;
5use crate::saved::{Reader, SavedState, Writer, invalid};
6use crate::{Complex, Error, Result};
7use nalgebra::DMatrix;
8
9// Status zero means a saved gradient; the remaining tags identify the only
10// errors chirality and SvdvalsResidual::pullback can save in MetricResidual.
11const GRADIENT_ERRORS: [&str; 4] = [
12    "chirality is not differentiable at zero contrast",
13    "invalid singular-value cotangent",
14    "nonzero singular-value weight at zero is not differentiable",
15    "repeated singular values require equal cotangent weights",
16];
17
18impl MetricResidual {
19    /// Number of bytes in the fixed derivative state for a square metric input.
20    pub fn state_size(dimension: usize) -> Result<usize> {
21        if dimension == 0 {
22            return Err(invalid());
23        }
24        dimension
25            .checked_mul(dimension)
26            .and_then(|count| count.checked_mul(16))
27            .and_then(|bytes| bytes.checked_add(25))
28            .ok_or_else(invalid)
29    }
30}
31
32impl SavedState for MetricResidual {
33    fn save_state(&self) -> Result<Vec<u8>> {
34        let status = match &self.gradient {
35            Ok(_) => 0,
36            Err(Error::InvalidInput(message)) => GRADIENT_ERRORS
37                .iter()
38                .zip(1_u8..)
39                .find_map(|(&known, status)| (known == message).then_some(status))
40                .ok_or_else(invalid)?,
41            Err(_) => return Err(invalid()),
42        };
43        let mut writer = Writer::new(Self::state_size(self.dimension)?);
44        writer.usize(self.dimension);
45        writer.byte(status);
46        for gradient in self.ks_gradient {
47            writer.f64(gradient);
48        }
49        match &self.gradient {
50            Ok(gradient) => {
51                for &value in gradient.iter() {
52                    writer.complex(value);
53                }
54            }
55            Err(_) => {
56                // An undefined gradient has the same shape as a valid one and
57                // reserves canonical zeros instead of variable-length text.
58                for _ in 0..self.dimension * self.dimension {
59                    writer.complex(Complex::default());
60                }
61            }
62        }
63        Ok(writer.finish())
64    }
65
66    fn from_state(bytes: &[u8]) -> Result<Self> {
67        let mut reader = Reader::new(bytes);
68        let dimension = reader.usize()?;
69        if bytes.len() != Self::state_size(dimension)? {
70            return Err(invalid());
71        }
72        let status = reader.byte()?;
73        let error = if status == 0 {
74            None
75        } else {
76            Some(
77                *GRADIENT_ERRORS
78                    .get(usize::from(status) - 1)
79                    .ok_or_else(invalid)?,
80            )
81        };
82        let ks_gradient = [reader.f64()?, reader.f64()?];
83        let gradient = if let Some(message) = error {
84            for _ in 0..dimension * dimension {
85                if reader.complex()? != Complex::default() {
86                    return Err(invalid());
87                }
88            }
89            Err(Error::InvalidInput(message.into()))
90        } else {
91            let values = (0..dimension * dimension)
92                .map(|_| reader.complex())
93                .collect::<Result<Vec<_>>>()?;
94            Ok(DMatrix::from_vec(dimension, dimension, values))
95        };
96        reader.finish()?;
97        Ok(Self {
98            gradient,
99            ks_gradient,
100            dimension,
101        })
102    }
103}

112    #[test]
113    fn roundtrip_preserves_repeated_pushforward_and_pullback() {
114        let matrix = DMatrix::from_row_slice(
115            2,
116            2,
117            &[
118                Complex::new(-0.3, 0.03),
119                Complex::new(0.02, 0.01),
120                Complex::new(0.04, -0.02),
121                Complex::new(-0.2, -0.01),
122            ],
123        );
124        let tangent = matrix.map(|z| Complex::new(z.im, z.re));
125        for kind in [
126            Metric::CircularDichroism,
127            Metric::DualityBreaking,
128            Metric::Chirality,
129        ] {
130            let (_, residual) = metric(&matrix, &[0, 1], [1.0, 1.3], kind).unwrap();
131            let bytes = residual.save_state().unwrap();
132            assert_eq!(bytes.len(), MetricResidual::state_size(2).unwrap());
133            let restored = MetricResidual::from_state(&bytes).unwrap();
134            assert_eq!(restored.shape(), (2, 2));
135            assert_eq!(restored.save_state().unwrap(), bytes);
136            for weight in [1.2, 0.0, -0.7, 1.2] {
137                let expected = residual.pullback(weight).unwrap();
138                let actual = restored.pullback(weight).unwrap();
139                assert_eq!(actual.matrix, expected.matrix);
140                assert_eq!(actual.ks, expected.ks);
141                assert_eq!(
142                    restored.pushforward(&tangent, [0.2, -0.1]).unwrap(),
143                    residual.pushforward(&tangent, [0.2, -0.1]).unwrap()
144                );
145            }
146        }
147    }

149    #[test]
150    fn roundtrip_preserves_deferred_errors_and_zero_directions() {
151        for message in GRADIENT_ERRORS {
152            let residual = MetricResidual {
153                gradient: Err(Error::InvalidInput(message.into())),
154                ks_gradient: [0.0; 2],
155                dimension: 2,
156            };
157            let bytes = residual.save_state().unwrap();
158            assert!(bytes[25..].iter().all(|&byte| byte == 0));
159            let restored = MetricResidual::from_state(&bytes).unwrap();
160            assert_eq!(restored.save_state().unwrap(), bytes);
161            for _ in 0..2 {
162                assert_eq!(restored.pullback(1.0).unwrap_err().to_string(), message);
163                assert_eq!(
164                    restored
165                        .pushforward(&DMatrix::identity(2, 2), [0.0; 2])
166                        .unwrap_err()
167                        .to_string(),
168                    message
169                );
170                assert_eq!(restored.pullback(0.0).unwrap().matrix, DMatrix::zeros(2, 2));
171                assert_eq!(
172                    restored
173                        .pushforward(&DMatrix::zeros(2, 2), [0.0; 2])
174                        .unwrap(),
175                    0.0
176                );
177            }
178        }
179    }

181    #[test]
182    fn rejects_invalid_shape_status_and_length_before_allocation() {
183        let (_, residual) = metric(
184            &DMatrix::identity(2, 2),
185            &[0, 1],
186            [1.0; 2],
187            Metric::Chirality,
188        )
189        .unwrap();
190        let bytes = residual.save_state().unwrap();
191        for len in 0..bytes.len() {
192            assert!(MetricResidual::from_state(&bytes[..len]).is_err());
193        }
194        let mut trailing = bytes.clone();
195        trailing.push(0);
196        assert!(MetricResidual::from_state(&trailing).is_err());
197        for dimension in [0_u64, u64::MAX, u64::MAX / 16] {
198            let mut invalid = bytes.clone();
199            invalid[..8].copy_from_slice(&dimension.to_le_bytes());
200            assert!(MetricResidual::from_state(&invalid).is_err());
201        }
202        let mut invalid = bytes.clone();
203        invalid[8] = 255;
204        assert!(MetricResidual::from_state(&invalid).is_err());
205        let mut invalid = bytes;
206        invalid[25..33].copy_from_slice(&1.0_f64.to_le_bytes());
207        assert!(MetricResidual::from_state(&invalid).is_err());
208        assert!(MetricResidual::state_size(usize::MAX).is_err());
209    }
