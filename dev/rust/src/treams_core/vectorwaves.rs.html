1//! Vector waves one at a time: the vector spherical harmonics and the spherical,
2//! cylindrical and plane vector waves in their local component frames, with pullbacks.
3//!
4//! Upstream: `treams.special.vsh_X`, `vsh_Y`, `vsh_Z`, `vsw_*`, `vcw_*`, `vpw_*` and
5//! `sph_harm`.
6//!
7//! [`vector_wave`] evaluates one wave of a [`Family`] at six complex arguments,
8//! [`vector_wave_pullback`] gives the gradients of all six, and [`vector_wave_array`]
9//! broadcasts both over arrays of labels and arguments. Spherical waves are in spherical
10//! components `(r, theta, phi)`, cylindrical waves in `(rho, phi, z)` and plane waves in
11//! Cartesian components, as in treams.
12//!
13//! The inline `tests` module checks the waves against the Cartesian fields of
14//! [`crate::fields`], their pullbacks against finite differences, the small-argument
15//! branches and the normalization of the harmonics.
16#![allow(clippy::indexing_slicing)] // Fixed wave argument and component arrays.
17

20    numerics::{Jet, broadcast, finite, parallel::Parallel},
21    special::{self, Radial, SERIES_RADIUS, polar_trig, polarized_wave, radial_jet},
22};
23use std::f64::consts::PI;

25/// The kind of vector wave [`vector_wave`] evaluates.
26///
27/// Polarization 0/1 selects M/N (upstream `*_M`/`*_N`), or negative/positive helicity
28/// (upstream `*_A`) when `helicity` is true. Harmonics ignore the polarization.
29///
30/// Upstream: `treams.special.vsh_*`, `vsw_*`, `vcw_*` and `vpw_*`; each variant names
31/// its functions.
32#[derive(Clone, Copy, Debug)]
33pub enum Family {
34    /// Tangential X harmonic in spherical components (upstream `vsh_X`).
35    HarmonicX,
36    /// Tangential Y harmonic in spherical components (upstream `vsh_Y`).
37    HarmonicY,
38    /// Radial Z harmonic, equal to i times the scalar spherical harmonic (upstream
39    /// `vsh_Z`).
40    HarmonicZ,
41    /// Spherical vector wave at `(kr, theta, phi)`: upstream `vsw_rM`, `vsw_rN` and
42    /// `vsw_rA` when regular, `vsw_M`, `vsw_N` and `vsw_A` when singular.
43    Spherical(Radial),
44    /// Cylindrical vector wave at `(kz, k_rho*rho, phi, z, k)`: upstream `vcw_rM`,
45    /// `vcw_rN` and `vcw_rA` when regular, `vcw_M`, `vcw_N` and `vcw_A` when singular.
46    Cylindrical(Radial),
47    /// Plane vector wave at `(kx, ky, kz, x, y, z)`: upstream `vpw_M`, `vpw_N` and
48    /// `vpw_A`.
49    Plane,
50}
51
52/// The integer labels of one vector wave.
53///
54/// Each [`Family`] reads some of the fields:
55///
56/// | Family | Reads |
57/// |---|---|
58/// | `HarmonicX`, `HarmonicY`, `HarmonicZ` | `l`, `m` |
59/// | `Spherical` | `l`, `m`, `pol` |
60/// | `Cylindrical` | `m`, `pol` |
61/// | `Plane` | `pol` |
62///
63/// Every evaluation accepts `0 <= l <= 128`, `|m| <= 128` and `pol` 0 or 1, whatever the
64/// family. Every harmonic and spherical wave with `|m| > l` is zero. At `l = 0` the
65/// spherical waves and the X and Y harmonics are zero, and the Z harmonic is `i Y_00`.
66#[derive(Clone, Copy, Debug, PartialEq, Eq)]
67pub struct WaveLabel {
68    /// Degree.
69    pub l: i32,
70    /// Order.
71    pub m: i32,
72    /// Polarization index: M/N (0/1), or negative/positive helicity.
73    pub pol: u8,
74}

76/// The normalized scalar spherical harmonic `Y_lm(theta, phi)` and, if `tangential`
77/// and `l > 0`, the angular functions `pi = m Y_lm / sin(theta)` and
78/// `tau = dY_lm / dtheta`, both divided by `sqrt(l (l + 1))`: `[Y, pi, tau]`, or
79/// `[Y, 0, 0]` otherwise.
80///
81/// The recurrence runs over normalized associated Legendre functions: it starts from
82/// the normalized diagonal term of degree `|m|` and steps the degree up to `l`. Every
83/// term stays of order one, which avoids the factorial overflow of the unnormalized
84/// functions at high orders. For `m = 0` the formula for `tau` used at other orders
85/// would divide by `sin(theta)`, so a second recurrence gives the derivative with
86/// respect to `cos(theta)` instead.
87fn angular<const N: usize>(
88    l: i32,
89    m: i32,
90    theta: Jet<N>,
91    phi: Jet<N>,
92    tangential: bool,
93) -> [Jet<N>; 3] {
94    let [cosine, sine] = polar_trig(theta);
95    let order = m.abs();
96    // Normalize the Legendre recurrence before evaluating it. This removes
97    // factorial overflow for high orders and repeated adjacent-order recurrences.
98    let mut diagonal = (1.0 / (4.0 * PI)).sqrt();
99    for j in 1..=order {
100        diagonal *= -(f64::from(2 * j + 1) / f64::from(2 * j)).sqrt();
101    }
102    if m < 0 && order % 2 != 0 {
103        diagonal = -diagonal;
104    }
105    let mut previous = Jet::default();
106    let mut polynomial = Jet::constant(diagonal);
107    let mut previous_derivative = Jet::default();
108    let mut derivative = Jet::default();
109    for j in order + 1..=l {
110        let denominator = f64::from(j * j - order * order);
111        let a = (f64::from(4 * j * j - 1) / denominator).sqrt();
112        let b = if j == order + 1 {
113            0.0
114        } else {
115            (f64::from((2 * j + 1) * ((j - 1) * (j - 1) - order * order))
116                / (f64::from(2 * j - 3) * denominator))
117                .sqrt()
118        };
119        if order == 0 && tangential {
120            let next = a * (polynomial + cosine * derivative) - b * previous_derivative;
121            previous_derivative = derivative;
122            derivative = next;
123        }
124        let next = a * cosine * polynomial - b * previous;
125        previous = polynomial;
126        polynomial = next;
127    }
128    let phase = (Complex::i() * f64::from(m) * phi).exp();
129    let power = sine.powi((order - 1).max(0));
130    let harmonic = phase
131        * polynomial
132        * if order == 0 {
133            Jet::constant(1.0)
134        } else {
135            power * sine
136        };
137    if l == 0 || !tangential {
138        return [harmonic, Jet::default(), Jet::default()];
139    }
140    let norm = f64::from(l * (l + 1)).sqrt();
141    if order == 0 {
142        return [harmonic, Jet::default(), -phase * sine * derivative / norm];
143    }
144    let adjacent = (f64::from((l * l - order * order) * (2 * l + 1)) / f64::from(2 * l - 1)).sqrt();
145    [
146        harmonic,
147        phase * f64::from(m) * power * polynomial / norm,
148        phase * power * (f64::from(l) * cosine * polynomial - adjacent * previous) / norm,
149    ]
150}

152/// The scalar harmonic `Y` and, if `tangential`, the angular functions `pi` and
153/// `tau` of `label`; `None` when `|m| > l`.
154fn harmonics<const N: usize>(
155    label: WaveLabel,
156    theta: Jet<N>,
157    phi: Jet<N>,
158    tangential: bool,
159) -> Option<[Jet<N>; 3]> {
160    (label.m.abs() <= label.l).then(|| angular(label.l, label.m, theta, phi, tangential))
161}
162
163/// The tangential X and Y vector harmonics from `[Y, pi, tau]`.
164fn tangential<const N: usize>([_, pi, tau]: [Jet<N>; 3]) -> [[Jet<N>; 3]; 2] {
165    [
166        [Jet::default(), -pi, -Complex::i() * tau],
167        [Jet::default(), Complex::i() * tau, -pi],
168    ]
169}

171/// Spherical M and N waves at `(kr, theta, phi)`, or their helicity combinations.
172fn spherical<const N: usize>(
173    radial: Radial,
174    label: WaveLabel,
175    [kr, theta, phi]: [Jet<N>; 3],
176    helicity: bool,
177) -> Result<[Jet<N>; 3]> {
178    if label.l == 0 {
179        return Ok([Jet::default(); 3]);
180    }
181    let Some(harmonics) = harmonics(label, theta, phi, true) else {
182        return Ok([Jet::default(); 3]);
183    };
184    let [x, tangent] = tangential(harmonics);
185    if !helicity && label.pol == 0 {
186        let value = radial_jet(label.l, kr, radial, true)?;
187        return Ok(x.map(|v| v * value));
188    }
189    let l = label.l.unsigned_abs();
190    let bessel = special::spherical_radial(l, kr.value, radial)?;
191    let value = kr.chain(bessel.value, bessel.first);
192    let first = kr.chain(bessel.first, bessel.second);
193    let divided = if radial == Radial::Regular && kr.value.norm() < SERIES_RADIUS {
194        let lower = special::spherical_radial(l - 1, kr.value, radial)?;
195        let upper = special::spherical_radial(l + 1, kr.value, radial)?;
196        kr.chain(
197            (lower.value + upper.value) / f64::from(2 * l + 1),
198            (lower.first + upper.first) / f64::from(2 * l + 1),
199        )
200    } else {
201        value / kr
202    };
203    let mut n = tangent.map(|v| v * (divided + first));
204    n[0] = Complex::i() * harmonics[0] * divided * f64::from(label.l * (label.l + 1)).sqrt();
205    Ok(polarized_wave(x.map(|v| v * value), n, label.pol, helicity))
206}

208/// Cylindrical M and N waves at `(kz, krr, phi, z, k)`, or their helicity combinations;
209/// `krr` is `k_rho rho`, as in upstream `vcw_*`.
210fn cylindrical<const N: usize>(
211    radial: Radial,
212    label: WaveLabel,
213    [kz, krr, phi, z, k, _]: [Jet<N>; 6],
214    helicity: bool,
215) -> Result<[Jet<N>; 3]> {
216    let bessel = special::cylindrical_radial(label.m, krr.value, radial)?;
217    let value = krr.chain(bessel.value, bessel.first);
218    let first = krr.chain(bessel.first, bessel.second);
219    let divided = if label.m == 0 {
220        Jet::default()
221    } else if radial == Radial::Regular && krr.value.norm() < SERIES_RADIUS {
222        let lower = special::cylindrical_radial(label.m - 1, krr.value, radial)?;
223        let upper = special::cylindrical_radial(label.m + 1, krr.value, radial)?;
224        krr.chain(
225            0.5 * (lower.value + upper.value),
226            0.5 * (lower.first + upper.first),
227        )
228    } else {
229        f64::from(label.m) * value / krr
230    };
231    let phase = (Complex::i() * (f64::from(label.m) * phi + kz * z)).exp();
232    let m = [
233        Complex::i() * divided * phase,
234        -first * phase,
235        Jet::default(),
236    ];
237    if !helicity && label.pol == 0 {
238        return Ok(m);
239    }
240    if k.value == Complex::default() {
241        return Err(Error::InvalidInput(
242            "cylindrical N waves require nonzero k".into(),
243        ));
244    }
245    // The principal root, as upstream `vcw_N` takes it. For a gain medium it can be the
246    // negative of the root with a nonnegative imaginary part that `cw` uses.
247    let transverse = (k * k - kz * kz).sqrt();
248    let factor = phase / k;
249    let n = [
250        Complex::i() * kz * first * factor,
251        -kz * divided * factor,
252        transverse * value * factor,
253    ];
254    Ok(polarized_wave(m, n, label.pol, helicity))
255}

257/// Plane wave `p exp(i k·r)` at `(kx, ky, kz, x, y, z)`, with the chain rule through
258/// the polarization `p(k)`.
259fn plane<const N: usize>(
260    label: WaveLabel,
261    args: [Jet<N>; 6],
262    helicity: bool,
263) -> Result<[Jet<N>; 3]> {
264    let vector = [args[0].value, args[1].value, args[2].value];
265    if N == 0 {
266        return Ok(crate::pw::field_value(
267            crate::pw::polarization(vector, label.pol, helicity)?,
268            vector,
269            [args[3].value, args[4].value, args[5].value],
270        )?
271        .map(Jet::constant));
272    }
273    let polarization = crate::pw::polarization_jet::<3>(vector, label.pol, helicity)?
274        .map(|p| p.compose([args[0], args[1], args[2]]));
275    let phase = (Complex::i() * (0..3).map(|a| args[a] * args[a + 3]).sum::<Jet<N>>()).exp();
276    Ok(polarization.map(|p| p * phase))
277}

279#[inline]
280fn evaluate<const N: usize>(
281    family: Family,
282    label: WaveLabel,
283    args: [Jet<N>; 6],
284    helicity: bool,
285) -> Result<[Jet<N>; 3]> {
286    if label.pol > 1
287        || label.l < 0
288        || label.l > MAX_DEGREE
289        || label.m.unsigned_abs() > MAX_DEGREE.unsigned_abs()
290        || args.iter().any(|v| !finite(v.value))
291    {
292        return Err(Error::InvalidInput(
293            // 128 is MAX_DEGREE.
294            "require finite wave arguments, 0 <= l <= 128, |m| <= 128 and polarization 0 or 1"
295                .into(),
296        ));
297    }
298    let [theta, phi] = [args[0], args[1]];
299    let zero = [Jet::default(); 3];
300    Ok(match family {
301        Family::HarmonicX => harmonics(label, theta, phi, true).map_or(zero, |h| tangential(h)[0]),
302        Family::HarmonicY => harmonics(label, theta, phi, true).map_or(zero, |h| tangential(h)[1]),
303        Family::HarmonicZ => harmonics(label, theta, phi, false).map_or(zero, |[y, _, _]| {
304            [Complex::i() * y, Jet::default(), Jet::default()]
305        }),
306        Family::Spherical(radial) => {
307            spherical(radial, label, [args[0], args[1], args[2]], helicity)?
308        }
309        Family::Cylindrical(radial) => cylindrical(radial, label, args, helicity)?,
310        Family::Plane => plane(label, args, helicity)?,
311    })
312}
313fn checked<const N: usize>(result: [Jet<N>; 3]) -> Result<[Jet<N>; 3]> {
314    if result.iter().any(|v| !v.finite()) {
315        return Err(Error::NonFinite(
316            "non-finite vector wave or undefined derivative".into(),
317        ));
318    }
319    Ok(result)
320}

322/// Evaluate a vector wave in its local component frame.
323///
324/// Harmonics use `(theta,phi)`, spherical waves `(kr,theta,phi)`, cylinders
325/// `(kz,k_rho*rho,phi,z,k)`, and planes `(kx,ky,kz,x,y,z)`. Unused arguments are zero.
326///
327/// Upstream: `treams.special.vsh_X`, `vsh_Y`, `vsh_Z`, `vsw_*`, `vcw_*` and `vpw_*`.
328#[inline]
329pub fn vector_wave(
330    family: Family,
331    label: WaveLabel,
332    args: [Complex; 6],
333    helicity: bool,
334) -> Result<[Complex; 3]> {
335    Ok(checked(evaluate::<0>(
336        family,
337        label,
338        args.map(Jet::constant),
339        helicity,
340    )?)?
341    .map(|v| v.value))
342}

344/// The gradients of all six complex arguments of [`vector_wave`] for a `cotangent` of
345/// the wave. A real physical input takes the real part of its gradient.
346pub fn vector_wave_pullback(
347    family: Family,
348    label: WaveLabel,
349    args: [Complex; 6],
350    helicity: bool,
351    cotangent: [Complex; 3],
352) -> Result<[Complex; 6]> {
353    if cotangent.iter().any(|&v| !finite(v)) {
354        return Err(Error::InvalidInput(
355            "vector-wave cotangent must be finite".into(),
356        ));
357    }
358    if cotangent == [Complex::default(); 3] {
359        vector_wave(family, label, args, helicity)?;
360        return Ok([Complex::default(); 6]);
361    }
362    match family {
363        Family::HarmonicX | Family::HarmonicY | Family::HarmonicZ => {
364            pullback_impl::<2>(family, label, args, helicity, cotangent)
365        }
366        Family::Spherical(_) => pullback_impl::<3>(family, label, args, helicity, cotangent),
367        Family::Cylindrical(_) => pullback_impl::<5>(family, label, args, helicity, cotangent),
368        Family::Plane => pullback_impl::<6>(family, label, args, helicity, cotangent),
369    }
370}
371/// [`vector_wave_pullback`] with derivatives in the first `N` arguments, the ones the
372/// family reads; the others get zero.
373fn pullback_impl<const N: usize>(
374    family: Family,
375    label: WaveLabel,
376    args: [Complex; 6],
377    helicity: bool,
378    cotangent: [Complex; 3],
379) -> Result<[Complex; 6]> {
380    let values = checked(evaluate::<N>(
381        family,
382        label,
383        std::array::from_fn(|a| Jet::variable(args[a], a)),
384        helicity,
385    )?)?;
386    Ok(std::array::from_fn(|a| {
387        values
388            .iter()
389            .zip(cotangent)
390            .map(|(v, g)| v.derivative.get(a).copied().unwrap_or_default().conj() * g)
391            .sum()
392    }))
393}

395/// Vector-wave arrays evaluate in parallel from this many elements.
396const PARALLEL: Parallel = Parallel::AtLeast(1024);

398/// What [`vector_wave_array`] saves for its pullback: the family, the labels, the
399/// convention and the arguments, each one value for all outputs or one per output.
400///
401/// The pullback computes each output's gradients on its own and adds the gradients of a
402/// scalar argument in output order on the calling thread, so its result does not
403/// depend on the thread count.
404#[derive(Debug)]
405pub struct VectorWaveResidual {
406    family: Family,
407    labels: Vec<WaveLabel>,
408    arguments: [Vec<Complex>; 6],
409    helicity: bool,
410    size: usize,
411}
412impl VectorWaveResidual {
413    fn element(&self, i: usize) -> (WaveLabel, [Complex; 6]) {
414        (
415            broadcast::element(&self.labels, i),
416            self.arguments.each_ref().map(|a| broadcast::element(a, i)),
417        )
418    }
419    /// The label and wave vector that several plane waves share, and hence their
420    /// one polarization, when they differ only in their points.
421    fn fixed_plane(&self) -> Option<(WaveLabel, [Complex; 3])> {
422        let [kx, ky, kz, ..] = self.arguments.each_ref().map(Vec::as_slice);
423        match (self.family, self.labels.as_slice(), kx, ky, kz) {
424            (Family::Plane, &[label], &[kx], &[ky], &[kz]) if self.size > 1 => {
425                Some((label, [kx, ky, kz]))
426            }
427            _ => None,
428        }
429    }
430    /// The point `(x, y, z)` of plane wave `i`.
431    fn point(&self, i: usize) -> [Complex; 3] {
432        let [_, _, _, x, y, z] = &self.arguments;
433        [x, y, z].map(|v| broadcast::element(v, i))
434    }
435    /// The gradients of the six arguments from one complex vector cotangent per output;
436    /// an argument given as one value for all outputs gets the sum of its gradients.
437    pub fn pullback(self, cotangent: &[[Complex; 3]]) -> Result<[Vec<Complex>; 6]> {
438        // Waves of one direction share the polarization jet; each element then
439        // reads only its point.
440        let plane = match self.fixed_plane() {
441            Some((label, k)) if cotangent.iter().flatten().any(|&v| v != Complex::default()) => {
442                let p = crate::pw::polarization_jet::<3>(k, label.pol, self.helicity)?;
443                Some((p, k))
444            }
445            _ => None,
446        };
447        broadcast::pullback(
448            cotangent,
449            self.size,
450            "wave cotangent must be finite and match output",
451            self.arguments.each_ref().map(Vec::len),
452            PARALLEL,
453            |i, &g| {
454                if let Some((p, k)) = &plane {
455                    fixed_plane_pullback(p, *k, self.point(i), g)
456                } else {
457                    let (m, a) = self.element(i);
458                    vector_wave_pullback(self.family, m, a, self.helicity, g)
459                }
460            },
461        )
462    }
463}

465/// Pullback to `(k, r)` of a plane wave `p exp(i k·r)`, given the polarization jet
466/// `p` in the wave vector.
467fn fixed_plane_pullback(
468    p: &[Jet<3>; 3],
469    k: [Complex; 3],
470    r: [Complex; 3],
471    g: [Complex; 3],
472) -> Result<[Complex; 6]> {
473    let phase = (Complex::i() * (0..3).map(|j| k[j] * r[j]).sum::<Complex>()).exp();
474    let g = g.map(|v| v * phase.conj());
475    let amplitude = p
476        .iter()
477        .zip(g)
478        .map(|(p, g)| p.value.conj() * g)
479        .sum::<Complex>();
480    let gradient = std::array::from_fn(|j| {
481        if j < 3 {
482            p.iter()
483                .zip(g)
484                .map(|(p, g)| p.derivative[j].conj() * g)
485                .sum::<Complex>()
486                + (Complex::i() * r[j]).conj() * amplitude
487        } else {
488            (Complex::i() * k[j - 3]).conj() * amplitude
489        }
490    });
491    if gradient.iter().any(|&v| !finite(v)) {
492        return Err(Error::NonFinite("non-finite plane-wave pullback".into()));
493    }
494    Ok(gradient)
495}

497/// [`vector_wave`] over arrays of labels and arguments, each one value for all outputs
498/// or one per output; the residual keeps only the inputs.
499///
500/// Upstream: `treams.special.vsh_*`, `vsw_*`, `vcw_*` and `vpw_*` on broadcast arrays.
501pub fn vector_wave_array(
502    family: Family,
503    labels: Vec<WaveLabel>,
504    arguments: [Vec<Complex>; 6],
505    helicity: bool,
506) -> Result<(Vec<[Complex; 3]>, VectorWaveResidual)> {
507    let [a, b, c, d, e, f] = arguments.each_ref().map(Vec::len);
508    let size = broadcast::size(
509        &[labels.len(), a, b, c, d, e, f],
510        "wave arrays must have equal lengths or scalar inputs",
511    )?;
512    let residual = VectorWaveResidual {
513        family,
514        labels,
515        arguments,
516        helicity,
517        size,
518    };
519    let plane = match residual.fixed_plane() {
520        Some((label, k)) => {
521            // Evaluating the first wave checks the shared label and wavevector once.
522            vector_wave(family, label, residual.element(0).1, helicity)?;
523            Some((crate::pw::polarization(k, label.pol, helicity)?, k))
524        }
525        None => None,
526    };
527    let values = broadcast::map(size, PARALLEL, |i| {
528        if let Some((p, k)) = plane {
529            crate::pw::field_value(p, k, residual.point(i))
530        } else {
531            let (m, a) = residual.element(i);
532            vector_wave(family, m, a, helicity)
533        }
534    })?;
535    Ok((values, residual))
536}

538/// Scalar spherical harmonic `Y_l^m(theta, phi)` at the polar angle `theta` and the
539/// azimuth `phi`.
540///
541/// `Y` is `-i` times the radial component of the [`Family::HarmonicZ`] wave
542/// ([`sph_harm_from_vsh_z`]).
543///
544/// Upstream: `treams.special.sph_harm`, which takes `(m, l, phi, theta)`.
545/// Differences: degrees outside `[0, 128]`, orders with `|m| > 128` and non-finite angles
546/// give an error, where treams returns a value or NaN; other orders with `|m| > l` give
547/// zero, where treams returns NaN.
548#[inline]
549pub fn sph_harm(l: i32, m: i32, theta: Complex, phi: Complex) -> Result<Complex> {
550    let zero = Complex::default();
551    let label = WaveLabel { l, m, pol: 0 };
552    let wave = vector_wave(
553        Family::HarmonicZ,
554        label,
555        [theta, phi, zero, zero, zero, zero],
556        false,
557    )?;
558    Ok(sph_harm_from_vsh_z(wave))
559}

561/// The scalar spherical harmonic `Y = -i Z_r` of a [`Family::HarmonicZ`] wave `Z`
562/// (`treams.special.vsh_Z`), whose only nonzero component is the radial one, `Z_r = i Y`.
563#[inline]
564#[must_use]
565pub fn sph_harm_from_vsh_z([z_r, _, _]: [Complex; 3]) -> Complex {
566    -Complex::i() * z_r
567}
568
569/// Pullback of [`sph_harm_from_vsh_z`]: the [`Family::HarmonicZ`] wave cotangent
570/// `[i cotangent, 0, 0]` for the scalar-harmonic `cotangent`.
571#[inline]
572#[must_use]
573pub fn sph_harm_from_vsh_z_pullback(cotangent: Complex) -> [Complex; 3] {
574    [
575        Complex::i() * cotangent,
576        Complex::default(),
577        Complex::default(),
578    ]
579}

660    /// Spherical and cylindrical waves in their local frames, rotated to Cartesian
661    /// components, equal the Cartesian fields of `fields`, on both sides of the
662    /// small-argument series switch `|z| = SERIES_RADIUS` (0.5).
663    fn check_reconstruction(
664        l: i32,
665        m: i32,
666        [r, theta, phi]: [f64; 3],
667        helicity: bool,
668        radial: Radial,
669    ) -> Result<(), TestCaseError> {
670        let k = Complex::new(1.3, 0.05);
671        let kz = 0.2;
672        let krho = (k * k - kz * kz).sqrt();
673        for pol in [0, 1] {
674            let label = WaveLabel { l, m, pol };
675            let spherical = [r, theta, phi];
676            let args = arguments([k * r, theta.into(), phi.into()]);
677            let local = vector_wave(Family::Spherical(radial), label, args, helicity).unwrap();
678            let actual = coordinates::vector(local, spherical, Transform::SphToCar).unwrap();
679            let position = coordinates::point(spherical, Transform::SphToCar).unwrap();
680            let mode = sw::Mode { l, m, pol };
681            let expected = fields::spherical_wave(mode, k, position, helicity, radial)
682                .unwrap()
683                .value;
684            let scale = 1e-11 * (1.0 + expected.map(Complex::norm).iter().sum::<f64>());
685            prop_assert_close!(actual, expected, scale, "spherical {label:?}");
686            let cylindrical = [r, phi, 0.3];
687            let args = arguments([kz.into(), krho * r, phi.into(), 0.3.into(), k]);
688            let local = vector_wave(Family::Cylindrical(radial), label, args, helicity).unwrap();
689            let actual = coordinates::vector(local, cylindrical, Transform::CylToCar).unwrap();
690            let position = coordinates::point(cylindrical, Transform::CylToCar).unwrap();
691            let expected =
692                fields::cylindrical_wave(cw::Mode { kz, m, pol }, k, position, helicity, radial)
693                    .unwrap()
694                    .value;
695            let scale = 1e-11 * (1.0 + expected.map(Complex::norm).iter().sum::<f64>());
696            prop_assert_close!(actual, expected, scale, "cylindrical {label:?}");
697        }
698        Ok(())
699    }

701    /// Central differences of `vector_wave` along a complex `direction` in argument `a`,
702    /// paired with the cotangent `g`, and the scale of the paired terms.
703    fn paired_difference(
704        family: Family,
705        label: WaveLabel,
706        args: [Complex; 6],
707        helicity: bool,
708        g: [Complex; 3],
709        a: usize,
710        direction: Complex,
711    ) -> (Complex, f64) {
712        let h = 1e-6;
713        let step = |t: f64| {
714            let mut shifted = args;
715            shifted[a] += t * h * direction;
716            vector_wave(family, label, shifted, helicity).unwrap()
717        };
718        let (plus, minus) = (step(1.0), step(-1.0));
719        let terms: Vec<Complex> = (0..3)
720            .map(|c| g[c].conj() * (plus[c] - minus[c]) / (2.0 * h))
721            .collect();
722        (terms.iter().sum(), terms.iter().map(|t| t.norm()).sum())
723    }

725    /// Every family's pullback, paired with a complex direction in each of the six
726    /// arguments, is the holomorphic directional derivative of `vector_wave`; the
727    /// radial argument lies on either side of the small-argument threshold.
728    fn check_adjoints(
729        label: WaveLabel,
730        z: Complex,
731        [theta, phi]: [f64; 2],
732        helicity: bool,
733        g: [Complex; 3],
734    ) -> Result<(), TestCaseError> {
735        let k = Complex::new(1.3, 0.05);
736        let families = [
737            (Family::HarmonicX, arguments([theta.into(), phi.into()])),
738            (Family::HarmonicY, arguments([theta.into(), phi.into()])),
739            (Family::HarmonicZ, arguments([theta.into(), phi.into()])),
740            (
741                Family::Spherical(Radial::Regular),
742                arguments([z, theta.into(), phi.into()]),
743            ),
744            (
745                Family::Spherical(Radial::Singular),
746                arguments([z, theta.into(), phi.into()]),
747            ),
748            (
749                Family::Cylindrical(Radial::Regular),
750                arguments([0.2.into(), z, phi.into(), 0.3.into(), k]),
751            ),
752            (
753                Family::Cylindrical(Radial::Singular),
754                arguments([0.2.into(), z, phi.into(), 0.3.into(), k]),
755            ),
756            (
757                Family::Plane,
758                [
759                    Complex::new(0.3, 0.1),
760                    Complex::new(0.4, -0.05),
761                    z,
762                    0.2.into(),
763                    0.3.into(),
764                    0.4.into(),
765                ],
766            ),
767        ];
768        let direction = Complex::new(0.2, 0.1);
769        for (family, args) in families {
770            let gradient = vector_wave_pullback(family, label, args, helicity, g).unwrap();
771            for (a, gradient) in gradient.into_iter().enumerate() {
772                let (numeric, scale) =
773                    paired_difference(family, label, args, helicity, g, a, direction);
774                prop_assert_close!(
775                    gradient.conj() * direction,
776                    numeric,
777                    2e-7 * (1.0 + scale),
778                    "{family:?} argument {a}"
779                );
780            }
781        }
782        Ok(())
783    }

785    /// Regular spherical and cylindrical waves and their pullbacks agree just below
786    /// and just above `|z| = SERIES_RADIUS` (0.5), where their radial functions switch
787    /// from power series to Bessel evaluations.
788    fn check_small_argument_continuity(
789        l: i32,
790        m: i32,
791        phase: f64,
792        [theta, phi]: [f64; 2],
793        g: [Complex; 3],
794    ) -> Result<(), TestCaseError> {
795        let at = |family: Family, label: WaveLabel, helicity: bool, magnitude: f64| {
796            let z = Complex::from_polar(magnitude, phase);
797            assert_eq!(
798                z.norm() < SERIES_RADIUS,
799                magnitude < SERIES_RADIUS,
800                "{z} is on the wrong branch"
801            );
802            let args = match family {
803                Family::Cylindrical(_) => arguments([
804                    0.2.into(),
805                    z,
806                    phi.into(),
807                    0.3.into(),
808                    Complex::new(1.3, 0.05),
809                ]),
810                _ => arguments([z, theta.into(), phi.into()]),
811            };
812            (
813                vector_wave(family, label, args, helicity).unwrap(),
814                vector_wave_pullback(family, label, args, helicity, g).unwrap(),
815            )
816        };
817        for family in [
818            Family::Spherical(Radial::Regular),
819            Family::Cylindrical(Radial::Regular),
820        ] {
821            for (pol, helicity) in [(0, false), (1, false), (0, true), (1, true)] {
822                let label = WaveLabel { l, m, pol };
823                // Nine and four ulps from the threshold, respectively.
824                let (below, below_gradient) =
825                    at(family, label, helicity, SERIES_RADIUS * (1.0 - 1e-15));
826                let (above, above_gradient) =
827                    at(family, label, helicity, SERIES_RADIUS * (1.0 + 1e-15));
828                let scale = below.iter().map(|v| v.norm()).sum::<f64>();
829                prop_assert_close!(
830                    below,
831                    above,
832                    1e-12 * scale + f64::MIN_POSITIVE,
833                    "{family:?} {label:?} {helicity}"
834                );
835                let scale = below_gradient.iter().map(|v| v.norm()).sum::<f64>();
836                prop_assert_close!(
837                    below_gradient,
838                    above_gradient,
839                    1e-12 * scale + f64::MIN_POSITIVE,
840                    "{family:?} {label:?} {helicity} pullback"
841                );
842            }
843        }
844        Ok(())
845    }

847    /// Unsöld's theorem `Σ_m |X_lm|² = (2l + 1) / 4π` for the normalized vector
848    /// spherical harmonics at high degree, including both poles and a polar angle
849    /// of `1e-10`.
850    fn check_unsolds_theorem(l: i32, theta: f64, phi: f64) -> Result<(), TestCaseError> {
851        let args = arguments([theta.into(), phi.into()]);
852        for family in [Family::HarmonicX, Family::HarmonicY, Family::HarmonicZ] {
853            let total = (-l..=l)
854                .map(|m| {
855                    vector_wave(family, WaveLabel { l, m, pol: 0 }, args, false)
856                        .unwrap()
857                        .iter()
858                        .map(Complex::norm_sqr)
859                        .sum::<f64>()
860                })
861                .sum::<f64>();
862            let expected = f64::from(2 * l + 1) / (4.0 * PI);
863            prop_assert_close!(total, expected, 2e-11 * expected, "{family:?}");
864        }
865        Ok(())
866    }

894    /// At the poles the azimuthal cotangent is exactly that of `∂_φ X = i m X`, and
895    /// the polar cotangent is the one-sided derivative from inside `(0, π)`.
896    #[test]
897    fn pole_pullbacks_are_one_sided_derivatives() {
898        let g = [
899            Complex::new(0.3, 0.2),
900            Complex::new(-0.1, 0.4),
901            Complex::new(0.2, -0.3),
902        ];
903        let h = 1e-5;
904        let kr = Complex::new(1.2, 0.1);
905        for (l, m) in (1..=3).flat_map(|l| (-l..=l).map(move |m| (l, m))) {
906            for (pol, helicity) in [(0, false), (1, false), (1, true)] {
907                let label = WaveLabel { l, m, pol };
908                for (family, offset) in [
909                    (Family::HarmonicX, 0),
910                    (Family::HarmonicY, 0),
911                    (Family::HarmonicZ, 0),
912                    (Family::Spherical(Radial::Regular), 1),
913                    (Family::Spherical(Radial::Singular), 1),
914                ] {
915                    for (pole, inward) in [(0.0, 1.0), (PI, -1.0)] {
916                        let args = |theta: f64| {
917                            let mut args = arguments([kr, theta.into(), 0.3.into()]);
918                            if offset == 0 {
919                                args.rotate_left(1);
920                            }
921                            args
922                        };
923                        let at =
924                            |t: f64| vector_wave(family, label, args(pole + t), helicity).unwrap();
925                        let pole_value = at(0.0);
926                        let gradient =
927                            vector_wave_pullback(family, label, args(pole), helicity, g).unwrap();
928                        let azimuthal: Complex = pole_value
929                            .iter()
930                            .zip(g)
931                            .map(|(x, g)| (Complex::i() * f64::from(m) * x).conj() * g)
932                            .sum();
933                        let scale = 1.0 + pole_value.iter().map(|x| x.norm()).sum::<f64>();
934                        let context = format!("{family:?} {label:?} {helicity} at {pole}");
935                        assert!(
936                            (gradient[offset + 1] - azimuthal).norm() < 1e-14 * scale,
937                            "{context}: {} != {azimuthal}",
938                            gradient[offset + 1]
939                        );
940                        let (one, two) = (at(inward * h), at(2.0 * inward * h));
941                        let polar: Complex = (0..3)
942                            .map(|c| {
943                                let derivative = inward
944                                    * (-3.0 * pole_value[c] + 4.0 * one[c] - two[c])
945                                    / (2.0 * h);
946                                derivative.conj() * g[c]
947                            })
948                            .sum();
949                        assert!(
950                            (gradient[offset] - polar).norm() < 1e-7 * (1.0 + polar.norm()),
951                            "{context}: {} != {polar}",
952                            gradient[offset]
953                        );
954                    }
955                }
956            }
957        }
958    }
