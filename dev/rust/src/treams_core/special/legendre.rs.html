23/// Associated Legendre functions `F_l^m` after factoring out `sin(theta)^|m|`, visited
24/// for every `l` in `|m|..=max` in order from one run of the degree recurrence. Keeping
25/// the factor explicit avoids 0/0 and cancellation in angular functions near the poles.
26pub(crate) fn legendre_factors<const N: usize>(
27    max: i32,
28    m: i32,
29    z: Jet<N>,
30    mut visit: impl FnMut(Jet<N>),
31) {
32    let order = m.abs();
33    if max < order {
34        return;
35    }
36    let mut p = Jet::constant(1.0);
37    if m >= 0 {
38        for k in 1..=m {
39            p *= -f64::from(2 * k - 1);
40        }
41    } else {
42        for k in 1..=-m {
43            p *= 1.0 / f64::from(2 * k);
44        }
45    }
46    visit(p);
47    if max == order {
48        return;
49    }
50    let mut prev = p;
51    p *= (f64::from(2 * order + 1) / f64::from(order - m + 1)) * z;
52    visit(p);
53    for k in (order + 2)..=max {
54        let next =
55            (f64::from(2 * k - 1) * z * p - f64::from(k + m - 1) * prev) * (1.0 / f64::from(k - m));
56        prev = p;
57        p = next;
58        visit(p);
59    }
60}

62/// `F_l^m` and `F_(l-1)^m` of [`legendre_factors`], with `F_(l-1)^m = 0` for `l = |m|`
63/// and both zero for `|m| > l`.
64fn legendre_factor_pair<const N: usize>(l: i32, m: i32, z: Jet<N>) -> [Jet<N>; 2] {
65    let mut pair = [Jet::default(); 2];
66    legendre_factors(l, m, z, |p| pair = [p, pair[0]]);
67    pair
68}

70/// Associated Legendre polynomial `F_l^m` after factoring out `sin(theta)^|m|`.
71pub(crate) fn legendre_factor<const N: usize>(l: i32, m: i32, z: Jet<N>) -> Jet<N> {
72    let [value, _] = legendre_factor_pair(l, m, z);
73    value
74}

76/// `tau_l^m / sin(theta)^(|m|-1)` for `m != 0` from one Legendre recurrence, by DLMF
77/// 14.10.5: `sin(theta) tau = l cos(theta) P_l^m - (l + m) P_(l-1)^m`.
78fn tau_factor<const N: usize>(
79    l: i32,
80    m: i32,
81    [upper, lower]: [Jet<N>; 2],
82    cosine: Jet<N>,
83) -> Jet<N> {
84    f64::from(l) * cosine * upper - f64::from(l + m) * lower
85}

87/// The angular functions `[pi, tau]` from the polar cosine and sine, with the sine
88/// factor kept explicit at the poles.
89fn angular_jets<const N: usize>(l: i32, m: i32, cosine: Jet<N>, sine: Jet<N>) -> [Jet<N>; 2] {
90    if m == 0 {
91        // tau_l^0 = P_l^1.
92        return [Jet::default(), sine * legendre_factor(l, 1, cosine)];
93    }
94    let pair = legendre_factor_pair(l, m, cosine);
95    let power = sine.powi(m.abs() - 1);
96    [
97        f64::from(m) * power * pair[0],
98        power * tau_factor(l, m, pair, cosine),
99    ]
100}

102/// Angular factor coupling the vector spherical wave `(l, m)` of polarization
103/// `mode_pol` to a plane wave of polarization `pol` whose direction has the polar
104/// cosine and sine given: `tau -+ pi` for negative (0) or positive (1) helicity; in
105/// parity polarization `tau` for the mode's own polarization and `pi` for the other.
106/// The polarization module states this rule for the vector waves themselves.
107pub(crate) fn polarized_angular<const N: usize>(
108    l: i32,
109    m: i32,
110    [cosine, sine]: [Jet<N>; 2],
111    [mode_pol, pol]: [u8; 2],
112    helicity: bool,
113) -> Jet<N> {
114    let [pi, tau] = angular_jets(l, m, cosine, sine);
115    if helicity {
116        tau + helicity_sign(pol) * pi
117    } else if mode_pol == pol {
118        tau
119    } else {
120        pi
121    }
122}

124/// Associated Legendre polynomial or one of the two vector-wave angular functions.
125///
126/// Upstream: `treams.special.lpmv`, `pi_fun` and `tau_fun`.
127#[derive(Clone, Copy, Debug)]
128pub enum Angular {
129    /// Associated Legendre P, including its Condon-Shortley phase.
130    Legendre,
131    /// m P / sqrt(1-z^2), with analytic polar limits.
132    Pi,
133    /// Polar-angle derivative of P(cos(theta)).
134    Tau,
135}

137/// `sin(theta)^power = (1 - z^2)^(power / 2)` at the cosine `z`.
138fn sine_power<const N: usize>(z: Jet<N>, power: i32) -> Jet<N> {
139    let w = 1.0 - z * z;
140    if power % 2 == 0 {
141        w.powi(power / 2)
142    } else if power > 1 && w.value == Complex::default() {
143        // Odd powers >=3 have zero value and first derivative at the poles.
144        Jet::default()
145    } else {
146        w.powi(power / 2) * w.sqrt()
147    }
148}
149
150/// `F_l^m(z) sin(theta)^power`: the Legendre function for `power = |m|`, and `pi / m`
151/// for `power = |m| - 1`.
152fn angular_legendre<const N: usize>(l: i32, m: i32, z: Jet<N>, power: i32) -> Jet<N> {
153    if m.abs() > l {
154        return Jet::default();
155    }
156    legendre_factor(l, m, z) * sine_power(z, power)
157}

159/// The angular function `kind` of degree `l` and order `m` at `z`, with its derivative
160/// for `N = 1`, after checking the labels and the argument.
161#[allow(clippy::cast_possible_truncation)] // Integer labels are checked and bounded before conversion.
162fn angular_jet<const N: usize>(l: f64, m: f64, z: Complex, kind: Angular) -> Result<Jet<N>> {
163    if !finite(z)
164        || !l.is_finite()
165        || !m.is_finite()
166        || m.fract() != 0.0
167        || !(0.0..=f64::from(MAX_DEGREE)).contains(&l)
168    {
169        return Err(Error::InvalidInput(
170            // 128 is MAX_DEGREE.
171            "angular functions require 0 <= degree <= 128, integer order and finite argument"
172                .into(),
173        ));
174    }
175    if l.fract() != 0.0 && (!matches!(kind, Angular::Legendre) || z.im != 0.0) {
176        return Err(Error::InvalidInput(
177            "noninteger degrees require real Legendre arguments".into(),
178        ));
179    }
180    if m.abs() > l {
181        return Ok(Jet::default());
182    }
183    if l.fract() != 0.0 {
184        let (value, derivative) = if N == 0 {
185            super::ferrers::ferrers_real_degree::<false>(l, m as i32, z.re)?
186        } else {
187            super::ferrers::ferrers_real_degree::<true>(l, m as i32, z.re)?
188        };
189        return Ok(Jet::<N>::variable(z, 0).chain(value.into(), derivative.into()));
190    }
191    let (l, m) = (l as i32, m as i32);
192    let z = Jet::<N>::variable(z, 0);
193    let value = match kind {
194        Angular::Legendre => angular_legendre(l, m, z, m.abs()),
195        Angular::Pi if m == 0 => Jet::default(),
196        Angular::Pi => f64::from(m) * angular_legendre(l, m, z, m.abs() - 1),
197        Angular::Tau if m == 0 => angular_legendre(l, 1, z, 1),
198        Angular::Tau => {
199            tau_factor(l, m, legendre_factor_pair(l, m, z), z) * sine_power(z, m.abs() - 1)
200        }
201    };
202    if !value.finite() {
203        return Err(Error::SpecialFunction(
204            if N == 0 {
205                "non-finite angular result"
206            } else {
207                "angular argument derivative is undefined or non-finite"
208            }
209            .into(),
210        ));
211    }
212    Ok(value)
213}

215/// Angular function at a cosine argument; real Legendre values also accept real degrees.
216///
217/// Upstream: `treams.special.lpmv` (which takes `(order, degree, x)`), `pi_fun` and
218/// `tau_fun`. Differences: degrees outside `[0, 128]`, non-integer orders, and
219/// non-integer degrees with `pi`, `tau` or a complex argument give an error.
220pub fn angular_value(l: f64, m: f64, z: Complex, kind: Angular) -> Result<Complex> {
221    Ok(angular_jet::<0>(l, m, z, kind)?.value)
222}

224/// Associated Legendre function `P_degree^order(x)` of a real argument, as a real value.
225///
226/// Outside `[-1, 1]` the odd orders with `|order| <= degree` have imaginary values: this
227/// function rejects them, and [`angular_value`] returns them for a complex argument.
228/// Orders with `|order| > degree` give zero.
229///
230/// Upstream: `treams.special.lpmv` with real arguments, which takes `(order, degree, x)`.
231/// Differences: for an integer degree, `x` outside `[-1, 1]` and `0 < |order| <= degree`,
232/// treams returns NaN, while this function returns the real value for even orders and an
233/// error for odd orders. Degrees outside `[0, 128]`, non-integer orders, non-finite `x`
234/// and non-integer degrees with `x` outside `(-1, 1]` give an error, where treams returns
235/// a value or NaN.
236#[inline]
237pub fn lpmv_real(degree: f64, order: f64, x: f64) -> Result<f64> {
238    let value = angular_value(degree, order, x.into(), Angular::Legendre)?;
239    if value.im == 0.0 {
240        Ok(value.re)
241    } else {
242        Err(Error::InvalidInput(
243            "real lpmv requires -1 <= x <= 1 for odd orders |order| <= degree; \
244             pass complex z for the continuation"
245                .into(),
246        ))
247    }
248}

250/// Borrowed angular values with scalar-or-equal-length broadcasting.
251pub(crate) fn angular_values(
252    degrees: &[f64],
253    orders: &[f64],
254    arguments: &[Complex],
255    kind: Angular,
256) -> Result<Vec<Complex>> {
257    let size = broadcast::size(
258        &[degrees.len(), orders.len(), arguments.len()],
259        "angular arrays must have equal lengths or scalar inputs",
260    )?;
261    broadcast::map(size, PARALLEL, |i| {
262        angular_value(
263            element(degrees, i),
264            element(orders, i),
265            element(arguments, i),
266            kind,
267        )
268    })
269}

271/// What [`angular_array`] saves for its pullback: the degrees, orders, arguments and
272/// function. The pullback recomputes the local derivatives.
273#[derive(Debug)]
274pub struct AngularResidual {
275    degrees: Vec<f64>,
276    orders: Vec<f64>,
277    arguments: Vec<Complex>,
278    kind: Angular,
279    size: usize,
280}
281
282/// Broadcast angular values and a complex-argument residual; labels stay fixed.
283///
284/// Upstream: the functions of [`angular_value`] over arrays.
285pub fn angular_array(
286    degrees: Vec<f64>,
287    orders: Vec<f64>,
288    arguments: Vec<Complex>,
289    kind: Angular,
290) -> Result<(Vec<Complex>, AngularResidual)> {
291    let values = angular_values(&degrees, &orders, &arguments, kind)?;
292    let size = values.len();
293    Ok((
294        values,
295        AngularResidual {
296            degrees,
297            orders,
298            arguments,
299            kind,
300            size,
301        },
302    ))
303}
304
305impl AngularResidual {
306    /// The gradient with respect to the arguments, given the gradient `cotangent` with
307    /// respect to the values. A zero cotangent skips derivatives that are singular at
308    /// the poles.
309    pub fn pullback(self, cotangent: &[Complex]) -> Result<Vec<Complex>> {
310        let [gradient] = broadcast::pullback(
311            cotangent,
312            self.size,
313            "angular cotangent must be finite and match output",
314            [self.arguments.len()],
315            PARALLEL,
316            |i, &g| {
317                if g == Complex::default() {
318                    return Ok([g]);
319                }
320                let jet = angular_jet::<1>(
321                    element(&self.degrees, i),
322                    element(&self.orders, i),
323                    element(&self.arguments, i),
324                    self.kind,
325                )?;
326                Ok([g * jet.derivative.first().copied().unwrap_or_default().conj()])
327            },
328        )?;
329        Ok(gradient)
330    }
331}

333/// `[cos(theta), sin(theta)]` with their derivatives. The sine takes the principal
334/// branch (nonnegative real part) and is exactly zero at the poles, where `theta` is a
335/// floating-point multiple of `PI`.
336#[allow(clippy::float_cmp)] // Exact coordinate pole labels.
337pub(crate) fn polar_trig<const N: usize>(theta: Jet<N>) -> [Jet<N>; 2] {
338    let cosine = theta.chain(theta.value.cos(), -theta.value.sin());
339    let mut sine = theta.chain(theta.value.sin(), theta.value.cos());
340    // Associated Legendre functions use the principal sine factor. Computing
341    // its magnitude from sin(theta) preserves small angles lost in 1-cos²(theta).
342    if sine.value.re < 0.0 {
343        sine = -sine;
344    }
345    // Multipole translations amplify sin(PI)'s rounding residue. An exact
346    // floating-point multiple of the PI constant denotes the coordinate pole.
347    if theta.value.im == 0.0 && theta.value.re % PI == 0.0 {
348        sine.value = Complex::default();
349    }
350    [cosine, sine]
