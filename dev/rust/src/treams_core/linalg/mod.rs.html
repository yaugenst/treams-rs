1//! Dense linear solves and general complex eigensystems with analytic derivatives, and
2//! restarted GMRES. treams-rs extension.
3//!
4//! Two libraries share the linear algebra, by one rule:
5//!
6//! - nalgebra's `DMatrix` stores every dense matrix.
7//! - faer factors and multiplies dense complex matrices of any size, through this
8//!   module: LU solves ([`solve`], `Lu`), singular values ([`svdvals`]), eigensystems
9//!   ([`eig`]) and products (`product` and its variants). `view` and `view_mut` lend a
10//!   `DMatrix` to faer without a copy.
11//! - Real symmetric eigenproblems use nalgebra's `SymmetricEigen`: `special::wigner`
12//!   diagonalizes the tridiagonal generator of the small-d matrix, of size 2l + 1.
13//! - Fixed-size matrices of at most 4 x 4 stay in nalgebra: the 3 x 3 inverse of
14//!   `lattice::reciprocal`, the closed-form 2 x 2 inverse of `coeffs::mie`, the 4 x 4
15//!   and 2 x 2 inverses of `coeffs::mie_cyl` and the 4 x 4 inverse of
16//!   `smatrix::interface`.
17#![allow(clippy::indexing_slicing)] // Validated matrix dimensions and eigenvector pivots.

58/// `out = left right` on the treams-rs pool ([`crate::threads::product`]). Pass
59/// `.adjoint()` views for conjugate-transposed factors; faer conjugates them inside
60/// the product kernel.
61pub(crate) fn product_into<L, R>(
62    out: MatMut<'_, Complex>,
63    left: MatRef<'_, L>,
64    right: MatRef<'_, R>,
65) where
66    L: Conjugate<Canonical = Complex>,
67    R: Conjugate<Canonical = Complex>,
68{
69    let (m, n, k) = (out.nrows(), out.ncols(), left.ncols());
70    crate::threads::product(m, n, k, |par| {
71        matmul(
72            out,
73            Accum::Replace,
74            left,
75            right,
76            Complex::new(1.0, 0.0),
77            par,
78        );
79    });
80}

82/// `left right` in a new matrix; see [`product_into`].
83pub(crate) fn product_views<L, R>(left: MatRef<'_, L>, right: MatRef<'_, R>) -> DMatrix<Complex>
84where
85    L: Conjugate<Canonical = Complex>,
86    R: Conjugate<Canonical = Complex>,
87{
88    let mut result = DMatrix::zeros(left.nrows(), right.ncols());
89    product_into(view_mut(&mut result), left, right);
90    result
91}

115/// Positive diagonal scales for the equivalent operator `R A C`.
116///
117/// A normal solve scales its right-hand side by `row` and its solution by
118/// `column`. A conjugate-transpose solve uses the opposite order. These scales
119/// are numerical coordinates, not additional inputs to the implicit derivative.
120#[derive(Clone, Debug)]
121pub(crate) struct Equilibration {
122    /// Left (equation) scale.
123    pub(crate) row: Vec<f64>,
124    /// Right (unknown) scale.
125    pub(crate) column: Vec<f64>,
126}

128/// Equilibrate severely unbalanced finite square operators in place.
129///
130/// Small-argument, high-order multipole operators can have unit diagonal and
131/// off-diagonal entries exceeding `1e20`. Their unscaled LU loses accuracy even
132/// when the equivalent equilibrated system is well conditioned. The scan that
133/// rejects non-finite entries also measures the spread of entry sizes. An operator
134/// whose nonzero entries, or whose diagonal entries, all lie within a factor of 10 of
135/// the largest entry stays unchanged.
136pub(crate) fn equilibrate(operator: &mut DMatrix<Complex>) -> Result<Option<Equilibration>> {
137    let n = operator.nrows();
138    if n == 0 || !operator.is_square() {
139        return Err(Error::InvalidInput(
140            "require a finite nonempty square operator".into(),
141        ));
142    }
143    // LAPACK xLAQGE uses 0.1 for row/column scale ratios. Waiting until
144    // sqrt(epsilon) loses observable digits in moderately unbalanced multipole
145    // systems even though a change of coordinates makes them well conditioned.
146    let relative_floor = 0.1;
147    let mut minimum = f64::INFINITY;
148    let mut maximum = 0.0_f64;
149    for &z in operator.iter() {
150        if !finite(z) {
151            return Err(Error::InvalidInput(
152                "require a finite nonempty square operator".into(),
153            ));
154        }
155        let magnitude = z.re.abs().max(z.im.abs());
156        if magnitude > 0.0 {
157            minimum = minimum.min(magnitude);
158        }
159        maximum = maximum.max(magnitude);
160    }
161    if maximum > 0.0
162        && (minimum / maximum >= relative_floor
163            || (0..n).all(|i| {
164                let z = operator[(i, i)];
165                z.re.abs().max(z.im.abs()) / maximum >= relative_floor
166            }))
167    {
168        return Ok(None);
169    }
170    // Only potentially unbalanced operators allocate scratch or need another
171    // pass. Isolated tiny entries do not by themselves require equilibration.
172    let mut row = vec![0.0_f64; n];
173    let mut column_min = f64::INFINITY;
174    for values in operator.as_slice().chunks_exact(n) {
175        let mut column_max = 0.0_f64;
176        for (row_max, &z) in row.iter_mut().zip(values) {
177            let magnitude = z.re.abs().max(z.im.abs());
178            *row_max = row_max.max(magnitude);
179            column_max = column_max.max(magnitude);
180        }
181        column_min = column_min.min(column_max);
182    }
183    let row_min = row.iter().copied().fold(f64::INFINITY, f64::min);
184    if row_min == 0.0 || column_min == 0.0 {
185        return Err(Error::Singular);
186    }
187    if row_min / maximum >= relative_floor && column_min / maximum >= relative_floor {
188        return Ok(None);
189    }
190    if column_min < row_min {
191        // Scale the more disparate axis first. Otherwise normalizing large rows
192        // can underflow an entire small column before it gets its own scale.
193        // Transposition swaps the two measured spreads, so this recurses once.
194        operator.transpose_mut();
195        let result = equilibrate(operator);
196        operator.transpose_mut();
197        return result.map(|scales| {
198            scales.map(|scales| Equilibration {
199                row: scales.column,
200                column: scales.row,
201            })
202        });
203    }
204    let inverse = |value: f64| {
205        value
206            .clamp(f64::MIN_POSITIVE, f64::MIN_POSITIVE.recip())
207            .recip()
208    };
209    for value in &mut row {
210        *value = inverse(*value);
211    }
212    let mut column = Vec::with_capacity(n);
213    for values in operator.as_mut_slice().chunks_exact_mut(n) {
214        let mut column_max = 0.0_f64;
215        for (value, &scale) in values.iter_mut().zip(&row) {
216            *value *= scale;
217            column_max = column_max.max(value.re.abs().max(value.im.abs()));
218        }
219        let scale = inverse(column_max);
220        column.push(scale);
221        for value in values {
222            *value *= scale;
223        }
224    }
225    Ok(Some(Equilibration { row, column }))
226}
227
228fn scale_rows(mut matrix: MatMut<'_, Complex>, scales: &[f64]) {
229    for j in 0..matrix.ncols() {
230        for (value, &scale) in matrix.as_mut().col_mut(j).iter_mut().zip(scales) {
231            *value *= scale;
232        }
233    }
234}

236/// The faer workers of an LU factorization or solve with `rows` rows and `columns`
237/// right-hand sides, on a pool of `budget` threads.
238///
239/// Solves with fewer than 32 right-hand sides run on one worker. Recursive LU and
240/// triangular solves split the work into panels, one task each. A pool of up to four
241/// threads runs whole: the panels stay wide enough for four workers. Larger pools get one worker per 512 rows up to four, one per 2048 rows
242/// beyond that, and at most one per 16 right-hand sides, because narrower panels cost
243/// more in task overhead than they gain. The constants are tuned up to 8192 rows;
244/// `benches/lu_scheduling.rs` times a factorization and its solves for any worker
245/// count, to check them on other hardware.
246fn lu_threads(rows: usize, columns: usize, budget: usize) -> usize {
247    // Narrow solves: faer's parallel triangular solves and products split the inner
248    // dimension by worker count, which changes the last bits of the solution, and
249    // their O(rows²) work per column gains little from workers.
250    if columns < 32 {
251        return 1;
252    }
253    if budget <= 4 {
254        return budget;
255    }
256    (rows / 512)
257        .min(4)
258        .max(rows / 2048)
259        .min(columns / 16)
260        .clamp(1, budget)
261}
262
263/// The faer workers of an LU call on the treams-rs budget.
264fn lu_workers(rows: usize, columns: usize) -> usize {
265    // Faer takes no parallel branch here (LU and row swaps below 64 rows, products
266    // below its M·N·K threshold, at most 64 right-hand sides in the triangular
267    // solves), so the result is the same without the hand-off to the pool.
268    if rows < 64 && columns <= 64 {
269        return 1;
270    }
271    lu_threads(rows, columns, crate::threads::current_num_threads())
272}

274/// A faer workspace of `req`, or [`Error::OutOfMemory`] when the system refuses it or
275/// the requirement overflowed.
276fn scratch(req: StackReq) -> Result<MemBuffer> {
277    MemBuffer::try_new(req).map_err(|_| {
278        // dyn-stack marks an overflowed requirement with alignment 0 and size 0; it
279        // asked for more than usize::MAX bytes.
280        if req.align_bytes() == 0 {
281            Error::out_of_memory(usize::MAX, 2)
282        } else {
283            Error::out_of_memory(req.size_bytes(), 1)
284        }
285    })
286}

288/// Packed pivoted LU: overwrite the operator and share its triangular storage.
289#[derive(Clone, Debug)]
290pub(crate) struct Lu {
291    factors: DMatrix<Complex>,
292    permutation: Perm<usize>,
293    equilibration: Option<Equilibration>,
294}
295
296impl Lu {
297    pub(crate) fn new(mut factors: DMatrix<Complex>) -> Result<Self> {
298        let n = factors.nrows();
299        let equilibration = equilibrate(&mut factors)?;
300        let mut forward = numerics::filled(n, 0_usize)?;
301        let mut inverse = numerics::filled(n, 0_usize)?;
302        crate::threads::dense(lu_workers(n, n), |par| -> Result<()> {
303            factor::lu_in_place(
304                view_mut(&mut factors),
305                &mut forward,
306                &mut inverse,
307                par,
308                MemStack::new(&mut scratch(
309                    factor::lu_in_place_scratch::<usize, Complex>(n, n, par, Spec::default()),
310                )?),
311                Spec::default(),
312            );
313            Ok(())
314        })?;
315        if factors.diagonal().iter().any(|&z| z == Complex::default()) {
316            return Err(Error::Singular);
317        }
318        Ok(Self {
319            factors,
320            equilibration,
321            permutation: Perm::new_checked(
322                forward.into_boxed_slice(),
323                inverse.into_boxed_slice(),
324                n,
325            ),
326        })
327    }
328
329    /// Solve `A X = B` in place, with `A` the operator given to [`Lu::new`].
330    pub(crate) fn solve_in_place(&self, rhs: MatMut<'_, Complex>) -> Result<()> {
331        self.solve_with(rhs, false)
332    }
333
334    /// Solve `Aᴴ X = B` in place.
335    pub(crate) fn solve_adjoint_in_place(&self, rhs: MatMut<'_, Complex>) -> Result<()> {
336        self.solve_with(rhs, true)
337    }
338
339    // The factors hold `R A C`. A forward solve scales by the row scales, solves
340    // and scales by the column scales; an adjoint solve uses the reverse order.
341    fn solve_with(&self, mut rhs: MatMut<'_, Complex>, adjoint: bool) -> Result<()> {
342        let (n, columns) = (self.factors.nrows(), rhs.ncols());
343        let scales = self.equilibration.as_ref().map(|scales| {
344            if adjoint {
345                (&scales.column, &scales.row)
346            } else {
347                (&scales.row, &scales.column)
348            }
349        });
350        if let Some((before, _)) = scales {
351            scale_rows(rhs.as_mut(), before);
352        }
353        let (factors, permutation) = (view(&self.factors), self.permutation.as_ref());
354        crate::threads::dense(lu_workers(n, columns), |par| -> Result<()> {
355            if adjoint {
356                lu_solve::solve_transpose_in_place_with_conj(
357                    factors,
358                    factors,
359                    permutation,
360                    Conj::Yes,
361                    rhs.as_mut(),
362                    par,
363                    MemStack::new(&mut scratch(lu_solve::solve_transpose_in_place_scratch::<
364                        usize,
365                        Complex,
366                    >(n, columns, par))?),
367                );
368            } else {
369                lu_solve::solve_in_place(
370                    factors,
371                    factors,
372                    permutation,
373                    rhs.as_mut(),
374                    par,
375                    MemStack::new(&mut scratch(lu_solve::solve_in_place_scratch::<
376                        usize,
377                        Complex,
378                    >(n, columns, par))?),
379                );
380            }
381            Ok(())
382        })?;
383        if let Some((_, after)) = scales {
384            scale_rows(rhs, after);
385        }
386        Ok(())
387    }
388}

400    solve_owned(operator.clone(), rhs)
401}

403/// Solve while reusing the owned operator buffer for its LU factors.
404pub fn solve_owned(operator: DMatrix<Complex>, mut rhs: DMatrix<Complex>) -> Result<SolveResidual> {
405    let n = operator.nrows();
406    if n == 0
407        || !operator.is_square()
408        || rhs.nrows() != n
409        || rhs.ncols() == 0
410        || rhs.iter().any(|&z| !finite(z))
411    {
412        return Err(Error::InvalidInput(
413            "require a finite nonempty square operator and matching right-hand sides".into(),
414        ));
415    }
416    let lu = Lu::new(operator)?;
417    lu.solve_in_place(view_mut(&mut rhs))?;
418    if rhs.iter().any(|&z| !finite(z)) {
419        return Err(Error::Singular);
420    }
421    Ok(SolveResidual { lu, value: rhs })
422}

424/// Eigen- and singular-value decompositions run on the calling thread. faer's
425/// parallel Hessenberg and bidiagonal reductions split inner products by worker
426/// count, so their last bits, and the order of eigenvalues, would follow the
427/// budget. On four workers they gained at most 1.3× (eigenvalues) and 1.7×
428/// (singular values) at n = 768, and nothing below n = 384.
429const DECOMPOSITION: faer::Par = faer::Par::Seq;
430
431/// Thin singular vectors and values. faer's `Svd` reads faer's global parallelism,
432/// so the decomposition is called with [`DECOMPOSITION`] instead.
433#[derive(Debug)]
434struct ThinSvd {
435    u: Mat<Complex>,
436    s: Diag<Complex>,
437    v: Mat<Complex>,
438}
439
440/// Reserve decomposition vectors before initializing their entries.
441fn faer_zeros(rows: usize, columns: usize) -> Result<Mat<Complex>> {
442    let mut matrix = Mat::new();
443    matrix
444        .try_reserve(rows, columns)
445        .map_err(|_| Error::out_of_memory(rows.saturating_mul(columns), size_of::<Complex>()))?;
446    matrix.resize_with(rows, columns, |_, _| Complex::default());
447    Ok(matrix)
448}
449
450fn thin_svd(a: MatRef<'_, Complex>) -> Result<ThinSvd> {
451    let (m, n) = a.shape();
452    let size = m.min(n);
453    let mut factors = ThinSvd {
454        u: faer_zeros(m, size)?,
455        s: Diag::zeros(size),
456        v: faer_zeros(n, size)?,
457    };
458    let thin = svd::ComputeSvdVectors::Thin;
459    let par = DECOMPOSITION;
460    svd::svd(
461        a,
462        factors.s.as_mut(),
463        Some(factors.u.as_mut()),
464        Some(factors.v.as_mut()),
465        par,
466        MemStack::new(&mut scratch(svd::svd_scratch::<Complex>(
467            m,
468            n,
469            thin,
470            thin,
471            par,
472            Spec::default(),
473        ))?),
474        Spec::default(),
475    )
476    .map_err(|err| Error::NotConverged(format!("singular-value decomposition: {err:?}")))?;
477    Ok(factors)
478}

480/// Eigenvalues and right eigenvectors, called with an explicit `Par` as [`thin_svd`].
481fn eigen(a: MatRef<'_, Complex>) -> Result<(Diag<Complex>, Mat<Complex>)> {
482    let n = a.nrows();
483    let mut values = Diag::zeros(n);
484    let mut vectors = faer_zeros(n, n)?;
485    let par = DECOMPOSITION;
486    evd::evd_cplx(
487        a,
488        values.as_mut(),
489        None,
490        Some(vectors.as_mut()),
491        par,
492        MemStack::new(&mut scratch(evd::evd_scratch::<Complex>(
493            n,
494            evd::ComputeEigenvectors::No,
495            evd::ComputeEigenvectors::Yes,
496            par,
497            Spec::default(),
498        ))?),
499        Spec::default(),
500    )
501    .map_err(|err| Error::NotConverged(format!("eigendecomposition: {err:?}")))?;
502    Ok((values, vectors))
503}

505/// What [`svdvals`] saves for its derivatives: the thin singular vectors and the singular
506/// values.
507#[derive(Debug)]
508pub struct SvdvalsResidual {
509    u: Mat<Complex>,
510    v: Mat<Complex>,
511    values: Vec<f64>,
512}
513
514/// Singular values of a finite nonempty rectangular complex matrix.
515pub fn svdvals(operator: &DMatrix<Complex>) -> Result<SvdvalsResidual> {
516    if operator.is_empty() || operator.iter().any(|&z| !finite(z)) {
517        return Err(Error::InvalidInput(
518            "SVD requires a finite nonempty matrix".into(),
519        ));
520    }
521    let scale = operator.iter().map(|z| z.norm()).fold(0.0, f64::max);
522    let scale = if scale == 0.0 { 1.0 } else { scale };
523    let decomposition = thin_svd(view(&operator.map(|z| z / scale)))?;
524    let values: Vec<_> = (0..operator.nrows().min(operator.ncols()))
525        .map(|i| decomposition.s.as_ref()[i].re * scale)
526        .collect();
527    if values.iter().any(|x| !x.is_finite()) {
528        return Err(Error::NonFinite("non-finite singular values".into()));
529    }
530    Ok(SvdvalsResidual {
531        u: decomposition.u,
532        v: decomposition.v,
533        values,
534    })
535}

537impl SvdvalsResidual {
538    /// The singular values, in descending order, which the pullback reads.
539    #[must_use]
540    pub fn values(&self) -> &[f64] {
541        &self.values
542    }
543
544    /// The shape of the input operator.
545    #[must_use]
546    pub fn shape(&self) -> (usize, usize) {
547        (self.u.nrows(), self.v.nrows())
548    }
549
550    /// Singular-value tangents `Re(uᴴ dA v)` for simple positive singular values.
551    ///
552    /// The ordered individual singular values have no linear derivative at repeated
553    /// or zero values. Unlike the pullback of a smooth spectral sum, a pushforward
554    /// returns each individual derivative and therefore rejects these cases.
555    /// Values within `64 ε` of the largest value's scale are numerically unresolved
556    /// and are treated as zero, including roundoff from rank-deficient matrices.
557    pub fn pushforward(&self, operator: &DMatrix<Complex>) -> Result<Vec<f64>> {
558        if operator.shape() != self.shape() || operator.iter().any(|&z| !finite(z)) {
559            return Err(Error::InvalidInput(
560                "invalid singular-value operator tangent".into(),
561            ));
562        }
563        let tolerance = 64.0 * f64::EPSILON * self.values[0];
564        if self.values.iter().any(|&value| value <= tolerance) {
565            return Err(Error::InvalidInput(
566                "singular values at zero or below numerical resolution have no pushforward".into(),
567            ));
568        }
569        if self
570            .values
571            .windows(2)
572            .any(|pair| pair[0] - pair[1] <= tolerance)
573        {
574            return Err(Error::InvalidInput(
575                "individual repeated singular values have no pushforward".into(),
576            ));
577        }
578        let moved = product_views(view(operator), self.v.as_ref());
579        let u = self.u.as_ref();
580        Ok((0..self.values.len())
581            .map(|j| {
582                (0..u.nrows())
583                    .map(|i| (u[(i, j)].conj() * moved[(i, j)]).re)
584                    .sum()
585            })
586            .collect())
587    }
588
589    /// Equal weights are supported at repeated positive values; numerically zero
590    /// values require numerically zero weights because the singular value itself
591    /// is not smooth there. Both tests use relative tolerance `64 ε`.
592    pub fn pullback(&self, cotangent: &[f64]) -> Result<DMatrix<Complex>> {
593        if cotangent.len() != self.values.len() || cotangent.iter().any(|x| !x.is_finite()) {
594            return Err(Error::InvalidInput(
595                "invalid singular-value cotangent".into(),
596            ));
597        }
598        let tolerance = 64.0 * f64::EPSILON * self.values[0];
599        let weight_tolerance =
600            64.0 * f64::EPSILON * cotangent.iter().map(|x| x.abs()).fold(0.0, f64::max);
601        for i in 0..cotangent.len() {
602            if self.values[i] <= tolerance && cotangent[i].abs() > weight_tolerance {
603                return Err(Error::InvalidInput(
604                    "nonzero singular-value weight at zero or below numerical resolution is not supported".into(),
605                ));
606            }
607            if i > 0
608                && self.values[i - 1] > tolerance
609                && (self.values[i - 1] - self.values[i]).abs() <= tolerance
610                && (cotangent[i - 1] - cotangent[i]).abs()
611                    > 64.0 * f64::EPSILON * cotangent[i - 1].abs().max(cotangent[i].abs())
612            {
613                return Err(Error::InvalidInput(
614                    "repeated singular values require equal cotangent weights".into(),
615                ));
616            }
617        }
618        let u = self.u.as_ref();
619        let weighted =
620            DMatrix::from_fn(u.nrows(), cotangent.len(), |i, j| u[(i, j)] * cotangent[j]);
621        Ok(product_views(view(&weighted), self.v.as_ref().adjoint()))
622    }
623}

625/// Cotangents of the inputs of [`solve`] and [`solve_owned`], in the order of their
626/// arguments.
627#[derive(Clone, Debug)]
628pub struct SolveGradient {
629    /// Cotangent of the operator `A`.
630    pub operator: DMatrix<Complex>,
631    /// Cotangent of the right-hand sides `B`.
632    pub rhs: DMatrix<Complex>,
633}
634
635impl SolveResidual {
636    /// The solution `X` of `A X = B`, one column per right-hand side, which the pullback
637    /// reads.
638    #[must_use]
639    pub const fn value(&self) -> &DMatrix<Complex> {
640        &self.value
641    }
642
643    /// The shape of the solution: unknowns and right-hand sides.
644    #[must_use]
645    pub fn shape(&self) -> (usize, usize) {
646        self.value.shape()
647    }
648
649    /// Solve `A dX = dB - dA X`, reusing the forward LU and the tangent RHS buffer.
650    pub fn pushforward(
651        &self,
652        operator: &DMatrix<Complex>,
653        mut rhs: DMatrix<Complex>,
654    ) -> Result<DMatrix<Complex>> {
655        let n = self.value.nrows();
656        if operator.shape() != (n, n)
657            || rhs.shape() != self.shape()
658            || operator.iter().chain(rhs.iter()).any(|&z| !finite(z))
659        {
660            return Err(Error::InvalidInput("invalid linear-solve tangent".into()));
661        }
662        rhs -= product(operator, &self.value);
663        self.lu.solve_in_place(view_mut(&mut rhs))?;
664        if rhs.iter().any(|&z| !finite(z)) {
665            return Err(Error::Singular);
666        }
667        Ok(rhs)
668    }
669
670    pub(crate) fn adjoint_rhs(&self, mut cotangent: DMatrix<Complex>) -> Result<DMatrix<Complex>> {
671        if cotangent.shape() != self.shape() || cotangent.iter().any(|&z| !finite(z)) {
672            return Err(Error::InvalidInput("invalid linear-solve cotangent".into()));
673        }
674        self.lu.solve_adjoint_in_place(view_mut(&mut cotangent))?;
675        Ok(cotangent)
676    }
677
678    /// Return operator and right-hand-side cotangents, reusing the forward LU.
679    pub fn pullback(&self, cotangent: DMatrix<Complex>) -> Result<SolveGradient> {
680        let rhs = self.adjoint_rhs(cotangent)?;
681        Ok(SolveGradient {
682            operator: -product_adjoint_right(&rhs, &self.value),
683            rhs,
684        })
685    }
686}

688/// What [`eig`] saves for its derivatives: the right eigensystem, with unit vectors whose
689/// largest component is real and positive.
690#[derive(Debug)]
691pub struct EigResidual {
692    values: Vec<Complex>,
693    vectors: DMatrix<Complex>,
694    pivots: Vec<usize>,
695    scale: f64,
696    /// Factor the eigenvectors only when a derivative needs their inverse;
697    /// every later direction and cotangent shares the same factors.
698    vectors_lu: OnceLock<Result<Lu>>,
699}
700
701/// General complex eigendecomposition. Repeated eigenvalues are allowed in the primal.
702pub fn eig(operator: &DMatrix<Complex>) -> Result<EigResidual> {
703    let n = operator.nrows();
704    if n == 0 || !operator.is_square() || operator.iter().any(|&z| !finite(z)) {
705        return Err(Error::InvalidInput(
706            "eigensystem requires a finite nonempty square matrix".into(),
707        ));
708    }
709    let scale = operator.iter().map(|z| z.norm()).fold(0.0, f64::max);
710    let scale = if scale == 0.0 { 1.0 } else { scale };
711    let (eigenvalues, eigenvectors) = eigen(view(&operator.map(|z| z / scale)))?;
712    let values: Vec<_> = (0..n).map(|i| eigenvalues.as_ref()[i] * scale).collect();
713    let mut vectors = DMatrix::from_fn(n, n, |i, j| eigenvectors[(i, j)]);
714    let mut pivots = Vec::with_capacity(n);
715    for mut column in vectors.column_iter_mut() {
716        let pivot = (0..n)
717            .max_by(|&i, &j| column[i].norm_sqr().total_cmp(&column[j].norm_sqr()))
718            .ok_or(Error::Singular)?;
719        let phase = column[pivot] / column[pivot].norm();
720        let normalization = phase * column.norm();
721        column /= normalization;
722        pivots.push(pivot);
723    }
724    if values.iter().chain(vectors.iter()).any(|&z| !finite(z)) {
725        return Err(Error::NonFinite("non-finite eigensystem".into()));
726    }
727    Ok(EigResidual {
728        values,
729        vectors,
730        pivots,
731        scale,
732        vectors_lu: OnceLock::new(),
733    })
734}
735
736/// Eigenvalues ordered by real part, then imaginary part. The same eigensystem
737/// residual supplies derivatives without imposing an eigenvector phase constraint.
738pub fn eigvals(operator: &DMatrix<Complex>) -> Result<EigResidual> {
739    let mut residual = eig(operator)?;
740    let n = residual.values.len();
741    let mut order: Vec<_> = (0..n).collect();
742    order.sort_by(|&i, &j| {
743        residual.values[i]
744            .re
745            .total_cmp(&residual.values[j].re)
746            .then_with(|| residual.values[i].im.total_cmp(&residual.values[j].im))
747    });
748    residual.values = order.iter().map(|&j| residual.values[j]).collect();
749    residual.vectors = DMatrix::from_fn(n, n, |i, j| residual.vectors[(i, order[j])]);
750    residual.pivots = order.iter().map(|&j| residual.pivots[j]).collect();
751    Ok(residual)
752}

754impl EigResidual {
755    fn vectors_lu(&self) -> Result<&Lu> {
756        self.vectors_lu
757            .get_or_init(|| Lu::new(self.vectors.clone()))
758            .as_ref()
759            .map_err(Clone::clone)
760    }
761
762    /// The eigenvalues in the solver's order, which the pullback reads.
763    #[must_use]
764    pub fn values(&self) -> &[Complex] {
765        &self.values
766    }
767
768    /// The right eigenvectors, one column per eigenvalue, which the pullback reads.
769    #[must_use]
770    pub const fn vectors(&self) -> &DMatrix<Complex> {
771        &self.vectors
772    }
773
774    /// Whether the phase pivot is tied with another largest component.
775    fn tied_pivot(&self, column: usize) -> bool {
776        let pivot = self.pivots[column];
777        let size = self.vectors[(pivot, column)].norm();
778        (0..self.values.len()).any(|i| {
779            i != pivot
780                && (size - self.vectors[(i, column)].norm()).abs() <= 64.0 * f64::EPSILON * size
781        })
782    }
783
784    /// Express the perturbation in the eigenbasis; its diagonal is the
785    /// eigenvalue derivative and its off-diagonal entries determine eigenvectors.
786    fn eigenbasis_tangent(&self, operator: &DMatrix<Complex>) -> Result<DMatrix<Complex>> {
787        let n = self.values.len();
788        if operator.shape() != (n, n) || operator.iter().any(|&z| !finite(z)) {
789            return Err(Error::InvalidInput("invalid eigensystem tangent".into()));
790        }
791        for j in 0..n {
792            for i in 0..j {
793                if (self.values[j] - self.values[i]).norm() <= 64.0 * f64::EPSILON * self.scale {
794                    return Err(Error::InvalidInput(
795                        "individual eigenmodes have no pushforward at repeated eigenvalues".into(),
796                    ));
797                }
798            }
799        }
800        let mut moved = product(operator, &self.vectors);
801        self.vectors_lu()?.solve_in_place(view_mut(&mut moved))?;
802        if moved.iter().any(|&z| !finite(z)) {
803            return Err(Error::Singular);
804        }
805        Ok(moved)
806    }
807
808    /// An ordered spectrum switches modes when real parts cross. Only equal
809    /// loss weights across a tied group make that change irrelevant.
810    fn check_value_order(&self, weights: Option<&[Complex]>) -> Result<()> {
811        for j in 0..self.values.len() {
812            for i in 0..j {
813                let tied = (self.values[j].re - self.values[i].re).abs()
814                    <= 64.0 * f64::EPSILON * self.scale;
815                let equal_weights = weights.is_some_and(|g| {
816                    (g[j] - g[i]).norm() <= 64.0 * f64::EPSILON * g[j].norm().max(g[i].norm())
817                });
818                if tied && !equal_weights {
819                    return Err(Error::InvalidInput(
820                        "ordered eigenvalue derivatives require distinct real parts or equal loss weights at ordering ties".into(),
821                    ));
822                }
823            }
824        }
825        Ok(())
826    }
827
828    /// Eigenvalue tangents with no eigenvector phase constraint. Distinct real
829    /// parts keep [`eigvals`]' ordering fixed; tied phase pivots are supported.
830    pub fn pushforward_values(&self, operator: &DMatrix<Complex>) -> Result<Vec<Complex>> {
831        let moved = self.eigenbasis_tangent(operator)?;
832        self.check_value_order(None)?;
833        Ok(moved.diagonal().iter().copied().collect())
834    }
835
836    /// Pullback of [`eigvals`], allowing equal weights at ordering ties.
837    pub fn pullback_values(&self, values: &[Complex]) -> Result<DMatrix<Complex>> {
838        let n = self.values.len();
839        let gradient = self.pullback(values, DMatrix::zeros(n, n))?;
840        self.check_value_order(Some(values))?;
841        Ok(gradient)
842    }
843
844    /// Eigenpair tangents with the same unit norm and real-positive pivot phase as
845    /// the primal. Repeated eigenvalues and tied phase pivots have no pushforward of
846    /// the complete eigenpair output. Use [`Self::pushforward_values`] when only
847    /// eigenvalues are needed.
848    pub fn pushforward(
849        &self,
850        operator: &DMatrix<Complex>,
851    ) -> Result<(Vec<Complex>, DMatrix<Complex>)> {
852        let n = self.values.len();
853        let mut moved = self.eigenbasis_tangent(operator)?;
854        if (0..n).any(|j| self.tied_pivot(j)) {
855            return Err(Error::InvalidInput(
856                "eigenvector pushforward requires a unique largest component; use eigvals for eigenvalues alone".into(),
857            ));
858        }
859        let values: Vec<_> = moved.diagonal().iter().copied().collect();
860        // E = V⁻¹ dA V; divide off-diagonal entries by λ_j - λ_i to get
861        // the unnormalized eigenvector changes. Its diagonal already gave dλ.
862        for j in 0..n {
863            for i in 0..n {
864                moved[(i, j)] = if i == j {
865                    Complex::default()
866                } else {
867                    ratio(moved[(i, j)], self.values[j] - self.values[i])
868                };
869            }
870        }
871        let mut vectors = product(&self.vectors, &moved);
872        for j in 0..n {
873            let pivot = self.pivots[j];
874            let normalization = self.vectors.column(j).dotc(&vectors.column(j)).re;
875            let phase = vectors[(pivot, j)].im / self.vectors[(pivot, j)].re;
876            let correction = Complex::new(normalization, phase);
877            for i in 0..n {
878                vectors[(i, j)] -= self.vectors[(i, j)] * correction;
879            }
880        }
881        if values.iter().chain(vectors.iter()).any(|&z| !finite(z)) {
882            return Err(Error::Singular);
883        }
884        Ok((values, vectors))
885    }
886
887    /// The operator gradient from the eigenvalue and eigenvector cotangents, with the
888    /// phase of each vector fixed at its pivot.
889    ///
890    /// Repeated eigenvalues support equal eigenvalue weights and zero vector
891    /// cotangents within each repeated group. Individual modes there have no pullback.
892    pub fn pullback(
893        &self,
894        values: &[Complex],
895        mut vectors: DMatrix<Complex>,
896    ) -> Result<DMatrix<Complex>> {
897        let n = self.values.len();
898        if values.len() != n
899            || vectors.shape() != (n, n)
900            || values.iter().chain(vectors.iter()).any(|&z| !finite(z))
901        {
902            return Err(Error::InvalidInput("invalid eigensystem cotangent".into()));
903        }
904        let vector_active: Vec<_> = vectors
905            .column_iter()
906            .map(|v| v.iter().any(|&z| z != Complex::default()))
907            .collect();
908        if !vector_active.iter().any(|&active| active) && values.iter().all(|&g| g == values[0]) {
909            // Sum of all eigenvalues is trace(A), including defective matrices.
910            return Ok(DMatrix::identity(n, n) * values[0]);
911        }
912        // Project out normalization and pivot-phase changes before pairing the
913        // cotangent with the off-diagonal eigenvector derivative.
914        for j in 0..n {
915            let inner = vectors.column(j).dotc(&self.vectors.column(j));
916            let pivot = self.pivots[j];
917            if inner.im.abs() > 64.0 * f64::EPSILON * vectors.column(j).norm() && self.tied_pivot(j)
918            {
919                return Err(Error::InvalidInput(
920                    "phase-dependent eigenvector pullback requires a unique largest component"
921                        .into(),
922                ));
923            }
924            for i in 0..n {
925                vectors[(i, j)] -= inner.re * self.vectors[(i, j)];
926            }
927            vectors[(pivot, j)] += Complex::i() * inner.im / self.vectors[(pivot, j)].re;
928        }
929        let mut g = product_views(view(&self.vectors).adjoint(), view(&vectors));
930        for j in 0..n {
931            for i in 0..n {
932                if i == j {
933                    g[(i, j)] = values[i];
934                    continue;
935                }
936                let gap = self.values[j] - self.values[i];
937                if gap.norm() <= 64.0 * f64::EPSILON * self.scale {
938                    let weight_scale = values[i].norm().max(values[j].norm());
939                    if vector_active[i]
940                        || vector_active[j]
941                        || (values[i] - values[j]).norm() > 64.0 * f64::EPSILON * weight_scale
942                    {
943                        return Err(Error::InvalidInput(
944                            "individual eigenmodes have no pullback at repeated eigenvalues".into(),
945                        ));
946                    }
947                    g[(i, j)] = Complex::default();
948                } else {
949                    g[(i, j)] = ratio(g[(i, j)], gap.conj());
950                }
951            }
952        }
953        let mut result = product_adjoint_right(&g, &self.vectors);
954        self.vectors_lu()?
955            .solve_adjoint_in_place(view_mut(&mut result))?;
956        if result.iter().any(|&z| !finite(z)) {
957            return Err(Error::Singular);
958        }
959        Ok(result)
960    }
961}

1038    /// Power-of-two equation (row) and unknown (column) units of a linear system.
1039    #[derive(Clone, Debug)]
1040    enum Units {
1041        /// Independent row exponents.
1042        Rows(Vec<i32>),
1043        /// Independent column exponents.
1044        Columns(Vec<i32>),
1045        /// Rows alternating `2^∓e` and columns `2^±e/2`, or with the two spreads
1046        /// exchanged, which makes the columns the more disparate axis.
1047        Alternating { exponent: i32, transposed: bool },
1048    }
1049
1050    impl Units {
1051        /// Row and column scales `(d_r, d_c)` of `A = d_r H d_c`.
1052        fn scales(&self, n: usize) -> (Vec<f64>, Vec<f64>) {
1053            let power = |e: i32| 2.0_f64.powi(e);
1054            let unit = vec![1.0; n];
1055            match self {
1056                Self::Rows(e) => (e.iter().map(|&e| power(e)).collect(), unit),
1057                Self::Columns(e) => (unit, e.iter().map(|&e| power(e)).collect()),
1058                &Self::Alternating {
1059                    exponent,
1060                    transposed,
1061                } => {
1062                    let sign = |i: usize| if i.is_multiple_of(2) { -1 } else { 1 };
1063                    let (wide, narrow) = (exponent, exponent / 2);
1064                    let (row, column) = if transposed {
1065                        (narrow, wide)
1066                    } else {
1067                        (wide, narrow)
1068                    };
1069                    (
1070                        (0..n).map(|i| power(sign(i) * row)).collect(),
1071                        (0..n).map(|j| power(-sign(j) * column)).collect(),
1072                    )
1073                }
1074            }
1075        }
1076    }
1077
1078    fn system() -> impl Strategy<Value = (DMatrix<Complex>, Units, DMatrix<Complex>)> {
1079        (1_usize..=40, 1_usize..=3).prop_flat_map(|(n, columns)| {
1080            let exponents = || prop::collection::vec(-400_i32..=400, n);
1081            (
1082                complex_matrix(n, n, 0.5),
1083                Just((0..n).collect::<Vec<_>>()).prop_shuffle(),
1084                prop_oneof![
1085                    exponents().prop_map(Units::Rows),
1086                    exponents().prop_map(Units::Columns),
1087                    (0_i32..=400, any::<bool>()).prop_map(|(exponent, transposed)| {
1088                        Units::Alternating {
1089                            exponent,
1090                            transposed,
1091                        }
1092                    }),
1093                ],
1094                complex_matrix(n, columns, 1.0),
1095            )
1096                .prop_map(move |(random, rows, units, y)| {
1097                    // `random + n I` with permuted rows, so pivoting must reorder them.
1098                    let size = Complex::from(f64::from(u32::try_from(n).unwrap()));
1099                    let mut h = random + DMatrix::identity(n, n) * size;
1100                    h = DMatrix::from_fn(n, n, |i, j| h[(rows[i], j)]);
1101                    (h, units, y)
1102                })
1103        })
1104    }

1117    /// For `A = d_r H d_c` with a well-conditioned `H`, the solve of `A X = d_r H Y`
1118    /// gives `d_c X = Y` and the adjoint solve of `Aᴴ Z = d_c Hᴴ Y` gives `d_r Z = Y`,
1119    /// whatever the units: equilibration removes them in both the direct and the
1120    /// transposed branch. The returned scales reproduce the equilibrated operator,
1121    /// and the operator cotangent is `-Z Xᴴ`, so scaling `A` and `B` jointly leaves
1122    /// the loss unchanged.
1123    fn check_units(
1124        h: &DMatrix<Complex>,
1125        units: &Units,
1126        y: &DMatrix<Complex>,
1127    ) -> Result<(), TestCaseError> {
1128        let n = h.nrows();
1129        let (row, column) = units.scales(n);
1130        let operator = DMatrix::from_fn(n, n, |i, j| h[(i, j)] * row[i] * column[j]);
1131        let hy = h * y;
1132        let rhs = DMatrix::from_fn(n, y.ncols(), |i, j| hy[(i, j)] * row[i]);
1133        let tolerance = 1e-13 * y.norm();
1134        let residual = solve_owned(operator.clone(), rhs.clone()).unwrap();
1135        let unknowns = DMatrix::from_fn(n, y.ncols(), |i, j| residual.value[(i, j)] * column[i]);
1136        prop_assert_close!(&unknowns, y, tolerance);
1137        let hz = h.adjoint() * y;
1138        let cotangent = DMatrix::from_fn(n, y.ncols(), |i, j| hz[(i, j)] * column[i]);
1139        let adjoint = residual.adjoint_rhs(cotangent.clone()).unwrap();
1140        let equations = DMatrix::from_fn(n, y.ncols(), |i, j| adjoint[(i, j)] * row[i]);
1141        prop_assert_close!(&equations, y, tolerance);
1142
1143        let SolveGradient {
1144            operator: ga,
1145            rhs: gb,
1146        } = residual.pullback(cotangent).unwrap();
1147        prop_assert_eq!(&gb, &adjoint);
1148        for i in 0..n {
1149            for j in 0..n {
1150                let terms = (0..y.ncols()).map(|p| (adjoint[(i, p)], residual.value[(j, p)]));
1151                let expected: Complex = terms.clone().map(|(z, x)| -z * x.conj()).sum();
1152                let scale: f64 = terms.map(|(z, x)| z.norm() * x.norm()).sum();
1153                prop_assert_close!(ga[(i, j)], expected, 1e-14 * scale, "entry ({}, {})", i, j);
1154            }
1155        }
1156        // Scaling A and B jointly leaves X unchanged: Re⟨ga, A⟩ + Re⟨gb, B⟩ = 0.
1157        let pairs = ga.iter().zip(&operator).chain(gb.iter().zip(&rhs));
1158        let magnitude: f64 = pairs.clone().map(|(g, x)| g.norm() * x.norm()).sum();
1159        let joint: f64 = pairs.map(|(g, x)| (g.conj() * x).re).sum();
1160        prop_assert_close!(joint, 0.0, 1e-13 * magnitude);
1161
1162        // Equilibration in the direct or transposed branch.
1163        let Units::Alternating {
1164            exponent,
1165            transposed,
1166        } = *units
1167        else {
1168            return Ok(());
1169        };
1170        // Past 2^32 the spreads, not the random entries of H, decide the branch.
1171        if n < 2 || exponent < 32 {
1172            return Ok(());
