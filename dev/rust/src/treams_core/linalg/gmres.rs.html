1//! Restarted GMRES (generalized minimal residual method) for complex linear systems given only
2//! through matrix-vector products. treams-rs extension.
3//!
4//! Each cycle builds an orthonormal Krylov basis of at most `restart` vectors by
5//! Arnoldi iteration, with two passes of modified Gram-Schmidt per vector, and takes
6//! the least-squares update through Givens rotations. A cycle ends early when the
7//! rotated residual estimate meets the tolerance, the new Krylov vector vanishes, or
8//! the iteration limit is reached. Before it accepts an answer or
9//! restarts, the solver recomputes the true residual `b - A x` with one more operator
10//! application, and only that residual decides convergence.
11#![allow(clippy::indexing_slicing)] // Krylov basis and Hessenberg indices below `restart + 1`.
12
13use crate::{Complex, Error, Result};
14
15/// Accuracy and memory budget of restarted GMRES.
16///
17/// The solve converges when the true residual norm is at most
18/// `max(atol, rtol * |b|)`. The defaults are `rtol = 1e-10`, `atol = 0`,
19/// `restart = 30` and `max_iterations = 300`; the Python constants
20/// `iterative.DEFAULT_RTOL`, `DEFAULT_ATOL`, `DEFAULT_RESTART` and
21/// `DEFAULT_MAX_ITERATIONS` repeat them.
22#[derive(Clone, Copy, Debug)]
23pub struct GmresOptions {
24    /// Relative residual tolerance with respect to the right-hand side norm.
25    pub rtol: f64,
26    /// Absolute residual tolerance, for right-hand sides near zero.
27    pub atol: f64,
28    /// Maximum number of Krylov vectors retained before a restart.
29    pub restart: usize,
30    /// Maximum total operator applications in the Arnoldi iterations.
31    pub max_iterations: usize,
32}
33
34impl Default for GmresOptions {
35    fn default() -> Self {
36        Self {
37            rtol: 1e-10,
38            atol: 0.0,
39            restart: 30,
40            max_iterations: 300,
41        }
42    }
43}
44
45impl GmresOptions {
46    /// Reject non-finite or negative tolerances, two zero tolerances, and a zero
47    /// restart length or iteration limit.
48    pub(crate) fn validate(self) -> Result<()> {
49        if !self.rtol.is_finite()
50            || !self.atol.is_finite()
51            || self.rtol < 0.0
52            || self.atol < 0.0
53            || self.rtol + self.atol == 0.0
54            || self.restart == 0
55            || self.max_iterations == 0
56        {
57            return Err(Error::InvalidInput(
58                "GMRES needs finite nonnegative tolerances, a positive tolerance, restart and iteration limit".into(),
59            ));
60        }
61        Ok(())
62    }
63}
64
65/// Proof of convergence for one right-hand side: the residual `b - A x` recomputed
66/// from the returned answer `x`, not the estimate of the Arnoldi iteration.
67///
68/// `residual_norm <= max(atol, rtol * rhs_norm)` holds for every returned value.
69#[derive(Clone, Copy, Debug)]
70pub struct Convergence {
71    /// Arnoldi iterations performed, including all restarts.
72    pub iterations: usize,
73    /// Euclidean norm of the recomputed residual `b - A x`.
74    pub residual_norm: f64,
75    /// Euclidean norm of the right-hand side.
76    pub rhs_norm: f64,
77}
78
79/// Euclidean norm without overflow or underflow of the squared terms.
80pub(crate) fn norm(vector: &[Complex]) -> f64 {
81    vector.iter().fold(0.0_f64, |sum, z| sum.hypot(z.norm()))
82}
83
84/// Solve `A x = rhs` for the operator that `apply` multiplies with, starting from
85/// `x = 0`. Fails with [`Error::NotConverged`] when the residual is not finite, when
86/// the iterations reach `options.max_iterations`, or when the Arnoldi iteration breaks
87/// down before convergence; errors of `apply` pass through.
88pub(crate) fn gmres(
89    rhs: &[Complex],
90    options: GmresOptions,
91    apply: impl Fn(&[Complex]) -> Result<Vec<Complex>>,
92) -> Result<(Vec<Complex>, Convergence)> {
93    let dimension = rhs.len();
94    let restart = options.restart.min(dimension).min(options.max_iterations);
95    let rhs_norm = norm(rhs);
96    let tolerance = options.atol.max(options.rtol * rhs_norm);
97    let mut answer = vec![Complex::default(); dimension];
98    let mut residual = rhs.to_vec();
99    let mut iterations = 0;
100    loop {
101        let residual_norm = norm(&residual);
102        if residual_norm.is_finite() && residual_norm <= tolerance {
103            return Ok((
104                answer,
105                Convergence {
106                    iterations,
107                    residual_norm,
108                    rhs_norm,
109                },
110            ));
111        }
112        if !residual_norm.is_finite() || iterations >= options.max_iterations {
113            return Err(Error::NotConverged(format!(
114                "GMRES did not converge after {iterations} iterations: residual {residual_norm:e}, tolerance {tolerance:e}"
115            )));
116        }
117        let mut basis = Vec::with_capacity(restart + 1);
118        basis.push(
119            residual
120                .iter()
121                .map(|z| z / residual_norm)
122                .collect::<Vec<_>>(),
123        );
124        let mut h = vec![Complex::default(); (restart + 1) * restart];
125        let mut rotations = Vec::<(f64, Complex)>::with_capacity(restart);
126        let mut g = vec![Complex::default(); restart + 1];
127        g[0] = Complex::new(residual_norm, 0.0);
128        let mut used = 0;
129        for j in 0..restart {
130            let mut w = apply(&basis[j])?;
131            // Two modified Gram-Schmidt passes keep orthogonality near resonance.
132            for _ in 0..2 {
133                for i in 0..=j {
134                    let dot: Complex = basis[i].iter().zip(&w).map(|(v, w)| v.conj() * w).sum();
135                    h[j * (restart + 1) + i] += dot;
136                    for (w, v) in w.iter_mut().zip(&basis[i]) {
137                        *w -= dot * v;
138                    }
139                }
140            }
141            let next = norm(&w);
142            h[j * (restart + 1) + j + 1] = Complex::new(next, 0.0);
143            for (i, &(c, s)) in rotations.iter().enumerate() {
144                let offset = j * (restart + 1) + i;
145                let a = h[offset];
146                let b = h[offset + 1];
147                h[offset] = c * a + s * b;
148                h[offset + 1] = -s.conj() * a + c * b;
149            }
150            let diagonal = j * (restart + 1) + j;
151            let a = h[diagonal];
152            let b = h[diagonal + 1];
153            let length = a.norm().hypot(b.norm());
154            if !length.is_finite() || length == 0.0 {
155                return Err(Error::NotConverged(
156                    "GMRES Arnoldi breakdown before convergence".into(),
157                ));
158            }
159            let phase = if a.norm() == 0.0 {
160                Complex::new(1.0, 0.0)
161            } else {
162                a / a.norm()
163            };
164            let c = a.norm() / length;
165            let s = phase * b.conj() / length;
166            h[diagonal] = phase * length;
167            h[diagonal + 1] = Complex::default();
168            rotations.push((c, s));
169            g[j + 1] = -s.conj() * g[j];
170            g[j] *= c;
171            used = j + 1;
172            iterations += 1;
173            if g[j + 1].norm() <= tolerance || next == 0.0 || iterations == options.max_iterations {
174                break;
175            }
176            basis.push(w.into_iter().map(|z| z / next).collect());
177        }
178        for i in (0..used).rev() {
179            let tail: Complex = ((i + 1)..used)
180                .map(|j| h[j * (restart + 1) + i] * g[j])
181                .sum();
182            g[i] = (g[i] - tail) / h[i * (restart + 1) + i];
183            for (x, v) in answer.iter_mut().zip(&basis[i]) {
184                *x += g[i] * v;
185            }
186        }
187        // Never certify convergence using the Hessenberg estimate alone.
188        residual = rhs
189            .iter()
190            .zip(apply(&answer)?)
191            .map(|(b, a)| b - a)
192            .collect();
193    }
194}

222            (m, rhs, restart) in (1_usize..=24).prop_flat_map(|n| {
223                (complex_matrix(n, n, 1.0), complex_matrix(n, 1, 1.0), 1..=n + 2)

230    /// Restarted GMRES on the nonnormal `A = I + M / (2 ‖M‖)`, whose field of values
231    /// stays at least 1/2 from zero, converges for every restart length to the direct
232    /// solution, reports its true residual within the tolerance, and without restarts
233    /// terminates within `n` iterations.
234    fn check_gmres(
235        m: &DMatrix<Complex>,
236        rhs: &[Complex],
237        restart: usize,
238    ) -> Result<(), TestCaseError> {
239        let n = m.nrows();
240        let shrink = if m.norm() > 0.0 { 0.5 / m.norm() } else { 0.0 };
241        let a = DMatrix::identity(n, n) + m * Complex::from(shrink);
242        let options = GmresOptions {
243            rtol: 1e-10,
244            atol: 0.0,
245            restart,
246            max_iterations: 4000,
247        };
248        let apply = |x: &[Complex]| Ok((&a * DVector::from_column_slice(x)).as_slice().to_vec());
249        let (answer, report) = gmres(rhs, options, apply).unwrap();
250        let b = DVector::from_column_slice(rhs);
251        let direct = a.clone().lu().solve(&b).unwrap();
252        prop_assert_close!(
253            answer.as_slice(),
254            direct.as_slice(),
255            1e-8 * (1.0 + direct.norm())
256        );
257        prop_assert_close!(
258            report.rhs_norm,
259            b.norm(),
260            1e-14 * b.norm() + f64::MIN_POSITIVE
261        );
262        let residual = (&b - &a * DVector::from_column_slice(&answer)).norm();
263        prop_assert_close!(
264            report.residual_norm,
265            residual,
266            1e-14 * b.norm() + f64::MIN_POSITIVE
267        );
268        prop_assert!(report.residual_norm <= options.rtol * report.rhs_norm);
269        if restart >= n {
270            prop_assert!(report.iterations <= n, "{} iterations", report.iterations);
271        }
272        Ok(())
273    }

275    #[test]
276    fn gmres_zero_happy_breakdown_restarts_and_failure() {
277        let identity = |x: &[Complex]| Ok(x.to_vec());
278        let (zero, report) = gmres(&[Complex::default(); 3], options(), identity).unwrap();
279        assert_eq!(zero, vec![Complex::default(); 3]);
280        assert_eq!(report.iterations, 0);
281        let rhs = vec![Complex::new(0.2, -0.3); 3];
282        let (answer, report) = gmres(&rhs, options(), identity).unwrap();
283        assert!((norm(&answer) - norm(&rhs)).abs() < 1e-14);
284        assert_eq!(report.iterations, 1);
285        let diagonal = |x: &[Complex]| {
286            Ok(x.iter()
287                .enumerate()
288                .map(|(i, x)| {
289                    Complex::new(1.0 + f64::from(u32::try_from(i).unwrap()) * 0.3, 0.1) * x
290                })
291                .collect())
292        };
293        let limited = GmresOptions {
294            restart: 1,
295            max_iterations: 1,
296            ..options()
297        };
298        assert!(gmres(&rhs, limited, diagonal).is_err());
299        let restarted = GmresOptions {
300            restart: 1,
301            ..options()
302        };
303        let (_, report) = gmres(&rhs, restarted, diagonal).unwrap();
304        assert!(report.iterations > 1);
305        assert!(report.residual_norm <= restarted.rtol * report.rhs_norm);
306        assert!(gmres(&rhs, options(), |_| Ok(vec![Complex::default(); 3])).is_err());
307    }
