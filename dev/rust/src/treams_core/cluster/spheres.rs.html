23/// What [`sphere_cluster`] saves for its pullback: the Mie residual of each sphere,
24/// the translation plan between two spheres, the positions and the interaction solve.
25#[derive(Clone, Debug)]
26pub struct SphereClusterResidual {
27    k0: f64,
28    positions: Vec<[f64; 3]>,
29    plan: TranslationPlan,
30    spheres: Vec<SphereResidual>,
31    interaction: InteractionResidual,
32}

34/// Gradients of the inputs of [`sphere_cluster`], in the order of its arguments.
35#[derive(Clone, Debug)]
36pub struct SphereClusterGradient {
37    /// Gradient of the vacuum wavenumber.
38    pub k0: f64,

40    pub radii: Vec<f64>,

42    pub epsilon: Vec<Complex>,

47/// Solve a finite cluster of homogeneous, nonmagnetic spheres in vacuum.
48///
49/// Returns the interacting T-matrix in the local bases of the spheres, each in
50/// [`sw::modes`] order up to `lmax`. [`particle_cluster`](super::particle_cluster)
51/// couples any particle T-matrices, without the radius and permittivity gradients.
52///
53/// Upstream: `treams.TMatrix.cluster` of `TMatrix.sphere` T-matrices, followed by
54/// `.interaction.solve()`.
55pub fn sphere_cluster(
56    lmax: u32,
57    k0: f64,
58    radii: &[f64],
59    epsilon: &[Complex],
60    positions: &[[f64; 3]],
61) -> Result<SphereClusterResidual> {
62    let ClusterParts {
63        blocks,
64        spheres,
65        plan,
66        coupling,
67    } = cluster_parts(lmax, k0, radii, epsilon, positions)?;
68    Ok(SphereClusterResidual {
69        k0,
70        positions: positions.to_vec(),
71        spheres,
72        plan,
73        interaction: interaction_blocks(blocks, coupling)?,
74    })
75}

77/// Assemble and factor a sphere cluster for repeated requested illuminations.
78///
79/// The returned factor differentiates the local T blocks, the coupling and the incident
80/// fields. Use [`IterativeSphereCluster`](super::IterativeSphereCluster) for the
81/// gradients of the radii, permittivities and positions.
82///
83/// treams-rs extension.
84pub fn sphere_cluster_factor(
85    lmax: u32,
86    k0: f64,
87    radii: &[f64],
88    epsilon: &[Complex],
89    positions: &[[f64; 3]],
90) -> Result<super::InteractionFactor> {
91    let parts = cluster_parts(lmax, k0, radii, epsilon, positions)?;
92    super::InteractionFactor::from_blocks(parts.blocks, parts.coupling)
93}

95/// Check a cluster of homogeneous spheres in vacuum: a positive finite vacuum
96/// wavenumber and radii, one finite permittivity and position per sphere, and no two
97/// spheres touching or overlapping.
98pub(crate) fn validate_spheres(
99    k0: f64,
100    radii: &[f64],
101    epsilon: &[Complex],
102    positions: &[[f64; 3]],
103) -> Result<()> {
104    if !k0.is_finite()
105        || k0 <= 0.0
106        || radii.is_empty()
107        || radii.len() != epsilon.len()
108        || radii.len() != positions.len()
109        || radii.iter().any(|r| !r.is_finite() || *r <= 0.0)
110        || epsilon.iter().any(|&z| !finite(z))
111        || positions.iter().flatten().any(|x| !x.is_finite())
112    {
113        return Err(Error::InvalidInput(
114            "require positive finite k0 and radii, and matching finite permittivities and positions".into(),
115        ));
116    }
117    for i in 0..radii.len() {
118        for j in 0..i {
119            let distance = positions[i]
120                .iter()
121                .zip(positions[j])
122                .fold(0.0_f64, |norm, (a, b)| norm.hypot(a - b));
123            if distance <= radii[i] + radii[j] || !distance.is_finite() {
124                return Err(Error::InvalidInput(
125                    "spherical particles must have finite separation and must not touch or overlap"
126                        .into(),
127                ));
128            }
129        }
130    }
131    Ok(())
132}

134/// The media of a homogeneous nonmagnetic sphere in vacuum.
135pub(crate) fn vacuum_sphere(epsilon: Complex) -> [Material; 2] {
136    [
137        Material {
138            epsilon,
139            ..Material::default()
140        },
141        Material::default(),
142    ]
143}

145struct ClusterParts {
146    /// The T-matrix of each sphere.
147    blocks: Vec<DMatrix<Complex>>,
148    spheres: Vec<SphereResidual>,
149    plan: TranslationPlan,
150    coupling: DMatrix<Complex>,
151}
152
153fn cluster_parts(
154    lmax: u32,
155    k0: f64,
156    radii: &[f64],
157    epsilon: &[Complex],
158    positions: &[[f64; 3]],
159) -> Result<ClusterParts> {
160    validate_spheres(k0, radii, epsilon, positions)?;
161    let modes = sw::modes(lmax)?;
162    let modes_per_particle = modes.len();
163    let (blocks, spheres): (Vec<_>, Vec<_>) = crate::threads::install(|| {
164        radii
165            .par_iter()
166            .zip(epsilon)
167            .map(|(&radius, &epsilon)| sphere(lmax, k0, &[radius], &vacuum_sphere(epsilon)))
168            .collect::<Result<Vec<_>>>()
169    })?
170    .into_iter()
171    .unzip();
172    let dimension = modes_per_particle
173        .checked_mul(spheres.len())
174        .ok_or_else(|| Error::InvalidInput("cluster is too large".into()))?;
175    let mut coupling = numerics::zeros(dimension, dimension)?;
176    let plan = TranslationPlan::between(&modes, &modes, true)?;
177    // Each worker owns complete source-particle columns; no locks or dense pair copies.
178    crate::threads::install(|| {
179        coupling
180            .as_mut_slice()
181            .par_chunks_mut(dimension * modes_per_particle)
182            .enumerate()
183            .try_for_each(|(j, columns)| -> Result<()> {
184                for i in 0..spheres.len() {
185                    if i == j {
186                        continue;
187                    }
188                    let displacement =
189                        std::array::from_fn(|axis| positions[i][axis] - positions[j][axis]);
190                    let values =
191                        plan.evaluate(Complex::new(k0, 0.0), displacement, Radial::Singular)?;
192                    for col in 0..modes_per_particle {
193                        let start = col * dimension + i * modes_per_particle;
194                        columns[start..start + modes_per_particle].copy_from_slice(
195                            &values[col * modes_per_particle..(col + 1) * modes_per_particle],
196                        );
197                    }
198                }
199                Ok(())
200            })
201    })?;
202    Ok(ClusterParts {
203        blocks,
204        spheres,
205        plan,
206        coupling,
207    })
208}

210impl SphereClusterResidual {
211    /// Interacting T-matrix in the local multipole bases.
212    #[must_use]
213    pub const fn value(&self) -> &DMatrix<Complex> {
214        self.interaction.value()
215    }
216
217    /// The shape of the interacting T-matrix.
218    #[must_use]
219    pub fn shape(&self) -> (usize, usize) {
220        self.interaction.shape()
221    }
222
223    /// Gradients of `k0`, the radii, the permittivities and the positions from
224    /// `cotangent`, the gradient of a real loss with respect to the interacting T-matrix.
225    ///
226    /// The spheres run in parallel; their gradients add in sphere order, so the Rayon
227    /// pool does not change them. The LU solve and the matrix products of
228    /// [`InteractionResidual::pullback_blocks`] run in faer with the worker count of
229    /// [`linalg`](crate::linalg).
230    pub fn pullback(self, cotangent: &DMatrix<Complex>) -> Result<SphereClusterGradient> {
231        let InteractionGradient { local, coupling } =
232            self.interaction.pullback_blocks(cotangent)?;
233        let n = self.spheres.len();
234        let modes_per_particle = coupling.nrows() / n;
235        let (k0, plan, positions) = (self.k0, &self.plan, &self.positions);
236        // Each sphere's Mie pullback and the couplings into it run in parallel; the
237        // gradients are summed in sphere order.
238        let parts = crate::threads::install(|| {
239            self.spheres
240                .into_par_iter()
241                .zip(local)
242                .enumerate()
243                .map(|(i, (sphere, local))| {
244                    let gradient = sphere.pullback(&local)?;
245                    let mut block = DMatrix::zeros(modes_per_particle, modes_per_particle);
246                    let pairs = (0..n)
247                        .filter(|&j| j != i)
248                        .map(|j| {
249                            block.copy_from(&coupling.view(
250                                (i * modes_per_particle, j * modes_per_particle),
251                                (modes_per_particle, modes_per_particle),
252                            ));
253                            let displacement =
254                                std::array::from_fn(|axis| positions[i][axis] - positions[j][axis]);
255                            plan.pullback(
256                                Complex::new(k0, 0.0),
257                                displacement,
258                                Radial::Singular,
259                                block.as_slice(),
260                            )
261                            .map(|(position, k)| (position, k.re))
262                        })
263                        .collect::<Result<Vec<_>>>()?;
264                    Ok((gradient, pairs))
265                })
266                .collect::<Result<Vec<_>>>()
267        })?;
268        let mut result = SphereClusterGradient {
269            k0: 0.0,
270            radii: vec![0.0; n],
271            epsilon: vec![Complex::default(); n],
272            positions: vec![[0.0; 3]; n],
273        };
274        for (i, (sphere, pairs)) in parts.into_iter().enumerate() {
275            result.radii[i] = sphere.layers.radii[0];
276            result.epsilon[i] = sphere.layers.epsilon[0];
277            result.k0 += sphere.k0;
278            for (j, (gradient, gk)) in (0..n).filter(|&j| j != i).zip(pairs) {
279                result.k0 += gk;
280                for (axis, g) in gradient.into_iter().enumerate() {
281                    result.positions[i][axis] += g;
282                    result.positions[j][axis] -= g;
283                }
284            }
285        }
286        Ok(result)
287    }
