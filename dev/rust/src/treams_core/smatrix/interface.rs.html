15/// The chiral Fresnel coefficients `res[out][in][out pol][in pol]`, transcribed from
16/// upstream `treams.coeffs.fresnel` with its variable names.
17///
18/// `out` and `in` index the propagation direction (0 = up, 1 = down), so `res[0][0]`
19/// transmits upward from the negative medium. The inputs index the medium first,
20/// negative side (below) then positive side (above), and the polarization second.
21fn fresnel_values<const N: usize>(
22    ks: [[Jet<N>; 2]; 2],
23    kzs: [[Jet<N>; 2]; 2],
24    zs: [Jet<N>; 2],
25) -> [[[[Jet<N>; 2]; 2]; 2]; 2] {
26    let ap = std::array::from_fn::<_, 2, _>(|i| ks[i][0] * kzs[i][1] + ks[i][1] * kzs[i][0]);
27    let am = std::array::from_fn::<_, 2, _>(|i| ks[i][0] * kzs[i][1] - ks[i][1] * kzs[i][0]);
28    let bp = ks[0][1] * kzs[1][1] + ks[1][1] * kzs[0][1];
29    let cp = ks[0][0] * kzs[1][0] + ks[1][0] * kzs[0][0];
30    let zs_diff = zs[0] - zs[1];
31    let zs_prod = 4.0 * zs[0] * zs[1];
32    let pref = 1.0 / (zs_diff * zs_diff * ap[0] * ap[1] + zs_prod * bp * cp);
33    let mut res = [[[[Jet::default(); 2]; 2]; 2]; 2];
34    for i in 0..2 {
35        let j = 1 - i;
36        res[i][i][0][0] = (zs[0] + zs[1]) * ks[j][0] * kzs[i][0] * bp * zs[j] * 4.0 * pref;
37        res[i][i][0][1] = -(zs[i] - zs[j])
38            * ks[j][0]
39            * kzs[i][1]
40            * (ks[j][1] * kzs[i][0] - ks[i][0] * kzs[j][1])
41            * zs[j]
42            * 4.0
43            * pref;
44        res[i][i][1][0] = -(zs[i] - zs[j])
45            * ks[j][1]
46            * kzs[i][0]
47            * (ks[j][0] * kzs[i][1] - ks[i][1] * kzs[j][0])
48            * zs[j]
49            * 4.0
50            * pref;
51        res[i][i][1][1] = (zs[0] + zs[1]) * ks[j][1] * kzs[i][1] * cp * zs[j] * 4.0 * pref;
52        res[j][i][0][0] = (-zs_diff * zs_diff * am[i] * ap[j]
53            - zs_prod * bp * (ks[i][0] * kzs[j][0] - ks[j][0] * kzs[i][0]))
54            * pref;
55        res[j][i][0][1] =
56            -2.0 * (zs[j] * zs[j] - zs[i] * zs[i]) * ks[i][0] * kzs[i][1] * ap[j] * pref;
57        res[j][i][1][0] =
58            -2.0 * (zs[j] * zs[j] - zs[i] * zs[i]) * ks[i][1] * kzs[i][0] * ap[j] * pref;
59        res[j][i][1][1] = (zs_diff * zs_diff * am[i] * ap[j]
60            - zs_prod * cp * (ks[i][1] * kzs[j][1] - ks[j][1] * kzs[i][1]))
61            * pref;
62    }
63    res
64}

66/// What [`fresnel`] saves for its pullback: the wavenumbers, the normal wavenumbers and
67/// the impedances. The pullback differentiates the closed forms again instead of keeping
68/// a Jacobian.

70pub struct FresnelResidual {
71    ks: [[Complex; 2]; 2],
72    kzs: [[Complex; 2]; 2],
73    zs: [Complex; 2],
74}

76/// Planar interface from full and normal wavenumbers and impedances, each indexed by
77/// medium (negative side, then positive side) and then by polarization.
78///
79/// Returns four helicity scattering blocks with polarization order (0, 1).
80///
81/// Upstream: `treams.coeffs.fresnel(ks, kzs, zs)`.
82pub fn fresnel(
83    ks: [[Complex; 2]; 2],
84    kzs: [[Complex; 2]; 2],
85    zs: [Complex; 2],
86) -> Result<(Blocks, FresnelResidual)> {
87    if ks
88        .iter()
89        .flatten()
90        .chain(kzs.iter().flatten())
91        .chain(zs.iter())
92        .any(|&v| !finite(v))
93    {
94        return Err(Error::InvalidInput("Fresnel inputs must be finite".into()));
95    }
96    if ks[0] == ks[1]
97        && kzs[0] == kzs[1]
98        && zs[0] == zs[1]
99        && zs[0] != Complex::default()
100        && ks[0].iter().all(|&k| k != Complex::default())
101        && kzs[0].contains(&Complex::default())
102    {
103        // Identical media have no boundary. This is the exact continuation of
104        // transmission through a grazing channel, whose Fresnel formula is 0/0.
105        return Ok((identity_blocks(2), FresnelResidual { ks, kzs, zs }));
106    }
107    let values = fresnel_values::<0>(
108        ks.map(|r| r.map(Jet::constant)),
109        kzs.map(|r| r.map(Jet::constant)),
110        zs.map(Jet::constant),
111    );
112    let value =
113        std::array::from_fn(|b| DMatrix::from_fn(2, 2, |i, j| values[b / 2][b % 2][i][j].value));
114    dimension(&value).map_err(|_| Error::Singular)?;
115    Ok((value, FresnelResidual { ks, kzs, zs }))
116}

118/// Cotangents of the inputs of [`fresnel`], in the order of its arguments; each array
119/// lists the medium on the negative side (below), then the positive side (above).
120#[derive(Clone, Copy, Debug)]
121pub struct FresnelGradient {
122    /// Cotangents of the full wavenumbers `ks` of polarizations 0 and 1.
123    pub ks: [[Complex; 2]; 2],
124    /// Cotangents of the normal wavenumbers `kzs` of polarizations 0 and 1.
125    pub kz: [[Complex; 2]; 2],
126    /// Cotangents of the impedances `zs`.
127    pub z: [Complex; 2],
128}
129
130impl FresnelResidual {
131    /// The gradients of all ten complex inputs under the real pairing `Re Σ conj(g)·dx`.
132    pub fn pullback(self, cotangent: &Blocks) -> Result<FresnelGradient> {
133        // Jet seed offsets: `ks[i][j]` at `KS + 2 i + j`, `kzs[i][j]` at `KZS + 2 i + j`
134        // and `zs[i]` at `ZS + i`.
135        const KS: usize = 0;
136        const KZS: usize = 4;
137        const ZS: usize = 8;
138        checked_dimension(cotangent, 2, "Fresnel cotangent blocks must be 2 by 2")?;
139        let ks = std::array::from_fn(|i| {
140            std::array::from_fn(|j| Jet::<10>::variable(self.ks[i][j], KS + 2 * i + j))
141        });
142        let kzs = std::array::from_fn(|i| {
143            std::array::from_fn(|j| Jet::variable(self.kzs[i][j], KZS + 2 * i + j))
144        });
145        let zs = std::array::from_fn(|i| Jet::variable(self.zs[i], ZS + i));
146        let values = fresnel_values(ks, kzs, zs);
147        if values
148            .iter()
149            .flatten()
150            .flatten()
151            .flatten()
152            .any(|v| !v.finite())
153        {
154            return Err(Error::InvalidInput(
155                "Fresnel derivative is undefined for this grazing-channel limit".into(),
156            ));
157        }
158        let mut result = [Complex::default(); 10];
159        for (b, block) in cotangent.iter().enumerate() {
160            for i in 0..2 {
161                for j in 0..2 {
162                    for (gradient, d) in
163                        result.iter_mut().zip(values[b / 2][b % 2][i][j].derivative)
164                    {
165                        *gradient += block[(i, j)] * d.conj();
166                    }
167                }
168            }
169        }
170        Ok(FresnelGradient {
171            ks: std::array::from_fn(|i| std::array::from_fn(|j| result[KS + 2 * i + j])),
172            kz: std::array::from_fn(|i| std::array::from_fn(|j| result[KZS + 2 * i + j])),
173            z: [result[ZS], result[ZS + 1]],
174        })
175    }
176}

178/// The four 2-by-2 scattering blocks of one interface as a 4-by-4 matrix.
179type InterfaceMatrix = nalgebra::SMatrix<Complex, 4, 4>;
180
181/// One 4-by-4 matrix (`lhs` or `rhs`) of the tangential-field matching, as jets.
182type BoundaryJets<const N: usize> = [[Jet<N>; 4]; 4];
183
184/// The normal wavenumber `sqrt(k^2 - q.q)` on the branch with a nonnegative imaginary
185/// part.
186fn normal_component<const N: usize>(k: Jet<N>, q: [Jet<N>; 2]) -> Result<Jet<N>> {
187    let mut normal = (k * k - q[0] * q[0] - q[1] * q[1]).sqrt();
188    // A grazing wave still has finite tangential fields. Only the derivative
189    // of its square-root dispersion is singular, not the forward interface.
190    if N > 0 && normal.value == Complex::default() {
191        return Err(Error::InvalidInput(
192            "interface derivative is undefined at an exact diffraction threshold".into(),
193        ));
194    }
195    // The principal root has a nonnegative real part; take the outgoing branch.
196    if normal.value.im < 0.0 {
197        normal = -normal;
198    }
199    Ok(normal)
200}

202/// Tangential fields `[E_a, E_b, H_a, H_b]` of the plane waves in one medium, indexed
203/// by propagation direction (0 = up) and polarization; `a` and `b` are the two axes
204/// after `axis` in cyclic order.
205pub(super) fn port_boundary<const N: usize>(
206    ks: [Jet<N>; 2],
207    z: Jet<N>,
208    q: [Jet<N>; 2],
209    axis: usize,
210) -> Result<[[[Jet<N>; 4]; 2]; 2]> {
211    let mut waves = [[[Jet::default(); 4]; 2]; 2];
212    let a = (axis + 1) % 3;
213    let b = (axis + 2) % 3;
214    for (pol, &wave_number) in ks.iter().enumerate() {
215        let normal = normal_component(wave_number, q)?;
216        for (direction, polarizations) in waves.iter_mut().enumerate() {
217            let mut vector = [Jet::default(); 3];
218            vector[a] = q[0];
219            vector[b] = q[1];
220            vector[axis] = if direction == 0 { normal } else { -normal };
221            let e = crate::pw::polarization_from_inputs(vector, u8::from(pol != 0))?;
222            let impedance = -Complex::i() * (if pol == 0 { -1.0 } else { 1.0 }) / z;
223            polarizations[pol] = [e[a], e[b], impedance * e[a], impedance * e[b]];
224        }
225    }
226    Ok(waves)
227}

229/// The tangential-field matching `lhs s = rhs` of an interface. The columns of `lhs`
230/// are the outgoing waves (up in the positive medium, down in the negative one), those
231/// of `rhs` the incoming waves (up in the negative medium, down in the positive one).
232fn interface_boundary<const N: usize>(
233    ks: [[Jet<N>; 2]; 2],
234    zs: [Jet<N>; 2],
235    q: [Jet<N>; 2],
236    axis: usize,
237) -> Result<(BoundaryJets<N>, BoundaryJets<N>)> {
238    let waves = [
239        port_boundary(ks[0], zs[0], q, axis)?,
240        port_boundary(ks[1], zs[1], q, axis)?,
241    ];
242    let lhs = std::array::from_fn(|row| {
243        std::array::from_fn(|col| {
244            if col < 2 {
245                waves[1][0][col][row]
246            } else {
247                -waves[0][1][col - 2][row]
248            }
249        })
250    });
251    let rhs = std::array::from_fn(|row| {
252        std::array::from_fn(|col| {
253            if col < 2 {
254                waves[0][0][col][row]
255            } else {
256                -waves[1][1][col - 2][row]
257            }
258        })
259    });
260    Ok((lhs, rhs))
261}

263/// What [`interface()`] saves for its pullback: the media, the transverse wavevector,
264/// the axis, the inverse of the tangential-field matching matrix and the blocks.
265#[derive(Debug)]
266pub struct InterfaceResidual {
267    ks: [[Complex; 2]; 2],
268    zs: [Complex; 2],
269    q: [f64; 2],
270    axis: usize,
271    inverse: Option<InterfaceMatrix>,
272    /// The scattering blocks as one matrix over (up, down) outputs and inputs.
273    result: InterfaceMatrix,
274    fixed_q: bool,
275}
276
277/// Blocks `[[b0, b1], [b2, b3]]` of 2-by-2 polarization blocks as one 4-by-4 matrix.
278fn interface_matrix(blocks: &Blocks) -> InterfaceMatrix {
279    InterfaceMatrix::from_fn(|i, j| blocks[2 * (i / 2) + j / 2][(i % 2, j % 2)])
280}
281
282/// The four 2-by-2 polarization blocks of a 4-by-4 interface matrix.
283fn interface_blocks(matrix: &InterfaceMatrix) -> Blocks {
284    std::array::from_fn(|block| {
285        DMatrix::from_fn(2, 2, |i, j| {
286            matrix[(2 * (block / 2) + i, 2 * (block % 2) + j)]
287        })
288    })
289}

291/// Interface from wavenumbers and impedances, indexed by medium (negative side, then
292/// positive side), and the real transverse wavevector `q`.
293///
294/// `axis` is the Cartesian normal, and `q` holds the two components after it in cyclic
295/// order. Returns four helicity scattering blocks with polarization order (0, 1). With
296/// `fixed_q`, the pullback holds `q` fixed and returns a zero `q` gradient.
297///
298/// Upstream: `treams.SMatrices.interface`.
299pub fn interface(
300    ks: [[Complex; 2]; 2],
301    zs: [Complex; 2],
302    q: [f64; 2],
303    axis: usize,
304    fixed_q: bool,
305) -> Result<(Blocks, InterfaceResidual)> {
306    if axis > 2
307        || q.iter().any(|x| !x.is_finite())
308        || ks
309            .iter()
310            .flatten()
311            .chain(&zs)
312            .any(|&v| !finite(v) || v == Complex::default())
313    {
314        return Err(Error::InvalidInput("interface requires finite transverse components, nonzero finite wavenumbers/impedances, and a Cartesian normal".into()));
315    }
316    let (lhs, rhs) = interface_boundary::<0>(
317        ks.map(|k| k.map(Jet::constant)),
318        zs.map(Jet::constant),
319        q.map(Jet::constant),
320        axis,
321    )?;
322    let matrix = InterfaceMatrix::from_fn(|i, j| lhs[i][j].value);
323    let identical = ks[0] == ks[1] && zs[0] == zs[1];
324    let matched_grazing = identical
325        && ks[0]
326            .iter()
327            .any(|&k| k * k - q[0] * q[0] - q[1] * q[1] == Complex::default());
328    let inverse = if matched_grazing {
329        None
330    } else {
331        Some(matrix.lu().try_inverse().ok_or(Error::Singular)?)
332    };
333    // The same medium on both sides is transparent for every transverse
334    // wavevector. Tangential matching alone is ill-conditioned near grazing
335    // incidence and cannot distinguish the ports at it, so use the exact value;
336    // the saved inverse still differentiates it away from the threshold.
337    let result = match inverse {
338        Some(inverse) if !identical => inverse * InterfaceMatrix::from_fn(|i, j| rhs[i][j].value),
339        _ => InterfaceMatrix::identity(),
340    };
341    let value = interface_blocks(&result);
342    dimension(&value).map_err(|_| Error::Singular)?;
343    Ok((
344        value,
345        InterfaceResidual {
346            ks,
347            zs,
348            q,
349            axis,
350            inverse,
351            result,
352            fixed_q,
353        },
354    ))
355}

357/// Cotangents of the inputs of [`interface`], in the order of its arguments; the media
358/// are listed negative side (below), then positive side (above).
359#[derive(Clone, Copy, Debug)]
360pub struct InterfaceGradient {
361    /// Cotangents of the wavenumbers of polarizations 0 and 1 in each medium.
362    pub ks: [[Complex; 2]; 2],
363    /// Cotangents of the impedances `zs` of the media.
364    pub z: [Complex; 2],
365    /// Cotangents of the real transverse wavevector components; zero when the forward
366    /// holds `q` fixed.
367    pub q: [f64; 2],
368}
369
370impl InterfaceResidual {
371    /// Reuse the 4-by-4 inverse for the implicit solve adjoint; recompute local field derivatives.
372    pub fn pullback(self, cotangent: &Blocks) -> Result<InterfaceGradient> {
373        // Jet seed offsets: `ks[i][j]` at `KS + 2 i + j`, `zs[i]` at `ZS + i` and `q[i]`
374        // at `Q + i`.
375        const KS: usize = 0;
376        const ZS: usize = 4;
377        const Q: usize = 6;
378        checked_dimension(cotangent, 2, "interface cotangent blocks must be 2 by 2")?;
379        let cotangent = interface_matrix(cotangent);
380        let inverse = self.inverse.ok_or_else(|| {
381            Error::InvalidInput(
382                "interface derivative is undefined at an exact diffraction threshold".into(),
383            )
384        })?;
385        let adjoint = inverse.adjoint() * cotangent;
386        let operator = -adjoint * self.result.adjoint();
387        let ks = std::array::from_fn(|i| {
388            std::array::from_fn(|j| Jet::<8>::variable(self.ks[i][j], KS + 2 * i + j))
389        });
390        let zs = std::array::from_fn(|i| Jet::variable(self.zs[i], ZS + i));
391        // At normal incidence the xy-interface blocks depend on q only to second
392        // order. Use their zero first derivative instead of an undefined
393        // azimuth derivative in the intermediate polarization vectors.
394        let fixed_q = self.fixed_q || (self.axis == 2 && self.q.iter().all(|&q| q == 0.0));
395        let q = std::array::from_fn(|i| {
396            if fixed_q {
397                Jet::constant(self.q[i])
398            } else {
399                Jet::variable(self.q[i], Q + i)
400            }
401        });
402        let (lhs, rhs) = interface_boundary(ks, zs, q, self.axis)?;
403        let mut result = [Complex::default(); 8];
404        for i in 0..4 {
405            for j in 0..4 {
406                for (a, g) in result.iter_mut().enumerate() {
407                    *g += operator[(i, j)] * lhs[i][j].derivative[a].conj()
408                        + adjoint[(i, j)] * rhs[i][j].derivative[a].conj();
409                }
410            }
411        }
412        Ok(InterfaceGradient {
413            ks: std::array::from_fn(|i| std::array::from_fn(|j| result[KS + 2 * i + j])),
414            z: [result[ZS], result[ZS + 1]],
415            q: [result[Q].re, result[Q + 1].re],
416        })
417    }
418}

420/// What [`propagation`] saves for its pullback: the wavevectors, the displacement and
421/// the diagonal phases.
422#[derive(Debug)]
423pub struct PropagationResidual {
424    vectors: Vec<[Complex; 3]>,
425    distance: [f64; 3],
426    /// Upward and downward transmission phase of every mode.
427    phases: Vec<[Complex; 2]>,
428}
429
430/// Propagate by a Cartesian displacement, given upgoing wavevectors for every mode.
431///
432/// The blocks are diagonal transmission phases and zero reflections.
433///
434/// Upstream: `treams.SMatrices.propagation`.
435pub fn propagation(
436    vectors: Vec<[Complex; 3]>,
437    distance: [f64; 3],
438) -> Result<(Blocks, PropagationResidual)> {
439    if vectors.is_empty()
440        || vectors.iter().flatten().any(|&v| !finite(v))
441        || distance.iter().any(|r| !r.is_finite())
442    {
443        return Err(Error::InvalidInput(
444            "propagation requires finite nonempty wavevectors and displacement".into(),
445        ));
446    }
447    let phases = vectors
448        .iter()
449        .map(|k| {
450            [1.0, -1.0].map(|sign| {
451                (Complex::i()
452                    * (sign * (k[0] * distance[0] + k[1] * distance[1]) + k[2] * distance[2]))
453                    .exp()
454            })
455        })
456        .collect::<Vec<_>>();
457    // Evanescent waves displaced against their decay overflow.
458    if phases.iter().flatten().any(|&phase| !finite(phase)) {
459        return Err(Error::NonFinite("non-finite propagation phase".into()));
460    }
461    let n = vectors.len();
462    let mut value: Blocks = std::array::from_fn(|_| DMatrix::zeros(n, n));
463    for (i, phase) in phases.iter().enumerate() {
464        value[0][(i, i)] = phase[0];
465        value[3][(i, i)] = phase[1];
466    }
467    Ok((
468        value,
469        PropagationResidual {
470            vectors,
471            distance,
472            phases,
473        },
474    ))
475}

477/// Cotangents of the inputs of [`propagation`], in the order of its arguments.
478#[derive(Clone, Debug)]
479pub struct PropagationGradient {
480    /// Complex wavevector cotangents, one per mode.
481    pub vectors: Vec<[Complex; 3]>,
482    /// Real displacement cotangent.
483    pub distance: [f64; 3],
484}
485
486impl PropagationResidual {
487    /// The shape `(n, n)` of each scattering block, with `n` modes in each direction.
488    #[must_use]
489    pub fn shape(&self) -> (usize, usize) {
490        (self.vectors.len(), self.vectors.len())
491    }
492
493    /// Wavevector and displacement gradients from the cotangents of the four blocks.
494    pub fn pullback(self, cotangent: &Blocks) -> Result<PropagationGradient> {
495        let n = checked_dimension(
496            cotangent,
497            self.vectors.len(),
498            "propagation cotangent shape mismatch",
499        )?;
500        let mut vectors = vec![[Complex::default(); 3]; n];
501        let mut distance = [0.0; 3];
502        for (i, (k, phase)) in self.vectors.iter().zip(&self.phases).enumerate() {
503            for ((block, sign), phase) in [(0, 1.0), (3, -1.0)].into_iter().zip(phase) {
504                let weight = cotangent[block][(i, i)].conj() * Complex::i() * phase;
505                for (a, s) in [sign, sign, 1.0].into_iter().enumerate() {
506                    vectors[i][a] += (weight * s * self.distance[a]).conj();
507                    distance[a] += (weight * s * k[a]).re;
508                }
509            }
510        }
511        Ok(PropagationGradient { vectors, distance })
512    }
