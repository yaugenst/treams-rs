22/// `F_1(v) = Gamma(1, v) / (-v) = exp(-v) / (-v)` of the 3D spherical and 2D
23/// cylindrical reciprocal sums, in closed form with `dF_1/dv = F_2 = (1 + v) exp(-v) / v^2`.
24fn exponential_kernel<const N: usize>(v: Jet<N>) -> Jet<N> {
25    let z = lower_side(v.value);
26    let exponential = (-z).exp();
27    let value = exponential / (-z);
28    if N == 0 {
29        return Jet::constant(value);
30    }
31    v.chain(value, (1.0 + z) * exponential / (z * z))
32}

34/// The inner sum `S_lmn` of the 2D spherical summand of [`reciprocal_term`] (upstream
35/// `_s_sum_sw2d` times `e^(i m phi)`), with `z = k r_z`, `beta^2 = beta2` and
36/// `beta e^(i phi) = (q_x + i q_y) / k`:
37/// `sum_s (-z)^(2n - s) beta^(l - s) e^(i m phi) / ((2n - s)! (s - n)! ((l + |m| - s) / 2)!
38/// ((l - |m| - s) / 2)!)` over `s = n..=min(l - |m|, 2n)` with even `l - |m| - s`.
39fn polynomial_sw2d<const N: usize>(
40    l: i32,
41    m: i32,
42    n: i32,
43    z: Jet<N>,
44    beta2: Jet<N>,
45    azimuth: Jet<N>,
46) -> Jet<N> {
47    let mut sum = Jet::default();
48    for s in n..=(l - m.abs()).min(2 * n) {
49        if (l - m.abs() - s) % 2 != 0 {
50            continue;
51        }
52        sum += (-z).powi(2 * n - s)
53            * beta2.powi((l - s - m.abs()) / 2)
54            * (-log_factorial(2 * n - s)
55                - log_factorial(s - n)
56                - log_factorial((l + m.abs() - s) / 2)
57                - log_factorial((l - m.abs() - s) / 2))
58            .exp();
59    }
60    sum * azimuth.powi(m.abs())
61}

63/// The inner sum `S_lmn` of the 1D spherical summand of [`reciprocal_term`] (upstream
64/// `_s_sum_sw1d` times `(-1)^m e^(i m phi)`), with `rho2 = (k rho)^2`, `beta = q_z / k`
65/// and the azimuth `phi` of the shift: `sum_s (k rho)^(2n - s) beta^(l - s) (-1)^m
66/// e^(i m phi) / ((n - (s + |m|) / 2)! (n - (s - |m|) / 2)! (l - s)! (s - n)!)` over
67/// `s = n..=min(2n - |m|, l)` with even `s - |m|`.
68fn polynomial_sw1d<const N: usize>(
69    l: i32,
70    m: i32,
71    n: i32,
72    rho2: Jet<N>,
73    beta: Jet<N>,
74    azimuth: Jet<N>,
75) -> Jet<N> {
76    let mut sum = Jet::default();
77    for s in n..=(2 * n - m.abs()).min(l) {
78        if (m.abs() - s) % 2 != 0 {
79            continue;
80        }
81        sum += rho2.powi((2 * n - s - m.abs()) / 2)
82            * beta.powi(l - s)
83            * (-log_factorial(n - s.midpoint(m.abs()))
84                - log_factorial(n - (s - m.abs()) / 2)
85                - log_factorial(l - s)
86                - log_factorial(s - n))
87            .exp();
88    }
89    sum * azimuth.powi(m.abs())
90}

92/// The inner sum `S_ln` of the 1D cylindrical summand of [`reciprocal_term`] (upstream
93/// `_s_sum_cw1d`, whose docstring formula omits the factor `l!` that its code includes)
94/// for `l = |m|`, with `y = k r_y` (`-k r_y` for negative `m`) and `beta = q_x / k`:
95/// `sum_s (-y)^(2n - s) beta^(l - s) l! / (2^s (2n - s)! (l - s)! (s - n)!)` over
96/// `s = n..=min(2n, l)`.
97fn polynomial_cw1d<const N: usize>(l: i32, n: i32, y: Jet<N>, beta: Jet<N>) -> Jet<N> {
98    let mut sum = Jet::default();
99    for s in n..=(2 * n).min(l) {
100        sum += (-y).powi(2 * n - s)
101            * beta.powi(l - s)
102            * (log_factorial(l)
103                - log_factorial(2 * n - s)
104                - log_factorial(l - s)
105                - log_factorial(s - n)
106                - f64::from(s) * 2.0_f64.ln())
107            .exp();
108    }
109    sum
110}

112/// The reciprocal term of a value-only 1D cylindrical sum on its axis, which needs half
113/// of the polynomial terms: adjacent half-integer gamma functions share one evaluation,
114/// recurring downward near zero and upward at large arguments, where a downward step
115/// would subtract the leading asymptotic term. On a `sheet` ([`reduced_gamma`]) the
116/// endpoint term takes the phase `e^(-i pi n)` of the lower side.
117fn axial_cylindrical_reciprocal(
118    order: i32,
119    beta: Complex,
120    z: Complex,
121    eta: Complex,
122    sheet: Option<Sheet>,
123) -> Complex {
124    let maximum = order / 2;
125    let upward = z.norm() > 4.0;
126    let mut gamma = reduced_gamma(1 - 2 * if upward { maximum } else { 0 }, z, sheet);
127    let endpoint = if maximum > 0 {
128        (if sheet.is_none() && z.im > 0.0 {
129            Complex::i()
130        } else {
131            -Complex::i()
132        }) * (-z).exp()
133    } else {
134        Complex::default()
135    };
136    let mut sum = Complex::default();
137    for step in 0..=maximum {
138        let n = if upward { maximum - step } else { step };
139        let weight = beta.powi(order - 2 * n)
140            * (0.5 * eta * eta).powi(n)
141            * (log_factorial(order) - log_factorial(n) - log_factorial(order - 2 * n)).exp()
142            / (2.0_f64.sqrt() * eta);
143        sum += weight * gamma;
144        if step < maximum {
145            let phase = if n % 2 == 0 { endpoint } else { -endpoint };
146            gamma = if upward {
147                ((0.5 - f64::from(n)) * gamma + phase) / (-z)
148            } else {
149                (-z * gamma + phase) / (-0.5 - f64::from(n))
150            };
151        }
152    }
153    sum
154}

156/// `q . q - k^2` of a diffraction order to a few ulps of itself, from exact products and
157/// compensated additions: next to a threshold the rounded `q . q / k^2 - 1` loses about
158/// `eps k^2 / |k_q^2|` of it. Only the zeroth order of an unreduced Bloch vector takes it
159/// (within [`NEAR_THRESHOLD`]): every other `q` carries the rounding of its reciprocal
160/// vector, as large as the rounding this removes.
161fn compensated_threshold_distance(q: [Complex; 3], k: Complex) -> Complex {
162    let (mut re, mut im) = (Compensated::default(), Compensated::default());
163    for q in q {
164        re.add_product(q.re, q.re);
165        re.add_product(-q.im, q.im);
166        im.add_product(2.0 * q.re, q.im);
167    }
168    re.add_product(-k.re, k.re);
169    re.add_product(k.im, k.im);
170    im.add_product(-2.0 * k.re, k.im);
171    Complex::new(re.total(), im.total())
172}

174/// Largest moduli of the real and imaginary parts of `q . q / k^2 - 1` of the zeroth
175/// diffraction order at which the reciprocal sums take
176/// [`compensated_threshold_distance`]: there the rounded form loses 50 ulps of `v` or
177/// more (about 3 ulps of `q . q / k^2` over the difference).
178const NEAR_THRESHOLD: f64 = 1.0 / 16.0;
179
180/// Largest `|v|` (for the reciprocal terms) or `|k_q^2 / k^2|` (for the spectral series)
181/// at which a diffraction order counts as on its threshold `|q| = k`, where the sums
182/// are singular.
183pub(super) const THRESHOLD_DISTANCE: f64 = 1e-14;
184
185/// A sum of products that keeps the rounding errors of its products and additions
186/// apart until the end.
187#[derive(Clone, Copy, Debug, Default)]
188struct Compensated {
189    sum: f64,
190    error: f64,
191}
192
193impl Compensated {
194    fn add_product(&mut self, a: f64, b: f64) {
195        let product = a * b;
196        let total = self.sum + product;
197        let back = total - self.sum;
198        self.error += a.mul_add(b, -product) + (self.sum - (total - back)) + (product - back);
199        self.sum = total;
200    }
201
202    fn total(self) -> f64 {
203        self.sum + self.error
204    }
205}

216/// One reciprocal summand `T(q)` of the sum, without the Bloch phase `exp(-i q . r)`
217/// that the caller multiplies in. Every sum shares the argument `v = (q . q / k^2 - 1) /
218/// (2 eta^2)`; `wave` and `dim` select the sum. 1D spherical sums, whose odd Kambe
219/// orders cancel by up to `e^(w^2)` at `w = k rho eta`, add the rounding bounds of its
220/// value and derivatives in units of epsilon to `bound`; the other sums keep their
221/// relative accuracy.
222pub(super) fn reciprocal_term<const N: usize>(
223    wave: Family,
224    dim: usize,
225    k: Jet<N>,
226    Diffraction { q, exact }: Diffraction<N>,
227    r: [Jet<N>; 3],
228    eta: Complex,
229    split: &Split,
230    measure: Jet<N>,
231    bound: &mut Rounding<N>,
232) -> Result<Jet<N>> {
233    let beta2 = q.into_iter().map(|q| q * q).sum::<Jet<N>>() / (k * k);
234    let mut value = (beta2 - 1.0) / (2.0 * eta * eta);
235    let distance = beta2.value - 1.0;
236    if exact && distance.re.abs().max(distance.im.abs()) < NEAR_THRESHOLD {
237        value.value = compensated_threshold_distance(q.map(|q| q.value), k.value)
238            / (k.value * k.value)
239            / (2.0 * eta * eta);
240    }
241    value.value = lower_side(value.value);
242    if value.value.norm() <= THRESHOLD_DISTANCE {
243        return Err(Error::InvalidInput(
244            "lattice sum is at a diffraction threshold; supply a limiting complex wavenumber"
245                .into(),
246        ));
247    }
248    let i = Complex::i();
249    // The five summands share this one match: the term loop of every sum inlines
250    // this function, and in separate functions the summands make the forward 3D sums
251    // up to 13% slower.
252    Ok(match (wave, dim) {
253        // 3D spherical sum (upstream `recsumsw3d`, summand `_recsw3d`), with the cell
254        // volume `V = measure` and `F_1` of `exponential_kernel`:
255        // `2 i (-i)^l pi Y_lm(q / |q|) (|q| / k)^l F_1(v) / (V k^3 eta^2)`.
256        (Family::Spherical { l, m }, 3) => {
257            let harmonic = harmonic_normalization(l, m) * solid_jet(l, m, q);
258            2.0 * i * (-i).powi(l) * PI / (measure * k.powi(3)) * harmonic / k.powi(l)
259                * exponential_kernel(value)
260                / (eta * eta)
261        }
262        // 2D cylindrical sum (upstream `recsumcw2d`, summand `_reccw2d`), with the cell
263        // area `A = measure`, `beta e^(i phi) = (q_x + i q_y) / k` and `F_1` of
264        // `exponential_kernel`: `2 i (-i)^m beta^|m| e^(i m phi) F_1(v) / (A k^2 eta^2)`.
265        (Family::Cylindrical { m }, 2) => {
266            let xy = azimuthal(q[0], q[1], m);
267            2.0 * i * (-i).powi(m) / (measure * k * k)
268                * (xy / k).powi(m.abs())
269                * exponential_kernel(value)
270                / (eta * eta)
271        }
272        // 2D spherical sum for any shift (upstream `recsumsw2d_shift`, summand
273        // `_n_sum_sw2d`; at `r_z = 0` it also gives upstream `recsumsw2d`), with the cell
274        // area `A = measure`:
275        //
276        // `sqrt(2l + 1) (-i)^m sqrt((l + m)! (l - m)!) / (A k^2 (-2)^l)
277        // sum_(n = 0..=l - |m|) (2 eta^2)^n / (sqrt(2) eta) F_(1/2 - n)(v, (k r_z eta)^2) S_lmn`
278        //
279        // with the reduced integrals `F` of `Reduced` and the inner sums `S_lmn` of
280        // `polynomial_sw2d`, summed by `half_integer_orders`.
281        (Family::Spherical { l, m }, 2) => {
282            let z = k * r[2];
283            let azimuth = azimuthal(q[0], q[1], m) / k;
284            let factor = 1.0 / (2.0_f64.sqrt() * eta);
285            // For Im k < 0 the sums in the plane depend on the split, and with
286            // Re(k eta) < 0 the real parts do not converge to the sum: both keep the
287            // principal branches.
288            let sheet = (k.value.im >= 0.0 && (k.value * eta).re > 0.0).then(|| {
289                let q2 = q[0].value * q[0].value + q[1].value * q[1].value;
290                split_sheet(split, q2)
291            });
292            let mut reduced = Reduced::new(value, (z * eta).powi(2), sheet);
293            let sum = half_integer_orders(&mut reduced, split, (factor, eta), l - m.abs(), |n| {
294                polynomial_sw2d(l, m, n, z, beta2, azimuth)
295            });
296            f64::from(2 * l + 1).sqrt()
297                * (-i).powi(m)
298                * log_factorial(l + m).midpoint(log_factorial(l - m)).exp()
299                / (measure * k * k * (-2.0_f64).powi(l))
300                * sum
301        }
302        // 1D spherical sum along `z` (upstream `recsumsw1d_shift`, summand
303        // `_n_sum_sw1d`), with the period `a = measure` and `rho^2 = r_x^2 + r_y^2`:
304        //
305        // `-i sqrt((2l + 1) / pi) (-i)^(l - m) sqrt((l + m)! (l - m)!) / (2 a k)
306        // sum_(n = |m|..=l) (eta^2 / 2)^n F_(-n)(v, (k rho eta)^2) S_lmn`
307        //
308        // with the reduced integrals `F` of `Reduced` and the inner sums `S_lmn` of
309        // `polynomial_sw1d`. Adds the rounding bounds of the integer orders `F_(-n)` to
310        // `bound`.
311        (Family::Spherical { l, m }, 1) => {
312            let beta = q[2] / k;
313            let rho2 = k * k * (r[0] * r[0] + r[1] * r[1]);
314            let azimuth = -k * azimuthal(r[0], r[1], m);
315            let mut sum = Jet::default();
316            let mult = 0.5 * eta * eta;
317            let mut factor = mult.powi(m.abs());
318            let sheet = split_sheet(split, q[2].value * q[2].value);
319            let mut reduced = Reduced::new(value, rho2 * eta * eta, Some(sheet));
320            let mut sum_bound = Rounding::default();
321            for n in m.abs()..=l {
322                let polynomial = polynomial_sw1d(l, m, n, rho2, beta, azimuth);
323                if !polynomial.is_zero() {
324                    let (integral, integral_bound) = reduced.get(-2 * n);
325                    sum += factor * integral * polynomial;
326                    sum_bound += integral_bound.times(&(factor * polynomial));
327                }
328                factor *= mult;
329            }
330            let prefactor = -i
331                * (f64::from(2 * l + 1) / PI).sqrt()
332                * (-i).powi(l - m)
333                * log_factorial(l + m).midpoint(log_factorial(l - m)).exp()
334                / (2.0 * measure * k);
335            *bound += sum_bound.times(&prefactor);
336            prefactor * sum
337        }
338        // 1D cylindrical sum along `x` (upstream `recsumcw1d_shift`, summand
339        // `_n_sum_cw1d`), with the period `a = measure`:
340        //
341        // `2 (-i)^m / (sqrt(pi) a k)
342        // sum_(n = 0..=|m|) (2 eta^2)^n / (sqrt(2) eta') F_(1/2 - n)(v, (k r_y eta)^2) S_(|m|n)`
343        //
344        // with `eta' = ±eta`, the root of `eta^2` with `Re(k eta') >= 0`, the reduced
345        // integrals `F` of `Reduced` and the inner sums `S_ln` of `polynomial_cw1d`,
346        // summed by `half_integer_orders`. Value-only sums on the axis (`r_y = 0`) take
347        // `axial_cylindrical_reciprocal`.
348        (Family::Cylindrical { m }, 1) => {
349            let beta = q[0] / k;
350            let y = k * if m < 0 { -r[1] } else { r[1] };
351            // The real part and the reduced integrals depend on eta^2 alone, the sum on
352            // the sign of eta only through the factor 1 / eta: splits with Re(k eta) < 0
353            // take the root -eta of eta^2, with which the Ewald parts converge to the
354            // sum. For Im k < 0 the sums on the axis depend on the split: they keep the
355            // principal branches.
356            let root = if (k.value * eta).re < 0.0 { -eta } else { eta };
357            let factor = 1.0 / (2.0_f64.sqrt() * root);
358            let sheet = (k.value.im >= 0.0).then(|| split_sheet(split, q[0].value * q[0].value));
359            if N == 0 && r[1].value == Complex::default() {
360                return Ok(Jet::constant(
361                    2.0 * (-i).powi(m) / (PI.sqrt() * measure.value * k.value)
362                        * axial_cylindrical_reciprocal(
363                            m.abs(),
364                            beta.value,
365                            value.value,
366                            root,
367                            sheet,
368                        ),
369                ));
370            }
371            let mut reduced = Reduced::new(value, (y * eta).powi(2), sheet);
372            let sum = half_integer_orders(&mut reduced, split, (factor, eta), m.abs(), |n| {
373                polynomial_cw1d(m.abs(), n, y, beta)
374            });
375            2.0 * (-i).powi(m) / (PI.sqrt() * measure * k) * sum
376        }
377        _ => return Err(Error::InvalidInput("invalid lattice dimension".into())),
378    })
379}

381/// `sum_n factor (2 eta^2)^n F_(1/2 - n) P(n)` over the half-integer orders
382/// `n = 0..=orders` of one reciprocal point with the degree polynomials `P` (zero ones
383/// skipped). Below every automatic split the orders weigh their terms, and are summed
384/// again from the single Kambe integrals where those bound the sum more tightly
385/// ([`Reduced::prefers_single`]).
386fn half_integer_orders<const N: usize>(
387    reduced: &mut Reduced<N>,
388    split: &Split,
389    (factor, eta): (Complex, Complex),
390    orders: i32,
391    polynomial: impl Fn(i32) -> Jet<N>,
392) -> Jet<N> {
393    let sum = |reduced: &mut Reduced<N>| {
394        let (mut sum, mut factor) = (Jet::default(), factor);
395        for n in 0..=orders {
396            let polynomial = polynomial(n);
397            if !polynomial.is_zero() {
398                if split.small {
399                    reduced.weigh(1 - 2 * n, factor * polynomial.value);
400                }
401                sum += factor * reduced.get(1 - 2 * n).0 * polynomial;
402            }
403            factor *= 2.0 * eta * eta;
404        }
405        sum
406    };
407    let from_recurrence = sum(reduced);
408    if split.small && reduced.prefers_single() {
409        sum(reduced)
410    } else {
411        from_recurrence
412    }
413}

415/// The self term: the Cartesian Taylor term of `real_Ewald(-r) - outgoing(-r)` at the
416/// origin. Upstream `zero2d` (cylindrical) and `zero3d` (spherical) give its value at
417/// degree and order zero. Keeping its first-order polynomial gives the regular
418/// image-sum derivative at self.
419///
420/// # Branch
421///
422/// Its incomplete gamma function of `v = -1 / (2 eta^2)` takes the sheet
423/// ([`self_sheet`]) of the principal `ln((k eta)^2)` of the real-space Kambe integrals
424/// less `2 ln k` of the outgoing waves. For a sheet `j` turns off the principal one,
425/// DLMF 8.2.10 gives `2 Gamma(a) - Gamma(a, v)` for half-integer `a` and odd `j`, and
426/// `Gamma(-n, v) - 2 pi i j (-1)^n / n!` for `a = -n`.
427pub(super) fn self_term<const N: usize>(
428    wave: Family,
429    k: Jet<N>,
430    r: [Jet<N>; 3],
431    eta: Complex,
432) -> Jet<N> {
433    let value = lower_side(-0.5 / (eta * eta));
434    let turns = sheet_turns(self_sheet(k.value, eta), value);
435    match wave {
436        Family::Spherical { l, m } => {
437            let degree = -f64::from(l) - 0.5;
438            let mut gamma = special::upper_gamma(degree, value);
439            if turns % 2.0 != 0.0 {
440                gamma = 2.0 * libm::tgamma(degree) - gamma;
441            }
442            k.powi(l)
443                * solid_jet(l, m, r)
444                * harmonic_normalization(l, m)
445                * (2.0_f64.powi(-l - 1) / PI.sqrt())
446                * gamma
447        }
448        Family::Cylindrical { m } => {
449            let (xy, sign) = (azimuthal(r[0], r[1], m), negative_order_sign(m));
450            let n = m.abs();
451            let parity = if n % 2 == 0 { 1.0 } else { -1.0 };
452            let gamma = special::upper_gamma(-f64::from(n), value)
453                - Complex::new(0.0, TAU * turns) * parity * (-log_factorial(n)).exp();
454            Complex::i() / PI * sign * (0.5 * k * xy).powi(n) * gamma
455        }
456    }
