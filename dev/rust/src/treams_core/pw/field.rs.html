17/// `exp(i k·r)` for a complex wavevector `k` and a real point `r`, from the real and
18/// imaginary parts of `k·r` separately.
19#[inline]
20pub(crate) fn phase(vector: [Complex; 3], point: [f64; 3]) -> Complex {
21    let angle = vector[0].re * point[0] + vector[1].re * point[1] + vector[2].re * point[2];
22    let attenuation = vector[0].im * point[0] + vector[1].im * point[1] + vector[2].im * point[2];
23    let (sine, cosine) = angle.sin_cos();
24    let scale = (-attenuation).exp();
25    Complex::new(scale * cosine, scale * sine)
26}

28/// What [`phases`] saves for its pullback: the points and the wavevectors, without the
29/// phases of every point and mode.
30#[derive(Debug)]
31pub struct PhasesResidual {
32    points: Vec<[f64; 3]>,
33    vectors: Vec<[Complex; 3]>,
34}
35/// Cotangents of real displacements and complex full wavevectors.
36#[derive(Debug)]
37pub struct PhasesGradient {
38    /// Real displacement cotangents.
39    pub points: Vec<[f64; 3]>,
40    /// Complex wavevector cotangents in the real Hermitian pairing.
41    pub vectors: Vec<[Complex; 3]>,
42}

44/// Direct scalar plane translation phase.
45///
46/// Upstream: `treams.pw.translate`.
47#[inline]
48pub fn translate(vector: [Complex; 3], point: [f64; 3]) -> Result<Complex> {
49    if vector.iter().any(|&v| !finite(v)) || point.iter().any(|v| !v.is_finite()) {
50        return Err(Error::InvalidInput(
51            "plane translation requires finite arguments".into(),
52        ));
53    }
54    let value = phase(vector, point);
55    if !finite(value) {
56        return Err(Error::NonFinite("non-finite plane translation".into()));
57    }
58    Ok(value)
59}

61/// Plane-wave translation phases with shape (displacements, wavevectors).
62///
63/// Upstream: `treams.pw.translate` on arrays of points and wavevectors.
64pub fn phases(
65    points: Vec<[f64; 3]>,
66    vectors: Vec<[Complex; 3]>,
67) -> Result<(DMatrix<Complex>, PhasesResidual)> {
68    if vectors.is_empty()
69        || vectors.iter().flatten().any(|&k| !finite(k))
70        || points.iter().flatten().any(|r| !r.is_finite())
71    {
72        return Err(Error::InvalidInput(
73            "require finite displacements and nonempty finite wavevectors".into(),
74        ));
75    }
76    let mut value = DMatrix::zeros(points.len(), vectors.len());
77    let parallel = value.len() >= PARALLEL_ENTRIES;
78    try_fill_chunks(
79        value.as_mut_slice(),
80        points.len().max(1),
81        parallel,
82        |j, column| -> Result<()> {
83            for (out, &point) in column.iter_mut().zip(&points) {
84                *out = phase(vectors[j], point);
85            }
86            Ok(())
87        },
88    )?;
89    if value.iter().any(|&v| !finite(v)) {
90        return Err(Error::NonFinite("plane translation phase overflow".into()));
91    }
92    Ok((value, PhasesResidual { points, vectors }))
93}
94impl PhasesResidual {
95    /// Number of displacements and wavevectors.
96    #[must_use]
97    pub fn shape(&self) -> (usize, usize) {
98        (self.points.len(), self.vectors.len())
99    }
100    /// Recompute each phase once and add its terms to the point and wavevector gradients.
101    ///
102    /// The longer axis is split into at most 32 blocks, fixed by the shape, that
103    /// run in parallel; each block sums its own rows (or columns) in index order and
104    /// a partial gradient of the shorter axis, and the partials are added in block
105    /// order. The result is deterministic and the scratch is at most 32 gradients
106    /// of the shorter axis.
107    pub fn pullback<S: nalgebra::Storage<Complex, nalgebra::Dyn, nalgebra::Dyn> + Sync>(
108        self,
109        cotangent: &nalgebra::Matrix<Complex, nalgebra::Dyn, nalgebra::Dyn, S>,
110    ) -> Result<PhasesGradient> {
111        if cotangent.shape() != self.shape() || cotangent.iter().any(|&v| !finite(v)) {
112            return Err(Error::InvalidInput("invalid plane-phase cotangent".into()));
113        }
114        let (rows, columns) = self.shape();
115        // Add the phase terms of points `i0..` and wavevectors `j0..` to gradients
116        // indexed from those offsets; each entry sums over the other axis in order.
117        let contract =
118            |i0: usize, points: &mut [[f64; 3]], j0: usize, vectors: &mut [[Complex; 3]]| {
119                for (j, (vector_gradient, &vector)) in
120                    vectors.iter_mut().zip(&self.vectors[j0..]).enumerate()
121                {
122                    for (i, (point_gradient, &point)) in
123                        points.iter_mut().zip(&self.points[i0..]).enumerate()
124                    {
125                        let factor = cotangent[(i0 + i, j0 + j)].conj()
126                            * Complex::i()
127                            * phase(vector, point);
128                        for axis in 0..3 {
129                            point_gradient[axis] += (factor * vector[axis]).re;
130                            vector_gradient[axis] += factor.conj() * point[axis];
131                        }
132                    }
133                }
134            };
135        let long = rows.max(columns);
136        let block = long.div_ceil(32).max(16);
137        let blocks = long.div_ceil(block);
138        let task = |b: usize| {
139            let (start, stop) = (b * block, ((b + 1) * block).min(long));
140            if rows >= columns {
141                let mut points = vec![[0.0; 3]; stop - start];
142                let mut vectors = vec![[Complex::default(); 3]; columns];
143                contract(start, &mut points, 0, &mut vectors);
144                (points, vectors)
145            } else {
146                let mut points = vec![[0.0; 3]; rows];
147                let mut vectors = vec![[Complex::default(); 3]; stop - start];
148                contract(0, &mut points, start, &mut vectors);
149                (points, vectors)
150            }
151        };
152        let partials = try_map(
153            blocks,
154            cotangent.len() >= PARALLEL_ENTRIES,
155            |b| -> Result<_> { Ok(task(b)) },
156        )?;
157        let mut gradient = PhasesGradient {
158            points: Vec::with_capacity(rows),
159            vectors: Vec::with_capacity(columns),
160        };
161        if rows >= columns {
162            gradient.vectors = vec![[Complex::default(); 3]; columns];
163            for (points, vectors) in partials {
164                gradient.points.extend(points);
165                for (target, partial) in gradient
166                    .vectors
167                    .iter_mut()
168                    .flatten()
169                    .zip(vectors.iter().flatten())
170                {
171                    *target += partial;
172                }
173            }
174        } else {
175            gradient.points = vec![[0.0; 3]; rows];
176            for (points, vectors) in partials {
177                gradient.vectors.extend(vectors);
178                for (target, partial) in gradient
179                    .points
180                    .iter_mut()
181                    .flatten()
182                    .zip(points.iter().flatten())
183                {
184                    *target += partial;
185                }
186            }
187        }
188        Ok(gradient)
189    }
190}
191/// What [`field`] saves for its pullback: its inputs, for weighted fields or for the
192/// matrix that samples every mode.
193///
194/// The pullback splits the longer of the point and mode axes into at most 32 blocks,
195/// fixed by the shape, as [`PhasesResidual::pullback`] does. Each block writes the
196/// gradients of its own points or modes and a partial sum of the gradients of the
197/// other axis, and the partial sums add in block order; a mode's polarization pulls
198/// back once, from its total over all points. The blocks run in parallel from 4096
199/// points times modes (`PARALLEL_ENTRIES`), and in the same order on the calling thread
200/// below, so the thread count does not change the result.
201#[derive(Debug)]
202pub struct FieldResidual {
203    vectors: Vec<[Complex; 3]>,
204    polarizations: Vec<u8>,
205    points: Vec<[f64; 3]>,
206    coefficients: Option<Vec<Complex>>,
207    helicity: bool,
208    fixed_vectors: bool,
209}
210/// Cotangents in the real Hermitian pairing.
211#[derive(Debug)]
212pub struct FieldGradient {
213    /// Amplitude cotangents; empty for a field operator.
214    pub coefficients: Vec<Complex>,
215    /// Real sample-point cotangents.
216    pub points: Vec<[f64; 3]>,
217    /// Complex full-wavevector cotangents; zero when wavevectors are held fixed.
218    pub vectors: Vec<[Complex; 3]>,
219}
220/// Evaluate weighted plane fields, or the full operator when coefficients are absent.
221///
222/// The rows hold the three Cartesian components of each point in turn. With
223/// `fixed_vectors`, the pullback holds the wavevectors fixed and returns zero wavevector
224/// gradients.
225///
226/// Upstream: sums of `treams.special.vpw_M`, `vpw_N` or `vpw_A` over the plane modes.
227pub fn field(
228    vectors: Vec<[Complex; 3]>,
229    polarizations: Vec<u8>,
230    points: Vec<[f64; 3]>,
231    coefficients: Option<Vec<Complex>>,
232    helicity: bool,
233    fixed_vectors: bool,
234) -> Result<(DMatrix<Complex>, FieldResidual)> {
235    if vectors.is_empty()
236        || vectors.len() != polarizations.len()
237        || points.iter().flatten().any(|r| !r.is_finite())
238        || coefficients
239            .as_ref()
240            .is_some_and(|c| c.len() != vectors.len() || c.iter().any(|&v| !finite(v)))
241    {
242        return Err(Error::InvalidInput(
243            "require nonempty plane modes, finite points, matching finite amplitudes and polarizations"
244                .into(),
245        ));
246    }
247    let electric: Vec<_> = vectors
248        .iter()
249        .zip(&polarizations)
250        .map(|(&k, &p)| polarization(k, p, helicity))
251        .collect::<Result<_>>()?;
252    let mut value = DMatrix::zeros(
253        3 * points.len(),
254        if coefficients.is_some() {
255            1
256        } else {
257            vectors.len()
258        },
259    );
260    let parallel = points.len() * vectors.len() >= PARALLEL_ENTRIES;
261    if let Some(c) = &coefficients {
262        try_fill_chunks(value.as_mut_slice(), 3, parallel, |p, out| -> Result<()> {
263            for ((&k, e), &c) in vectors.iter().zip(&electric).zip(c) {
264                let weight = c * phase(k, points[p]);
265                for (out, &e) in out.iter_mut().zip(e) {
266                    *out += weight * e;
267                }
268            }
269            Ok(())
270        })?;
271    } else if !points.is_empty() {
272        let size = 3 * points.len();
273        try_fill_chunks(
274            value.as_mut_slice(),
275            size,
276            parallel,
277            |j, column| -> Result<()> {
278                for (out, &point) in column.as_chunks_mut::<3>().0.iter_mut().zip(&points) {
279                    let phase = phase(vectors[j], point);
280                    for (out, &e) in out.iter_mut().zip(&electric[j]) {
281                        *out = phase * e;
282                    }
283                }
284                Ok(())
285            },
286        )?;
287    }
288    Ok((
289        value,
290        FieldResidual {
291            vectors,
292            polarizations,
293            points,
294            coefficients,
295            helicity,
296            fixed_vectors,
297        },
298    ))
299}
300impl FieldResidual {
301    /// Flattened output dimensions (3 * samples, modes or one weighted column).
302    #[must_use]
303    pub fn shape(&self) -> (usize, usize) {
304        (
305            3 * self.points.len(),
306            if self.coefficients.is_some() {
307                1
308            } else {
309                self.vectors.len()
310            },
311        )
312    }
313    /// Gradients for a `cotangent` of the field's shape. The pullback recomputes the
314    /// polarization and phase derivatives instead of storing a dense field Jacobian.
315    pub fn pullback(self, cotangent: &DMatrix<Complex>) -> Result<FieldGradient> {
316        if cotangent.shape() != self.shape() || cotangent.iter().any(|&v| !finite(v)) {
317            return Err(Error::InvalidInput("invalid plane-field cotangent".into()));
318        }
319        let (samples, modes) = (self.points.len(), self.vectors.len());
320        let mut gradient = FieldGradient {
321            coefficients: vec![Complex::default(); self.coefficients.as_ref().map_or(0, Vec::len)],
322            points: vec![[0.0; 3]; samples],
323            vectors: vec![[Complex::default(); 3]; modes],
324        };
325        if samples == 0 {
326            return Ok(gradient);
327        }
328        // The blocks of `PhasesResidual::pullback`; one block adds in the order of a
329        // plain loop over modes and points.
330        let parallel = samples * modes >= PARALLEL_ENTRIES;
331        let block = samples.max(modes).div_ceil(32).max(16);
332        if modes >= samples {
333            // Blocks of modes write the amplitude and wavevector gradients of their
334            // modes and add into partial point gradients. An operator has no amplitude
335            // gradients, so its blocks get none.
336            let mut coefficient_blocks = gradient.coefficients.chunks_mut(block);
337            let blocks: Vec<_> = gradient
338                .vectors
339                .chunks_mut(block)
340                .map(|vectors| (vectors, coefficient_blocks.next()))
341                .collect();
342            gradient.points = try_fold_ordered(
343                blocks,
344                parallel,
345                || vec![[0.0; 3]; samples],
346                |mut points, b, (vectors, mut coefficients)| -> Result<_> {
347                    for (offset, vector) in vectors.iter_mut().enumerate() {
348                        let j = b * block + offset;
349                        let mode = self.mode(j)?;
350                        let mut sums = ModeSums::default();
351                        let rows = self.rows(cotangent, j);
352                        self.contract(&mode, &self.points, rows, &mut points, &mut sums);
353                        let coefficient = coefficients.as_deref_mut().map(|c| &mut c[offset]);
354                        self.finish(&mode, &sums, coefficient, vector);
355                    }
356                    Ok(points)
357                },
358                |mut total, partial| {
359                    for (a, b) in total.iter_mut().flatten().zip(partial.iter().flatten()) {
360                        *a += b;
361                    }
362                    total
363                },
364            )?;
365        } else {
366            // Blocks of points write the gradients of their points and add into partial
367            // sums of every mode, which finish once all blocks are added.
368            let terms = (0..modes)
369                .map(|j| self.mode(j))
370                .collect::<Result<Vec<_>>>()?;
371            let blocks: Vec<_> = gradient.points.chunks_mut(block).collect();
372            let sums = try_fold_ordered(
373                blocks,
374                parallel,
375                || vec![ModeSums::default(); modes],
376                |mut sums, b, point_gradients| -> Result<_> {
377                    let range = b * block..b * block + point_gradients.len();
378                    let points = &self.points[range.clone()];
379                    for (j, (mode, sums)) in terms.iter().zip(&mut sums).enumerate() {
380                        let rows = &self.rows(cotangent, j)[range.clone()];
381                        self.contract(mode, points, rows, point_gradients, sums);
382                    }
383                    Ok(sums)
384                },
385                |mut total, partial| {
386                    for (total, partial) in total.iter_mut().zip(&partial) {
387                        total.add(partial);
388                    }
389                    total
390                },
391            )?;
392            let mut coefficients = gradient.coefficients.iter_mut();
393            for ((mode, sums), vector) in terms.iter().zip(&sums).zip(&mut gradient.vectors) {
394                self.finish(mode, sums, coefficients.next(), vector);
395            }
396        }
397        Ok(gradient)
398    }
399
400    /// The wavevector, polarization jet and amplitude of plane mode `j`.
401    fn mode(&self, j: usize) -> Result<ModeTerms> {
402        let vector = self.vectors[j];
403        let electric = if self.fixed_vectors {
404            polarization(vector, self.polarizations[j], self.helicity)?.map(Jet::<3>::constant)
405        } else {
406            polarization_jet::<3>(vector, self.polarizations[j], self.helicity)?
407        };
408        let coefficient = self
409            .coefficients
410            .as_ref()
411            .map_or(Complex::new(1.0, 0.0), |c| c[j]);
412        Ok(ModeTerms {
413            vector,
414            electric,
415            coefficient,
416        })
417    }
418
419    /// The three Cartesian cotangents of each sample, contiguous in the cotangent
420    /// column of mode `j`, or in the one column of a weighted field.
421    fn rows<'c>(&self, cotangent: &'c DMatrix<Complex>, j: usize) -> &'c [[Complex; 3]] {
422        let length = 3 * self.points.len();
423        let column = if self.coefficients.is_some() { 0 } else { j };
424        cotangent.as_slice()[column * length..][..length]
425            .as_chunks::<3>()
426            .0
427    }
428
429    /// Add the terms of `mode` at `points`, whose cotangents are `rows`, to the point
430    /// gradients `point_gradients` and to the mode's `sums`, in point order.
431    fn contract(
432        &self,
433        mode: &ModeTerms,
434        points: &[[f64; 3]],
435        rows: &[[Complex; 3]],
436        point_gradients: &mut [[f64; 3]],
437        sums: &mut ModeSums,
438    ) {
439        let (weighted, fixed) = (self.coefficients.is_some(), self.fixed_vectors);
440        let ModeTerms {
441            vector,
442            electric,
443            coefficient,
444        } = *mode;
445        // Accumulate through references into the sums, which the loop can keep in
446        // registers, in the same order as indexing them would.
447        let ModeSums {
448            coefficient: coefficient_gradient,
449            vector: vector_gradient,
450            polarization: polarization_cotangent,
451        } = sums;
452        for ((&point, rows), point_gradient) in points.iter().zip(rows).zip(point_gradients) {
453            let phase = phase(vector, point);
454            let weighted_phase = coefficient * phase;
455            let paired: Complex = (0..3).map(|a| rows[a].conj() * electric[a].value).sum();
456            if weighted {
457                *coefficient_gradient += (phase * paired).conj();
458            }
459            let paired_field = weighted_phase * paired * Complex::i();
460            for axis in 0..3 {
461                point_gradient[axis] += (paired_field * vector[axis]).re;
462                if !fixed {
463                    vector_gradient[axis] += (paired_field * point[axis]).conj();
464                    polarization_cotangent[axis] += rows[axis] * weighted_phase.conj();
465                }
466            }
467        }
468    }
469
470    /// Write the amplitude gradient `coefficient` (none for an operator) and the
471    /// wavevector gradient `vector` of `mode` from its sums over all points.
472    fn finish(
473        &self,
474        mode: &ModeTerms,
475        sums: &ModeSums,
476        coefficient: Option<&mut Complex>,
477        vector: &mut [Complex; 3],
478    ) {
479        if let Some(coefficient) = coefficient {
480            *coefficient = sums.coefficient;
481        }
482        *vector = sums.vector;
483        // The polarization is the same at every point: pull it back once per mode.
484        if !self.fixed_vectors {
485            for (axis, gradient) in vector.iter_mut().enumerate() {
486                *gradient += (0..3)
487                    .map(|a| sums.polarization[a] * mode.electric[a].derivative[axis].conj())
488                    .sum::<Complex>();
489            }
490        }
491    }
492}

494/// What the field pullback needs of one plane mode at every point.
495#[derive(Clone, Copy)]
496struct ModeTerms {
497    /// The complex wavevector.
498    vector: [Complex; 3],
499    /// The polarization, with its derivatives in the wavevector unless it is fixed.
500    electric: [Jet<3>; 3],
501    /// The amplitude; one for an operator.
502    coefficient: Complex,
503}
504
505/// Sums over points of one plane mode's amplitude, wavevector and polarization
506/// cotangents; the last two stay zero when the wavevectors are fixed.
507#[derive(Clone, Copy, Default)]
508struct ModeSums {
509    coefficient: Complex,
510    vector: [Complex; 3],
511    polarization: [Complex; 3],
512}
513
514impl ModeSums {
515    /// Add the sums over other points.
516    fn add(&mut self, other: &Self) {
517        self.coefficient += other.coefficient;
518        for (a, b) in self
519            .vector
520            .iter_mut()
521            .chain(&mut self.polarization)
522            .zip(other.vector.iter().chain(&other.polarization))
523        {
524            *a += b;
525        }
526    }
527}

538    /// Field pullbacks add their partial sums in blocks fixed by the shape: blocks of
539    /// modes and blocks of points, for weighted fields and operators, with and without
540    /// wavevector derivatives, give gradients that repeat bit for bit on every pool
541    /// size once the blocks run in parallel.
542    #[test]
543    fn pullbacks_do_not_depend_on_the_thread_count() {
544        for (samples, modes) in [(30_u32, 200_u32), (300, 20)] {
545            let vectors: Vec<[Complex; 3]> = (0..modes)
546                .map(|j| {
547                    let t = f64::from(j);
548                    let (theta, phi) = (1.55 + 1.3 * (0.61 * t).sin(), 2.3 * t);
549                    [
550                        Complex::new(1.2 * theta.sin() * phi.cos(), 0.0),
551                        Complex::new(1.2 * theta.sin() * phi.sin(), 0.0),
552                        Complex::new(1.2 * theta.cos(), 0.01),
553                    ]
554                })
555                .collect();
556            let polarizations: Vec<u8> = (0..modes).map(|j| u8::from(j % 2 == 0)).collect();
557            let points: Vec<[f64; 3]> = (0..samples)
558                .map(|i| {
559                    let t = f64::from(i);
560                    [(0.37 * t).sin(), (0.53 * t).cos(), 0.8 * (0.71 * t).sin()]
561                })
562                .collect();
563            assert!(points.len() * vectors.len() >= PARALLEL_ENTRIES);
564            for weighted in [true, false] {
565                let coefficients =
566                    weighted.then(|| patterned(vectors.len(), 1, 0.3).as_slice().to_vec());
567                let columns = if weighted { 1 } else { vectors.len() };
568                let g = patterned(3 * points.len(), columns, 0.8);
569                for fixed_vectors in [false, true] {
570                    assert_same_bits_on_pools(|| {
571                        let (_, residual) = field(
572                            vectors.clone(),
573                            polarizations.clone(),
574                            points.clone(),
575                            coefficients.clone(),
576                            true,
577                            fixed_vectors,
578                        )
579                        .unwrap();
580                        let g = residual.pullback(&g).unwrap();
581                        bits(&[&g.coefficients, &g.points, &g.vectors])
582                    });
583                }
584            }
585        }
586    }
